oe_rw_renaissance.1 = {
    type = country_event
    outcome = neutral # FIX 2026-11-17 P2：补 outcome（默认）
    category = international_organization_event
    title = oe_rw_renaissance.1.title
    desc = oe_rw_renaissance.1.desc
    image = "gfx/interface/illustrations/institutions/renaissance.dds"
    trigger = {
    	exists = international_organization:oe_byz_roman_world
        international_organization:oe_byz_roman_world = {
            has_variable = oe_rw_renaissance_started
            var:oe_rw_renaissance_stage = 0
            NOT = { has_variable = oe_rw_renaissance_finished }
        }
    }
    immediate = {
        international_organization:oe_byz_roman_world = {
            set_variable = { name = oe_rw_renaissance_stage value = 1 }
        }
    }
    option = {
        name = oe_rw_renaissance.1.a
        add_gold = { value = monthly_income_trade_and_tax multiply = -6 }
        add_research_progress = 5
        custom_tooltip = {
            text = OE_RW_RENAISSANCE_ADD_SCORE
            international_organization:oe_byz_roman_world = { change_variable = { name = oe_rw_renaissance_score add = 1 } }
        }
        ai_chance = { base = 70 }
    }
    option = {
        name = oe_rw_renaissance.1.b
        add_prestige = 5
        ai_chance = { base = 30 }
    }
    after = {
        international_organization:oe_byz_roman_world = {
            remove_variable = oe_rw_renaissance_event_pending
            set_variable = { name = oe_rw_renaissance_wait value = yes years = 1 }
        }
    }
}
