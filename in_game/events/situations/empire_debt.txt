empire_debt.100 = {#贬值货币
	type = country_event
	outcome = neutral # FIX 2026-11-17 P2：补 outcome（默认）
	category = situation_event
	image = "gfx/interface/illustrations/situation/treaty_of_tordesillas.dds"
	title = empire_debt.100.t
	desc = empire_debt.100.d
	immediate = {
	}
	option = {
		name = empire_debt.100.a
		change_gold_effect = { scale = 10 }
		add_inflation = 0.05
		custom_tooltip = {
		text = empire_debt.100.tt
		if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money
			}
			set_variable = oe_byz_empier_money_I_no
			set_variable = oe_byz_empier_money_bII_allow
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money
			grant_estate_privilege = estate_privilege:oe_byz_empier_money_bII
		}
				else_if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money_bII
			}
			remove_variable = oe_byz_empier_money_bII_allow
			set_variable = oe_byz_empier_money_bIII_allow
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money_bII
			grant_estate_privilege = estate_privilege:oe_byz_empier_money_bIII
		}
				else_if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money_bIII
			}
			remove_variable = oe_byz_empier_money_bIII_allow
			set_variable = oe_byz_empier_money_bIV_allow
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money_bIII
			grant_estate_privilege = estate_privilege:oe_byz_empier_money_bIV
		}
				else_if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money_bIV
			}
			remove_variable = oe_byz_empier_money_bIV_allow
			set_variable = oe_byz_ed_money_break
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money_bIV
			set_bankruptcy = yes
		}
			else_if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money_gI
			}
			remove_variable = oe_byz_empier_money_gI_allow
			remove_variable = oe_byz_empier_money_I_no
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money_gI
			grant_estate_privilege = estate_privilege:oe_byz_empier_money
		}
			else_if = {
			limit = {
				has_estate_privilege = estate_privilege:oe_byz_empier_money_gII
			}
			remove_variable = oe_byz_empier_money_gII_allow
			set_variable = oe_byz_empier_money_gI_allow
			revoke_estate_privilege = estate_privilege:oe_byz_empier_money_gII
			grant_estate_privilege = estate_privilege:oe_byz_empier_money_gI
			add_stability = -13
			add_government_power = -13
		}
	}
	}
}
