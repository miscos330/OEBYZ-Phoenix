# A distinct event namespace prevents legacy queued timers from shortening a repaired movement.
namespace = oe_identity_start_repair

oe_identity_start_repair.17 = {
    type = location_event
    orphan = yes
    title = empty_text # FIX 2026-11-17 P2：隐藏事件按原版写法补 title
    hidden = yes
    immediate = {
        culture:oe_armenian_greek_culture = {
            if = {
                limit = { NOT = { has_variable = oe_armenia_two_start_timer } }
            set_variable = oe_identity_start_expired
            every_movement_in_culture = {
                limit = { movement_type = movement_definition:oe_armenian_identity_movement }
                destroy_movement = yes
            }
            }
        }
    }
}
