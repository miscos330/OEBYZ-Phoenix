    option = {
        name = oe_exp_sahara.3.a
        trigger = {
            gold >= 30
        }
        custom_tooltip = oe_exp_sahara.3.a.tt
        add_gold = -30
        resume_expedition = scope:expedition
        scope:expedition = {
            change_variable = {
                name = oe_exp_guards
                add = 5
            }
            clamp_variable = {
                name = oe_exp_guards
                min = 0
                max = 40
            }
            pause_expedition = 30
        }
    }
    option = {
        name = oe_exp_sahara.3.b
        custom_tooltip = oe_exp_sahara.3.b.tt
        resume_expedition = scope:expedition
        scope:expedition = {
            if = {
                limit = {
                    expedition_leader = {
                        mil >= 50
                    }
                }
                pause_expedition = 30
            }
            else = {
                change_variable = {
                    name = oe_exp_guards
                    add = -3
                }
                change_variable = {
                    name = oe_exp_water
                    add = -5
                }
                clamp_variable = {
                    name = oe_exp_water
                    min = 0
                    max = 100
                }
                clamp_variable = {
                    name = oe_exp_guards
                    min = 0
                    max = 40
                }
                pause_expedition = 75
            }
        }
    }
}

oe_exp_sahara.5 = {
    type = country_event
    title = oe_exp_sahara.5.t
    desc = oe_exp_sahara.5.d
    outcome = neutral
    image = "gfx/interface/illustrations/events/oe_cross_the_sahara.dds"
    option = {
        name = oe_exp_sahara.5.a
        custom_tooltip = oe_exp_sahara.5.a.tt
        resume_expedition = scope:expedition
        scope:expedition = {
            set_variable = {
                name = oe_exp_gold_cargo
                value = 20
            }
            if = {
                limit = {
                    var:oe_exp_water >= 50
                    var:oe_exp_guards >= 30
                }
                set_variable = {
                    name = oe_exp_gold_cargo
                    value = 40
                }
                if = {
                    limit = {
                        has_variable = oe_sahara_bargain
                    }
                    set_variable = {
                        name = oe_exp_gold_cargo
                        value = 60
                    }
                }
            }
            pause_expedition = 60
        }
    }
    option = {
        name = oe_exp_sahara.5.b
        custom_tooltip = oe_exp_sahara.5.b.tt
        resume_expedition = scope:expedition
        scope:expedition = {
            set_variable = {
                name = oe_exp_gold_cargo
                value = 20
            }
            change_variable = {
                name = oe_exp_water
                add = 20
            }
            clamp_variable = {
                name = oe_exp_water
                min = 0
                max = 100
            }
            pause_expedition = 30
        }
    }
}
