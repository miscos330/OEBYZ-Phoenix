namespace = oe_hegemony_triumphs

oe_hegemony_triumphs.1 = {
    type = country_event
    title = oe_hegemony_triumphs.1.t
    desc = oe_hegemony_triumphs.1.d
    outcome = positive
    illustration_tags = { 10 = regular 10 = exterior }
    trigger = {
        tag = BYZ
        has_variable = oe_heg_triumph_africa_pending
        NOT = { has_variable = oe_heg_triumph_africa_done }
        exists = international_organization:oe_byz_empire_pillars
    }
    immediate = {
        if = {
            limit = { has_variable = oe_heg_triumph_africa_pending NOT = { has_variable = oe_heg_triumph_africa_done } }
            set_variable = oe_heg_triumph_africa_done
            oe_heg_triumph_count_africa = yes
        }
        event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate }
    }
    option = {
        name = oe_hegemony_triumphs.1.a
        if = {
            limit = {
                has_variable = oe_heg_triumph_africa_pending
                has_variable = oe_heg_triumph_africa_done
                NOT = { has_variable = oe_heg_triumph_africa_rewarded }
            }
            set_variable = oe_heg_triumph_africa_rewarded
            oe_heg_triumph_rewards = { count_variable = oe_heg_triumph_africa_locations }
            remove_variable = oe_heg_triumph_africa_pending
        }
    }
    on_trigger_fail = { remove_variable = oe_heg_triumph_africa_pending }
}
