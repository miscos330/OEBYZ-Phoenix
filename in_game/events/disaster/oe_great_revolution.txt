oe_great_revolution.4 = {
    type = country_event
    category = disaster_event
    title = {
        first_valid = {
            triggered_desc = { trigger = { has_variable = oe_gr_royal_house } desc = oe_great_revolution.4.t }
            triggered_desc = { desc = oe_great_revolution.4.t_fallback }
        }
    }
    desc = oe_great_revolution.4.d
    outcome = neutral
    image = "gfx/interface/illustrations/events/oe_gr_republican_assembly.dds"
    trigger = {
        tag = BYZ oe_gr_active = yes has_variable = oe_gr_succession_pending
        NOT = { has_variable = oe_gr_emperor_nominee }
        NOT = { has_variable = oe_gr_royal_line_extinct }
    }
    immediate = {
        var:oe_gr_late_emperor = { save_scope_as = oe_gr_late_emperor }
        if = { limit = { has_variable = oe_gr_royal_house } var:oe_gr_royal_house = { save_scope_as = oe_gr_extinct_house } }
    }
    option = {
        name = oe_great_revolution.4.a
        custom_tooltip = oe_gr_extinction_tt
        hidden_effect = {
            if = {
                limit = { NOT = { has_variable = oe_gr_royal_line_extinct } }
                set_variable = oe_gr_royal_line_extinct
                remove_variable = oe_gr_succession_pending
                remove_variable = oe_gr_emperor
                remove_variable = oe_gr_detained_emperor
                change_variable = { name = oe_gr_muster subtract = 10 }
                clamp_variable = { name = oe_gr_muster min = 0 max = 100 }
            }
        }
    }
}
