oe_fate_of_the_phoenix.21 = {
	type = country_event
	outcome = neutral # FIX 2026-11-17 P2：补 outcome（默认）
	category = disaster_event
	title = oe_fate_of_the_phoenix.21.t
	desc = oe_fate_of_the_phoenix.21.d

	fire_only_once = yes

	trigger = {
	}

	immediate = {
		c:BYZ ?= {
			if = {
				limit = { has_ruler = yes }
				ruler ?= {
					save_scope_as = target_character
				}
			}
			else = {
				ruler_or_regent ?= {
					save_scope_as = target_character
				}
			}
		}
	}

	image = "gfx/interface/illustrations/disaster/fate_of_the_phoenix.dds"

	option = {	
		name = oe_fate_of_the_phoenix.21.a
		custom_tooltip = {
			text = oe_fate_of_the_phoenix.21.tt
			trigger_event_non_silently = oe_fate_of_the_phoenix.22
		}
		
	}
}

oe_fate_of_the_phoenix.200 = {
    type = country_event
    outcome = neutral # FIX 2026-11-17 P2：补 outcome（默认）
    category = disaster_event
    title = oe_fate_of_the_phoenix.200.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { scope:oe_raid_defended = 1 } desc = OE_RAID_TURK_DEFENDED_DESC }
            triggered_desc = { desc = oe_fate_of_the_phoenix.200.d }
        }
    }
    image = "gfx/interface/illustrations/disaster/fate_of_the_phoenix.dds"
    trigger = { oe_raid_turk_available = yes }
    immediate = {
        random_neighbor_country = {
            limit = { oe_raid_turk_attacker = yes }
            save_scope_as = oe_raid_attacker
            ruler_or_regent ?= { save_scope_as = target_character }
        }
        random_owned_location = {
            limit = { oe_raid_rural_target = yes is_neighbor_of = scope:oe_raid_attacker }
            save_scope_as = target_location
        }
        oe_raid_roll_defense = yes
    }
    option = {
        name = oe_fate_of_the_phoenix.200.a
        ai_chance = { base = 45 }
        trigger = { oe_raid_can_resolve = yes NOT = { scope:oe_raid_defended = 1 } }
        oe_raid_accept_effect = yes
    }
    option = {
        name = oe_fate_of_the_phoenix.200.b
        ai_chance = { base = 25 }
        trigger = { oe_raid_can_resolve = yes NOT = { scope:oe_raid_defended = 1 } manpower >= oe_raid_patrol_manpower }
        oe_raid_patrol_effect = yes
    }
    option = {
        name = oe_fate_of_the_phoenix.200.c
        ai_chance = { base = 20 }
        exclusive = yes
        trigger = { oe_raid_can_resolve = yes scope:oe_raid_defended = 1 }
        oe_raid_defended_effect = yes
    }
    option = {
        name = OE_RAID_NO_LONGER_RELEVANT
        ai_chance = { base = 10 }
        trigger = { NOT = { oe_raid_can_resolve = yes } }
        custom_tooltip = OE_RAID_NO_LONGER_RELEVANT_TT
    }
}
