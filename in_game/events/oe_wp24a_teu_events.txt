oe_wp24a_teu.1 = {
	type = country_event
	title = oe_wp24a_teu.1.t
	desc = oe_wp24a_teu.1.d
	outcome = neutral
	image = "gfx/interface/illustrations/situation/reformation.dds"
	trigger = {
		oe_wp24a_is_teu_host = yes
		oe_wp24a_is_teu_reformation_era = yes
		NOT = { has_variable = oe_wp24a_teu_branch_catholic }
	}
	immediate = {
		set_variable = { name = oe_wp24a_teu_branch_secular value = 1 }
		set_variable = { name = oe_teu_1525_choice_made value = 1 }
		remove_country_modifier = oe_wp24a_teu_marian_oath_mod
		remove_country_modifier = oe_wp24a_teu_papal_volunteers_mod
		remove_country_modifier = oe_wp24a_teu_ordensburg_bastions_mod
		remove_country_modifier = oe_wp24a_teu_tridentine_knights_mod
	}
	option = {
		name = oe_wp24a_teu.1.a
		historical_option = yes
		ai_chance = { base = 70 }
		if = {
			limit = { NOT = { has_variable = oe_teu_secularized_pru } }
			set_variable = { name = oe_teu_secularized_pru value = 1 }
			set_variable = { name = oe_teu_ordensstaat_active value = 1 }
			if = {
				limit = { has_estate_privilege = estate_privilege:clergy_military_orders }
				revoke_estate_privilege = estate_privilege:clergy_military_orders
			}
			if = {
				limit = { has_reform = government_reform:military_order_reform }
				remove_reform = government_reform:military_order_reform
			}
			if = {
				limit = { religion:lutheran = { is_religion_enabled = yes } }
				change_religion = religion:lutheran
			}
			change_government_type = government_type:monarchy
			if = {
				limit = { NOT = { tag = PRU } NOT = { country_exists = c:PRU } }
				form_country = formable_country:PRU_f
			}
		}
		if = {
			limit = { country_exists = c:PAP }
			c:PAP = { add_opinion = { target = root modifier = oe_wp24a_op_prussian_secular_rift } }
		}
		add_stability = 4
		add_country_modifier = { modifier = oe_wp24a_teu_prussian_duchy_mod years = 35 mode = replace }
		hidden_effect = {
			# [v70_mchain]
			set_variable = { name = oe_v70_mch_wp24a_teu_c01_s1 value = 1 }
			trigger_event_non_silently = { id = oe_wp24a_teu.2 days = { 365 730 } }
		}
	}
	option = {
		name = oe_wp24a_teu.1.b
		ai_chance = { base = 30 }
		set_variable = { name = oe_teu_secularized_pru value = 1 }
		set_variable = { name = oe_teu_ordensstaat_active value = 1 }
		if = {
			limit = { has_estate_privilege = estate_privilege:clergy_military_orders }
			revoke_estate_privilege = estate_privilege:clergy_military_orders
		}
		if = {
			limit = { has_reform = government_reform:military_order_reform }
			remove_reform = government_reform:military_order_reform
		}
		change_government_type = government_type:monarchy
		add_gold = 120
		add_country_modifier = { modifier = oe_wp24a_teu_prussian_duchy_mod years = 25 mode = replace }
		hidden_effect = {
			# [v70_mchain]
			set_variable = { name = oe_v70_mch_wp24a_teu_c01_s1 value = 1 }
			trigger_event_non_silently = { id = oe_wp24a_teu.2 days = { 365 730 } }
		}
	}
}
