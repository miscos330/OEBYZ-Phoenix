# 1. Core Imperial Theme Escheatment Event (overlord decides fate of heirless subject theme/vassal)
oe_wp47_escheat.1 = {
	type = country_event
	title = oe_wp47_escheat.1.t
	desc = oe_wp47_escheat.1.d
	outcome = neutral

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		is_rebel_country = no
		oe_wp47_is_escheat_overlord = yes
		exists = scope:oe_wp47_escheat_subject
		scope:oe_wp47_escheat_subject = {
			is_rebel_country = no
			is_subject_of = root
			num_locations > 0
		}
	}

	immediate = {
		event_illustration_government_estate_effect = yes
		oe_wp47_init_vars = yes
		set_variable = { name = oe_wp47_ev_cd value = 1 days = 75 }
		if = {
			limit = { has_variable = oe_wp47_escheat_count }
			change_variable = { name = oe_wp47_escheat_count add = 1 }
		}
		else = {
			set_variable = { name = oe_wp47_escheat_count value = 1 }
		}
	}

	# Option A: Reclaim Theme to Imperial Demesne
	option = {
		name = oe_wp47_escheat.1.a
		ai_chance = {
			base = 45
		}
		custom_tooltip = oe_wp47_escheat.1.a.tt
		set_variable = { name = oe_wp47_escheat_path value = 1 }
		annex_country = {
			country = scope:oe_wp47_escheat_subject
			reason = Diplomatic
			transfer = yes
		}
		add_stability = -4
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			add_estate_satisfaction = {
				type = estate_type:nobles_estate
				value = -0.08
			}
		}
		add_country_modifier = {
			modifier = oe_wp47_mod_reclaimed_theme
			years = 8
			mode = replace
		}
		capital ?= {
			if = {
				limit = { has_owner = yes owner ?= root }
				change_control = 0.04
				change_prosperity = 0.02
			}
		}
		every_subject = {
			limit = { oe_wp47_is_eligible_subject = yes }
			add_opinion = {
				target = root
				modifier = oe_wp47_bias_escheat_seizure
			}
		}
		trigger_event_non_silently = {
			id = oe_wp47_escheat.5
			days = 180
		}
	}

	# Option B: Appoint Loyal Imperial Strategos
	option = {
		name = oe_wp47_escheat.1.b
		ai_chance = {
			base = 35
		}
		custom_tooltip = oe_wp47_escheat.1.b.tt
		set_variable = { name = oe_wp47_escheat_path value = 2 }
		add_prestige = 8
		add_legitimacy = 5
		ruler ?= { add_mil = 2 }
		create_character = {
			age = 38
			adm = { 56 74 }
			dip = { 54 72 }
			mil = { 72 88 }
			estate = estate_type:nobles_estate
			culture = root.culture
			religion = root.religion
		}
		add_country_modifier = {
			modifier = oe_wp47_mod_overlord_strategos_patronage
			years = 10
			mode = replace
		}
		scope:oe_wp47_escheat_subject = {
			add_liberty_desire = liberty_desire_severe_minus
			add_opinion = {
				target = root
				modifier = oe_wp47_bias_loyal_strategos
			}
			add_country_modifier = {
				modifier = oe_wp47_mod_loyal_strategos
				years = 15
				mode = replace
			}
		}
		trigger_event_non_silently = {
			id = oe_wp47_escheat.2
			days = 180
		}
	}

	# Option C: Confirm Local Aristocratic Succession
	option = {
		name = oe_wp47_escheat.1.c
		ai_chance = {
			base = 20
		}
		custom_tooltip = oe_wp47_escheat.1.c.tt
		set_variable = { name = oe_wp47_escheat_path value = 3 }
		add_gold = 120
		ruler ?= { add_dip = 2 }
		if = {
			limit = { country_has_estate = estate_type:nobles_estate }
			add_estate_satisfaction = {
				type = estate_type:nobles_estate
				value = 0.08
			}
		}
		add_country_modifier = {
			modifier = oe_wp47_mod_aristocratic_charter
			years = 10
			mode = replace
		}
		every_subject = {
			limit = { oe_wp47_is_eligible_subject = yes }
			add_liberty_desire = liberty_desire_weak_minus
			add_opinion = {
				target = root
				modifier = oe_wp47_bias_confirmed_succession
			}
		}
		trigger_event_non_silently = {
			id = oe_wp47_escheat.4
			days = 180
		}
	}
}
