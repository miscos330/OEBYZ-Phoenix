		name = oe_byz_mission.5.a

			limit = {

			set_variable = oe_byz_mission_3_a_on
		}
		if = {
			limit = {
				has_variable = oe_byz_mission_4_b_on
			}
			set_variable = oe_byz_mission_3_b_on
		}
	}
	option = {
		name = oe_byz_mission.5.b
		ai_chance = { base = 40 }
		if = {
			limit = {
				has_variable = oe_byz_mission_4_a_on
			}
			set_variable = oe_byz_mission_3_b_on
		}
		if = {
			limit = {
				has_variable = oe_byz_mission_4_b_on
			}
			set_variable = oe_byz_mission_3_c_on
		}
		add_gold = {
			value = -5
			multiply = c:EPI.country_tax_base
		}
		c:EPI = {
		add_gold_to_estate = {
			estate_type = estate_type:nobles_estate
			value = {
				value = 5
			multiply = c:EPI.country_tax_base
			}
		}			
		}
	}
