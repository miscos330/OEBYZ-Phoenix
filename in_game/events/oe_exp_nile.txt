oe_exp_nile.5 = {
    type = country_event
    title = oe_exp_nile.5.t
    desc = oe_exp_nile.5.d
    outcome = neutral
    image = "gfx/interface/illustrations/events/oe_lower_nile_exploration.dds"
    option = {
        name = oe_exp_nile.5.a
        custom_tooltip = oe_exp_nile.5.a.tt
        if = {
            limit = {
                NOT = {
                    has_global_variable = oe_ark_recovered
                }
                NOT = {
                    exists = work_of_art:oe_ark_of_the_covenant
                }
            }
            random = {
                chance = {
                    value = 5
                    if = {
                        limit = {
                            scope:expedition = {
                                var:oe_exp_clues >= 60
                            }
                        }
                        add = 10
                    }
                    if = {
                        limit = {
                            scope:expedition = {
                                var:oe_exp_clues >= 80
                            }
                        }
                        add = 10
                    }
                }
                scope:expedition = {
                    set_variable = {
                        name = oe_nile_ark_found
                        value = 1
                    }
                }
            }
        }
        resume_expedition = scope:expedition
        scope:expedition = {
            if = {
                limit = {
                    var:oe_exp_provisions < 25
                }
                pause_expedition = 150
            }
            else = {
                pause_expedition = 90
            }
        }
    }
    option = {
        name = oe_exp_nile.5.b
        custom_tooltip = oe_exp_nile.5.b.tt
        resume_expedition = scope:expedition
        scope:expedition = {
            pause_expedition = 30
        }
    }
}
