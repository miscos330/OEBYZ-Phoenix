# 1. stratiotai
oe_levy.1 = {
	type = country_event
	title = oe_levy.1.title
	desc = oe_levy.1.desc
	outcome = positive
	image = "gfx/interface/illustrations/events/oe_ev_levy_eurasian_hosts.dds"
	trigger = { is_rebel_country = no }
	option = {
		name = oe_levy.1.a
		ai_chance = { base = 50 }
		add_country_modifier = { modifier = oe_levy_muster_mod years = 10 mode = replace }
		add_prestige = prestige_weak_bonus
	}
	option = {
		name = oe_levy.1.b
		ai_chance = { base = 30 }
		add_army_tradition = army_tradition_weak_bonus
		add_country_modifier = { modifier = oe_m2e_levy_elite_mod years = 10 mode = replace }
	}
	option = {
		name = oe_levy.1.c
		ai_chance = { base = 20 }
		if = { limit = { country_has_estate = estate_type:peasants_estate } add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus } }
		add_country_modifier = { modifier = oe_m2e_levy_charter_mod years = 10 mode = replace }
	}
}
