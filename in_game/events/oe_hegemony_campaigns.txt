namespace = oe_hegemony_campaigns
# Bounded watchdog for active expedition countries only; no global daily country scan.
oe_hegemony_campaigns.1 = {
    type = country_event
    title = empty_text # FIX 2026-11-17 P2：隐藏事件按原版写法补 title
    hidden = yes
    interface_lock = no
    trigger = { has_reform = government_reform:oe_heg_expedition_reform }
    immediate = {
        if = {
            limit = { at_war = no NOT = { has_variable = oe_heg_expedition_launching } }
            oe_heg_finish_expedition = yes
        }
        else = {
            set_variable = { name = oe_heg_expedition_watch_alive value = yes days = 3 }
            trigger_event_silently = { id = oe_hegemony_campaigns.1 days = 1 }
        }
    }
}
