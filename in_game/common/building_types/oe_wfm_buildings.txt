			is_rebel_country = no
			NOT = { oe_is_fourth_rome_exile_country = yes }
		}
	}

	on_built = {
		if = {
			limit = {
				NOT = { has_global_variable = oe_wfm_bld_angkor_wat_built }
				location = { has_owner = yes }
			}
			set_global_variable = { name = oe_wfm_bld_angkor_wat_built value = 1 }
			location.owner = {
				add_prestige = prestige_mild_bonus
				culture = { add_cultural_tradition = 4 }
			}
		}
	}

		temple_maintenance
	}

	modifier = {
		local_cultural_tradition = 0.18
		local_cultural_influence = 0.16
		local_monthly_food_modifier = 0.1
	}
