pe_oe_end_timur = {
    cost = { value = 100 }
    base_antagonism = 50
    antagonism_type = antagonism_peace_treaty_demands
    potential = {
        oe_historical_timur_alive = yes
        scope:winner = { tag = BYZ }
        scope:loser = { tag = TIM }
        scope:war = {
            is_no_cb_war = no
            casus_belli ?= casus_belli:oe_cb_confront_timur
            is_an_attacker = scope:winner
            original_defender_leader ?= scope:loser
        }
    }
    allow = {
        oe_historical_timur_alive = yes
        scope:loser = { any_owned_location = { oe_timur_treaty_location = yes } }
    }
    effect = {
        custom_tooltip = {
            text = OE_END_TIMUR_EFFECT
            scope:loser = { every_owned_location = { limit = { oe_timur_treaty_location = yes } add_to_list = oe_timur_ceded_locations } }
            every_in_list = {
                list = oe_timur_ceded_locations
                change_location_owner = scope:winner
                change_location_controller = scope:winner
                # This treaty explicitly grants conquered land without Byzantine cores.
                oe_reset_new_conquest_integration = { recipient = scope:winner }
            }
        }
    }
    ai_desire = { value = 100 }
}

                        limit = { region = region:caucasus_region }
