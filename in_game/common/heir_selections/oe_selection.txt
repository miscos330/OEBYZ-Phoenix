oe_sub_mam_elect = {
	include_ruler_siblings = yes
	all_in_country = yes
	through_female = yes
	allow_female = yes
	ignore_ruler = yes
	cached = yes
	allow_foreign_ruler = no

	allowed_estates = { nobles_estate }
	heir_is_allowed = {
	}
	potential = {
		has_reform = government_reform:oe_sub_mam_system_subject
	}
	locked = {
		has_reform = government_reform:oe_sub_mam_system_subject
	}
	calc = {
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
				multiply = 0.3
			}
			add = {
				desc = "HEIR_SELECTION_TOTAL_ABILITIES"
				value = total_abilities
				multiply = 0.1
			}
			if = { 
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = -1000 
				}
			}
			if = { 
				limit = { has_character_modifier = oe_byz_sub_mam_support_mod }
				add = {
					desc = "oe_sub_religion_empire_support_text"
					value = 100 
				}
			}
		}
		# Here it goes the description for explaining a heir gets +1 points per year.
		if = {
			limit = {
				not = { exists = root }
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 0.3
			}
			add = {
				desc = "HEIR_SELECTION_TOTAL_ABILITIES"
				value = 0.1
			}
				add = {
					desc = "oe_sub_religion_empire_support_text"
					value = 100
				}		
		}
	}
}

oe_byz_elect = {#拜占庭帝国继承法
	all_in_country = yes

	include_ruler_siblings = yes
	through_female = yes
	allow_children = yes
	allow_female = yes
	ignore_ruler = yes
	allow_foreign_ruler = yes
	include_other_countries = { oe_theme_member_trigger = yes }


	cached = yes
	term_duration = 0
	show_candidates = yes
	max_possible_candidates = 6

	allowed_estates = {} # Eligibility below preserves the estate gate except for imperial titles/adoption.
	heir_is_allowed = {
        oe_byz_honor_recipient = yes
        OR = {
            has_estate = estate_type:crown_estate
            has_estate = estate_type:nobles_estate
            has_character_modifier = oe_byz_royal_adopted_child
            has_character_modifier = oe_byz_caesar_modifier
        }
    }
	potential = {
		has_or_had_tag = BYZ
	}
	locked = {
		has_or_had_tag = BYZ
	}
	calc = {
		if = {
			limit = {
				exists = root
			}
            if = {
                limit = { has_character_modifier = oe_byz_royal_adopted_child }
                add = { desc = "OE_HEIR_SELECTION_ADOPTED_CHILD" value = 100 }
            }
            if = {
                limit = { has_character_modifier = oe_byz_caesar_modifier }
                add = { desc = "OE_HEIR_SELECTION_CAESAR" value = 50 }
            }

            # Use the native character modifier flag, not a nonexistent trait ID.
            if = {
                limit = { modifier:byz_born_in_the_purple = yes }
                add = { desc = "HEIR_SELECTION_BORN_IN_THE_PURPLE_DESC" value = 30 }
            }

			if = {
				limit = {
					is_female = yes
					total_abilities <= 200
				}
            add = {
                desc = "oe_byz_elect_text_II"
                value = -100              				
			}
		}
			if = {
				limit = {
					has_trait = oe_character_war_score_trait_VI
				}
            add = {
                desc = "oe_byz_elect_text_VIII"
                value = 100   				
			}
		}
			if = {
				limit = {
					has_character_modifier = oe_character_war_score_modifier_II
				}
            add = {
                desc = "oe_byz_elect_text_IX"
                value = 10      				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_war_score_modifier_III
				}
            add = {
                desc = "oe_byz_elect_text_IX"
                value = 180   				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_war_score_modifier_IV
				}
            add = {
                desc = "oe_byz_elect_text_IX"
                value = 240     				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_war_score_modifier_V
				}
            add = {
                desc = "oe_byz_elect_text_IX"
                value = 300     				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_cabinet_experience_modifier_III
				}
            add = {
                desc = "oe_byz_elect_text_XIV"
                value = 120 				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_cabinet_experience_modifier_IV
				}
            add = {
                desc = "oe_byz_elect_text_XIV"
                value = 160     				
			}
		}
					if = {
				limit = {
					has_character_modifier = oe_character_cabinet_experience_modifier_V
				}
            add = {
                desc = "oe_byz_elect_text_XIV"
                value = 200    				
			}
		}
			if = {
				limit = {
					OR = {
						has_character_modifier = oe_rr_put_university_fir_mod
						has_character_modifier = oe_rr_put_university_wea_mod
						has_character_modifier = oe_rr_put_university_can_mod
						has_character_modifier = oe_rr_put_university_pap_mod
						has_character_modifier = oe_rr_put_university_boo_mod
						has_character_modifier = oe_rr_put_university_tin_mod
						has_character_modifier = oe_rr_put_university_med_mod
						has_character_modifier = oe_rr_put_university_por_mod
						has_character_modifier = oe_rr_put_university_gla_mod
						has_character_modifier = oe_rr_put_university_dye_mod
					}
				}
            add = {
                desc = "oe_byz_elect_text_XII"
                value = 20  				
			}
		}
			if = {
				limit = {
					OR = {
						AND = {
					has_estate = estate_type:nobles_estate
					NOT = { has_character_modifier = oe_byz_select_as_second_empire_mod }
						}
						AND = {
							is_ruler = yes
							NOT = {
								owner = c:BYZ
							}
						}
					}
				}
            add = {
                desc = "oe_byz_elect_text_VI"
                value = {
					add = {
						value = -100
					}
					add = {
						value = 100
						multiply = "c:BYZ.estate_power(estate_type:nobles_estate)"
					}
				}
			}
		}
			if = {
				limit = {
					OR = {
						NOT = {
							root.religion = religion:orthodox
						}
						NOT = {
							root.culture = {
								has_culture_group = culture_group:greek_group
							}
						}
					}
				}
            add = {
                desc = "oe_byz_elect_text_IV"
                value = -1000             				
			}
		}
			if = {
				limit = {
					exists = scope:target.ruler # FIX 2026-11-17: a theme can be left without a ruler (revolt, recall)
					OR = {
				        father ?= { this = scope:target.ruler }    
                        mother ?= { this = scope:target.ruler }    
					}
				}
            add = {
                desc = "oe_byz_elect_text_III"
                value = 100           				
			}
		}
					if = {
				limit = {
					has_estate = estate_type:crown_estate
					exists = c:BYZ.ruler.dynasty # FIX 2026-11-18: no ruler during a regency (r5 log: 'Invalid right side ... dynasty')
					dynasty ?= c:BYZ.ruler.dynasty											
				}
            add = {
                desc = "oe_byz_elect_text_VII"
                value = 50        				
			}
		}
		if = {
			limit = {
							scope:target = {
								has_regent = yes
							}		
							is_ruler = yes	
							owner = c:BYZ														
			}
            add = {
                desc = "oe_byz_elect_text_XIII"
                value = 200      				
			}
		}
		if = {
				limit = {
					OR = {
						is_general = yes
						is_admiral = yes
						in_cabinet = yes
					}
				}
            add = {
                desc = "oe_byz_elect_text_V"
                value = 33            				
			}			
		}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
				multiply = 1
			}
			add = {
				desc = "HEIR_SELECTION_TOTAL_ABILITIES"
				value = total_abilities
				multiply = 0.5
			}
			if = { 
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = -1000 
				}
			}
			if = { 
				limit = { has_character_modifier = oe_byz_select_as_second_empire_mod }
				add = {
					desc = "oe_byz_elect_text_I"
					value = 200 
				}
			}
		}
		# Here it goes the description for explaining a heir gets +1 points per year.
		if = {
			limit = {
				not = { exists = root }
			}
            add = { desc = "OE_HEIR_SELECTION_ADOPTED_CHILD" value = 100 }
            add = { desc = "OE_HEIR_SELECTION_CAESAR" value = 50 }

            add = { desc = "HEIR_SELECTION_BORN_IN_THE_PURPLE_DESC" value = 30 }

			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_TOTAL_ABILITIES"
				value = 0.5
			}
            add = {
                desc = "oe_byz_elect_text_V"
                value = 33            				
			}	
            add = {
                desc = "oe_byz_elect_text_XII"
                value = 20  				
			}
			add = {
					desc = "oe_byz_elect_text_I"
					value = 200
			}
            add = {
                desc = "oe_byz_elect_text_III"
                value = 100            				
			}
			add = {
                desc = "oe_byz_elect_text_VII"
                value = 50        				
			}
            add = {
                desc = "oe_byz_elect_text_VIII"
                value = 100         				
			}
            add = {
                desc = "oe_byz_elect_text_IX"
                value = 60   				
			}
            add = {
                desc = "oe_byz_elect_text_XIV"
                value = 40				
			}
            add = {
                desc = "oe_byz_elect_text_XIII"
                value = 200      				
			}
			add = {
                desc = "oe_byz_elect_text_II"
                value = -100              				
			}
            add = {
                desc = "oe_byz_elect_text_VI"
                value = {
					add = {
						value = -100
					}
					add = {
						value = 100
						multiply = "c:BYZ.estate_power(estate_type:nobles_estate)"
					}
				}
			}
            add = {
                desc = "oe_byz_elect_text_IV"
                value = -1000             				
			}
		}
	}
