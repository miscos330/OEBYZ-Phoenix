oe_cross_the_sahara = {
    unique = yes
    repeatable = yes
    dynamic_first_waypoint = yes
    origin = none
    travel_mode = land
    travel_speed = 0.8
    # Return is explicitly part of the finite route, including a final home waypoint.
    should_end_at_last_waypoint = yes
    fail_if_no_leader = yes
    category = commercial
    icon = category_commercial
    show_start_message = no
    show_end_message = no
    variables = {
        oe_exp_water = {
            start = 100
            min = 0
            max = 100
            display = piechart
            icon = "gfx/interface/icons/unit_view/food.dds"
        }
        oe_exp_guards = {
            start = 40
            min = 0
            max = 40
            display = fraction
            icon = "gfx/interface/icons/expedition_types/guards.dds"
        }
        oe_exp_gold_cargo = {
            start = 0
            min = 0
            max = 60
            display = fraction
            icon = "gfx/interface/icons/trade_goods/icon_goods_goods_gold.dds"
        }
        oe_exp_elapsed = {
            start = 0
            min = 0
            max = 120
            display = fraction
            icon = "gfx/interface/icons/expedition_types/experience.dds"
            hidden = yes
        }
    }
    potential = {
        has_or_had_tag = BYZ
        oe_exp_early_ages = yes
    }
    can_start = {
        at_war = no
        gold >= 300
        oe_exp_early_ages = yes
        owns = location:tripoli
        NOT = {
            has_cooldown = oe_cross_the_sahara_cooldown
        }
    }
    leader = {
        is_adult = yes
        is_ruler = no
        is_expedition_leader = no
        has_exploration = no
    }
    on_start = {
        add_gold = -300
        add_cooldown = {
            type = oe_cross_the_sahara_cooldown
            years = 10
        }
        scope:expedition = {
            if = {
                limit = {
                    exists = expedition_leader
                }
                remove_commander = expedition_leader
            }
            add_new_waypoint = location:tripoli
            add_new_waypoint = location:ghadames
            add_new_waypoint = location:ghat
            add_new_waypoint = location:agadez
            add_new_waypoint = location:timbuktu
            add_new_waypoint = location:agadez
            add_new_waypoint = location:ghat
            add_new_waypoint = location:ghadames
            add_new_waypoint = location:tripoli
        }
        trigger_event_non_silently = oe_exp_sahara.1
    }
    on_monthly = {
        # Only expedition-local scalars. No location or population loops.
        if = {
            limit = {
                scope:expedition = {
                    NOT = {
                        expedition_is_stalled = yes
                    }
                }
            }
            scope:expedition = {
                change_variable = {
                    name = oe_exp_elapsed
                    add = 1
                }
                if = {
                    limit = {
                        expedition_is_traveling = yes
                        NOT = {
                            expedition_is_paused = yes
                        }
                    }
                    change_variable = {
                        name = oe_exp_water
                        subtract = 1
                    }
                    clamp_variable = {
                        name = oe_exp_water
                        min = 0
                        max = 100
                    }
                }
            }
            if = {
                limit = {
                    scope:expedition = {
                        var:oe_exp_elapsed >= 120
                    }
                }
                fail_expedition = scope:expedition
            }
        }
    }
    on_arrive_to_new_location = {
        scope:location = {
            discover_location = root
        }
    }
    on_arrive_to_waypoint = {
        if = {
            limit = {
                scope:location = location:ghadames
                NOT = {
                    scope:expedition = {
                        has_variable = oe_sahara_visited_ghadames
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_sahara_visited_ghadames
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_sahara.2
        }
        if = {
            limit = {
                scope:location = location:ghat
                NOT = {
                    scope:expedition = {
                        has_variable = oe_sahara_visited_ghat
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_sahara_visited_ghat
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_sahara.3
        }
        if = {
            limit = {
                scope:location = location:agadez
                NOT = {
                    scope:expedition = {
                        has_variable = oe_sahara_visited_agadez
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_sahara_visited_agadez
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_sahara.4
        }
        if = {
            limit = {
                scope:location = location:timbuktu
                NOT = {
                    scope:expedition = {
                        has_variable = oe_sahara_visited_timbuktu
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_sahara_visited_timbuktu
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_sahara.5
        }
        if = {
            limit = {
                scope:location = location:ghat
                scope:expedition = {
                    has_variable = oe_sahara_visited_timbuktu
                    NOT = {
                        has_variable = oe_sahara_home_event
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_sahara_home_event
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_sahara.6
        }
    }
    on_end = {
        set_variable = {
            name = oe_exp_last_sahara_gold
            value = scope:expedition.var:oe_exp_gold_cargo
        }
        location:constantinople.market = {
            add_goods_supply = {
                goods = goods:goods_gold
                amount = scope:expedition.var:oe_exp_gold_cargo
            }
        }
        add_prestige = 1
        trigger_event_non_silently = oe_exp_sahara.7
    }
    on_fail = {
        trigger_event_non_silently = oe_exp_sahara.8
    }
    # Player-authored adventures; avoid unattended AI spending and event choices.
    ai = no
}
