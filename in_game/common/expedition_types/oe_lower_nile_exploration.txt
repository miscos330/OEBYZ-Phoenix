oe_lower_nile_exploration = {
    unique = yes
    repeatable = yes
    dynamic_first_waypoint = yes
    origin = none
    travel_mode = land
    travel_speed = 0.65
    # Return is explicitly part of the finite route, including a final home waypoint.
    should_end_at_last_waypoint = yes
    fail_if_no_leader = yes
    category = religious
    icon = category_religious
    show_start_message = no
    show_end_message = no
    variables = {
        oe_exp_provisions = {
            start = 100
            min = 0
            max = 100
            display = piechart
            icon = "gfx/interface/icons/unit_view/food.dds"
        }
        oe_exp_clues = {
            start = 30
            min = 0
            max = 100
            display = piechart
            icon = "gfx/interface/icons/expedition_types/knowledge.dds"
        }
        oe_exp_guards = {
            start = 40
            min = 0
            max = 40
            display = fraction
            icon = "gfx/interface/icons/expedition_types/guards.dds"
        }
        oe_exp_elapsed = {
            start = 0
            min = 0
            max = 60
            display = fraction
            icon = "gfx/interface/icons/expedition_types/experience.dds"
            hidden = yes
        }
    }
    potential = {
        has_or_had_tag = BYZ
        oe_exp_early_ages = yes
        NOT = {
            has_global_variable = oe_ark_recovered
        }
        NOT = {
            exists = work_of_art:oe_ark_of_the_covenant
        }
    }
    can_start = {
        at_war = no
        gold >= 500
        oe_exp_early_ages = yes
        owns = location:cairo
        NOT = {
            has_global_variable = oe_ark_recovered
        }
        NOT = {
            exists = work_of_art:oe_ark_of_the_covenant
        }
        NOT = {
            has_cooldown = oe_lower_nile_exploration_cooldown
        }
    }
    leader = {
        is_adult = yes
        is_ruler = no
        is_expedition_leader = no
        has_exploration = no
    }
    on_start = {
        add_gold = -500
        add_cooldown = {
            type = oe_lower_nile_exploration_cooldown
            years = 5
        }
        scope:expedition = {
            if = {
                limit = {
                    exists = expedition_leader
                }
                remove_commander = expedition_leader
            }
            add_new_waypoint = location:cairo
            add_new_waypoint = location:bilbeis
            add_new_waypoint = location:damietta
            add_new_waypoint = {
                location = location:faqus
                hidden = yes
            }
            add_new_waypoint = location:rashid
            add_new_waypoint = location:cairo
        }
        trigger_event_non_silently = oe_exp_nile.1
    }
    on_monthly = {
        # Only expedition-local scalars. No location or population loops.
        if = {
            limit = {
                scope:expedition = {
                    NOT = {
                        expedition_is_stalled = yes
                    }
                }
            }
            scope:expedition = {
                change_variable = {
                    name = oe_exp_elapsed
                    add = 1
                }
                if = {
                    limit = {
                        expedition_is_traveling = yes
                        NOT = {
                            expedition_is_paused = yes
                        }
                    }
                    change_variable = {
                        name = oe_exp_provisions
                        subtract = 1
                    }
                    clamp_variable = {
                        name = oe_exp_provisions
                        min = 0
                        max = 100
                    }
                }
            }
            if = {
                limit = {
                    scope:expedition = {
                        var:oe_exp_elapsed >= 60
                    }
                }
                fail_expedition = scope:expedition
            }
        }
    }
    on_arrive_to_new_location = {
        scope:location = {
            discover_location = root
        }
    }
    on_arrive_to_waypoint = {
        if = {
            limit = {
                scope:location = location:bilbeis
                NOT = {
                    scope:expedition = {
                        has_variable = oe_nile_visited_bilbeis
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_nile_visited_bilbeis
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_nile.2
        }
        if = {
            limit = {
                scope:location = location:damietta
                NOT = {
                    scope:expedition = {
                        has_variable = oe_nile_visited_damietta
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_nile_visited_damietta
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_nile.3
        }
        if = {
            limit = {
                scope:location = location:faqus
                NOT = {
                    scope:expedition = {
                        has_variable = oe_nile_visited_faqus
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_nile_visited_faqus
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_nile.4
        }
        if = {
            limit = {
                scope:location = location:rashid
                NOT = {
                    scope:expedition = {
                        has_variable = oe_nile_visited_rashid
                    }
                }
            }
            scope:expedition = {
                set_variable = {
                    name = oe_nile_visited_rashid
                    value = 1
                }
            }
            stall_expedition = scope:expedition
            trigger_event_non_silently = oe_exp_nile.5
        }
    }
    on_end = {
        if = {
            limit = {
                scope:expedition = {
                    has_variable = oe_nile_ark_found
                }
                NOT = {
                    has_global_variable = oe_ark_recovered
                }
                NOT = {
                    exists = work_of_art:oe_ark_of_the_covenant
                }
            }
            set_global_variable = {
                name = oe_ark_recovered
                value = 1
            }
            capital = {
                create_art = {
                    type = work_of_art_type:holy_relic
                    quality = 100
                    key = oe_ark_of_the_covenant
                    save_scope_as = oe_recovered_ark
                }
            }
            scope:oe_recovered_ark = {
                move_art_and_owner = {
                    location = root.capital
                    country = root
                }
            }
            add_country_modifier = {
                modifier = oe_discovery_of_the_ark
                years = -1
                mode = replace
            }
            add_prestige = 3
            trigger_event_non_silently = oe_exp_nile.6
        }
        else = {
            add_prestige = 1
            trigger_event_non_silently = oe_exp_nile.7
        }
    }
    on_fail = {
        trigger_event_non_silently = oe_exp_nile.8
    }
    # Player-authored adventures; avoid unattended AI spending and event choices.
    ai = no
}
