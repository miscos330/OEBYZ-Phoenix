oe_all_rus_council = {
    unique = yes
    repeatable = yes
    dynamic_first_waypoint = yes
    origin = none
    travel_mode = both
    travel_speed = 1
    should_end_at_last_waypoint = no
    fail_if_no_leader = yes
    category = religious
    icon = category_religious
    show_start_message = no
    show_end_message = no
    stalled_status_text = oe_rsc_stalled_status
    ai = no
    variables = {
        oe_rsc_disagreements = { start = 0 min = 0 max = 3 display = fraction icon = "gfx/interface/icons/flat_icons/discretion.dds" }
        oe_rsc_visits = { start = 0 min = 0 max = 3 display = fraction icon = "gfx/interface/icons/flat_icons/opportunities.dds" }
        oe_rsc_days = { start = 0 min = 0 max = 200 display = fraction }
    }
    potential = { tag = BYZ }
    can_start = {
        tag = BYZ
        is_situation_active = situation:oe_rise_of_rus
        NOT = { has_cooldown = oe_rsc_cooldown }
        NOT = { has_variable = oe_rsc_active }
        oe_rsc_has_independent_rus = yes
        oe_rsc_has_venue = yes
    }
    leader = {
        owner = c:BYZ
        has_estate = estate_type:clergy_estate
        is_alive = yes
        is_adult = yes
        is_expedition_leader = no
        has_exploration = no
    }
    on_start = {
        add_cooldown = { type = oe_rsc_cooldown years = 5 }
        set_variable = { name = oe_rsc_active value = scope:expedition }
        scope:expedition = {
            set_variable = { name = oe_rsc_stage value = 0 }
            set_variable = { name = oe_rsc_pending value = 0 }
            add_new_waypoint = root.capital
            if = { limit = { exists = expedition_leader } remove_commander = expedition_leader }
        }
        stall_expedition = scope:expedition
        trigger_event_non_silently = oe_rus_council.1
    }
    on_arrive_to_waypoint = {
        stall_expedition = scope:expedition
        if = {
            limit = { scope:expedition = { var:oe_rsc_stage = 2 } }
            scope:expedition = {
                set_variable = { name = oe_rsc_stage value = 5 }
                change_variable = { name = oe_rsc_visits add = 1 }
                pause_expedition = 60
            }
            trigger_event_non_silently = { id = oe_rus_council.3 days = 60 }
        }
        else_if = {
            limit = { scope:expedition = { var:oe_rsc_stage = 6 } }
            oe_rsc_open_council = yes
        }
        else = { fail_expedition = scope:expedition }
    }
    on_end = { remove_variable = oe_rsc_active }
    on_fail = {
        remove_variable = oe_rsc_active
        trigger_event_non_silently = oe_rus_council.99
    }
}
