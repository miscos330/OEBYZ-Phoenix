# Location scope. Clears Kontor seat location modifiers cleanly.
oe_hbg_clear_seat_modifiers = {
	if = { limit = { has_location_modifier = oe_hbg_seat_prosperous_mod } remove_location_modifier = oe_hbg_seat_prosperous_mod }
	if = { limit = { has_location_modifier = oe_hbg_seat_friction_mod } remove_location_modifier = oe_hbg_seat_friction_mod }
}

# Kontor country scope. Clears subject loyalty tier modifiers cleanly.
oe_hbg_clear_kontor_modifiers = {
	if = { limit = { has_country_modifier = oe_hbg_kontor_compliant_mod } remove_country_modifier = oe_hbg_kontor_compliant_mod }
	if = { limit = { has_country_modifier = oe_hbg_kontor_restive_mod } remove_country_modifier = oe_hbg_kontor_restive_mod }
}

# Kontor country scope. Shifts oe_hbg_loyalty by $amount$ (clamped 0..100).
oe_hbg_shift_loyalty = {
	if = { limit = { NOT = { has_variable = oe_hbg_loyalty } } set_variable = { name = oe_hbg_loyalty value = 60 } }
	change_variable = { name = oe_hbg_loyalty add = $amount$ }
	clamp_variable = { name = oe_hbg_loyalty min = 0 max = 100 }
}

# Kontor country scope. Syncs oe_hbg_tier (3>=70, 2=40..69, 1<40) and country tier modifiers.
oe_hbg_sync_self_tier = {
	if = { limit = { NOT = { has_variable = oe_hbg_loyalty } } set_variable = { name = oe_hbg_loyalty value = 60 } }
	if = {
		limit = { has_variable = oe_hbg_loyalty var:oe_hbg_loyalty >= 70 }
		set_variable = { name = oe_hbg_tier value = 3 }
		oe_hbg_clear_kontor_modifiers = yes
		add_country_modifier = { modifier = oe_hbg_kontor_compliant_mod months = 14 mode = replace }
	}
	else_if = {
		limit = { has_variable = oe_hbg_loyalty var:oe_hbg_loyalty >= 40 }
		set_variable = { name = oe_hbg_tier value = 2 }
		oe_hbg_clear_kontor_modifiers = yes
	}
	else = {
		set_variable = { name = oe_hbg_tier value = 1 }
		oe_hbg_clear_kontor_modifiers = yes
		add_country_modifier = { modifier = oe_hbg_kontor_restive_mod months = 14 mode = replace }
	}
}

# Overlord country scope. Shifts loyalty of all active Kontor subjects by $amount$.
oe_hbg_shift_all_kontor_loyalty = {
	every_subject = {
		limit = { oe_hbg_is_kontor_subject = yes }
		oe_hbg_shift_loyalty = { amount = $amount$ }
		oe_hbg_sync_self_tier = yes
	}
}

# Any scope. F6-style clean dissolution of OHK1 (London Steelyard Kontor): annex via overlord or destroy_building_country, then clear seat.
oe_hbg_end_OHK1 = {
	set_global_variable = { name = oe_hbg_ended_ohk1 value = 1 }
	if = { limit = { exists = scope:oe_hbg_end_lord_OHK1 } clear_saved_scope = oe_hbg_end_lord_OHK1 }
	if = {
		limit = { country_exists = c:OHK1 }
		c:OHK1 = {
			set_variable = { name = oe_hbg_ended value = 1 }
			if = { limit = { is_subject = yes } overlord = { save_scope_as = oe_hbg_end_lord_OHK1 } }
		}
		if = {
			limit = { exists = scope:oe_hbg_end_lord_OHK1 }
			scope:oe_hbg_end_lord_OHK1 = { annex_country = { country = c:OHK1 reason = Diplomatic transfer = yes } }
		}
		else = {
			destroy_building_country = c:OHK1
		}
	}
	location:london = {
		if = { limit = { has_building = building_type:oe_hbg_kontor_seat_1 } destroy_all_buildings_of_type = building_type:oe_hbg_kontor_seat_1 }
		oe_hbg_clear_seat_modifiers = yes
	}
}

# Any scope. Paced yearly event dispatcher for all 12 events (oe_hbg.1..12, 0 orphans).
oe_hbg_dispatch_events = {
	# Theater 1: Hanseatic League (oe_hbg.1..3 -> LUB, fallback HAM/BRE)
	if = {
		limit = { country_exists = c:LUB c:LUB = { oe_hbg_is_hansa_host = yes NOT = { has_variable = oe_hbg_ev_cd } } }
		c:LUB = {
			if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev1_done } current_year >= 1340 }
				set_global_variable = { name = oe_hbg_ev1_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.1
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev3_done } current_year >= 1346 country_exists = c:TEU }
				set_global_variable = { name = oe_hbg_ev3_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.3
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev2_done } current_year >= 1352 }
				set_global_variable = { name = oe_hbg_ev2_done value = 1 }
				set_variable = { name = oe_hbg_ev2_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.2
			}
		}
	}
	# Theater 2: Burgundy & Lotharingia (oe_hbg.4..6 -> BUR)
	if = {
		limit = { country_exists = c:BUR c:BUR = { oe_hbg_is_burgundy_host = yes NOT = { has_variable = oe_hbg_ev_cd } } }
		c:BUR = {
			if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev4_done } current_year >= 1345 }
				set_global_variable = { name = oe_hbg_ev4_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.4
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev5_done } current_year >= 1360 }
				set_global_variable = { name = oe_hbg_ev5_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.5
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev6_done } current_year >= 1378 OR = { has_variable = oe_hbg_dec_lotharingia_done num_locations_owned_or_owned_by_subjects >= 22 } }
				set_global_variable = { name = oe_hbg_ev6_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.6
			}
		}
	}
	# Theater 3: Granada & Al-Andalus (oe_hbg.7..9 -> GRA, or ADU)
	if = {
		limit = { country_exists = c:GRA c:GRA = { oe_hbg_is_granada_host = yes NOT = { has_variable = oe_hbg_ev_cd } } }
		c:GRA = {
			if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev7_done } current_year >= 1339 }
				set_global_variable = { name = oe_hbg_ev7_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.7
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev8_done } OR = { owns_or_non_sovereign_subject_owns = location:cordoba owns_or_non_sovereign_subject_owns = location:sevilla AND = { current_year >= 1358 num_locations_owned_or_owned_by_subjects >= 18 } } }
				set_global_variable = { name = oe_hbg_ev8_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.8
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev9_done } OR = { has_variable = oe_hbg_dec_cordoba_caliphate_done AND = { current_year >= 1375 owns_or_non_sovereign_subject_owns = location:cordoba } } }
				set_global_variable = { name = oe_hbg_ev9_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.9
			}
		}
	}
	# Theater 4: Black Sea Trio — Trebizond (oe_hbg.10) & Theodoro (oe_hbg.11..12)
	if = {
		limit = { country_exists = c:TRE c:TRE = { oe_hbg_is_trebizond_host = yes NOT = { has_variable = oe_hbg_ev_cd } } }
		c:TRE = {
			if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev10_done } current_year >= 1341 }
				set_global_variable = { name = oe_hbg_ev10_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.10
			}
		}
	}
	if = {
		limit = { country_exists = c:FEO c:FEO = { oe_hbg_is_theodoro_host = yes NOT = { has_variable = oe_hbg_ev_cd } } }
		c:FEO = {
			if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev11_done } current_year >= 1341 }
				set_global_variable = { name = oe_hbg_ev11_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.11
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_hbg_ev12_done } OR = { oe_hbg_teu_reunion_partner_ready = yes AND = { current_year >= 1385 country_exists = c:TEU c:TEU = { is_rebel_country = no NOT = { is_at_war_with = c:FEO } } } } }
				set_global_variable = { name = oe_hbg_ev12_done value = 1 }
				set_variable = { name = oe_hbg_ev_cd value = 1 years = 4 }
				trigger_event_non_silently = oe_hbg.12
			}
		}
	}
}
