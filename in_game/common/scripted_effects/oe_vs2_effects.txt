	}
	if = {
		limit = { var:oe_vs2_ilk_legitimacy > 100 }

	}

# Country scope (JAL / GRG / ILK). Evaluates yearly Ilkhanid court legitimacy drift.
oe_vs2_yearly_ilk_one = {
	if = {
		limit = { NOT = { has_variable = oe_vs2_ilk_legitimacy } }
		set_variable = { name = oe_vs2_ilk_legitimacy value = 55 }
	}
	if = {
		limit = { var:oe_vs2_ilk_centralized ?= 1 }
		oe_vs2_shift_ilk_legitimacy = { amount = 2 }
	}
	else_if = {
		limit = {
			in_civil_war = no
			stability >= 10
			OR = {
				owns_or_non_sovereign_subject_owns = location:tabriz
				owns_or_non_sovereign_subject_owns = location:soltaniyeh
			}
		}
		oe_vs2_shift_ilk_legitimacy = { amount = 2 }
	}
	else_if = {
		limit = {
			in_civil_war = no
			stability >= 10
		}
		oe_vs2_shift_ilk_legitimacy = { amount = 1 }
	}
	else_if = {
		limit = {
			OR = {
				in_civil_war = yes
				stability < 0
			}
		}
		oe_vs2_shift_ilk_legitimacy = { amount = -2 }
	}
	else = {
		oe_vs2_sync_ilk_tier = yes
	}
}

oe_vs2_yearly_ilk = {
	if = {
		limit = { country_exists = c:JAL c:JAL = { is_rebel_country = no } }
		c:JAL = { oe_vs2_yearly_ilk_one = yes }
	}
	if = {
		limit = { country_exists = c:GRG c:GRG = { is_rebel_country = no } }
		c:GRG = { oe_vs2_yearly_ilk_one = yes }
	}
	if = {
		limit = { country_exists = c:ILK c:ILK = { is_rebel_country = no } }
		c:ILK = { oe_vs2_yearly_ilk_one = yes }
	}
}

oe_vs2_yearly_tib = {
	if = {
		limit = { country_exists = c:TIB c:TIB = { is_rebel_country = no } has_global_variable = oe_vs2_tib_cohesion }
		if = {
			limit = { global_var:oe_vs2_tib_centralized ?= 1 }
			oe_vs2_shift_tib_cohesion = { amount = 2 }
		}
		else_if = {
			limit = { c:TIB = { in_civil_war = no stability >= 10 } }
			oe_vs2_shift_tib_cohesion = { amount = 1 }
		}
		else_if = {
			limit = { c:TIB = { OR = { in_civil_war = yes stability < 0 } } }
			oe_vs2_shift_tib_cohesion = { amount = -2 }
		}
		else = {
			oe_vs2_sync_tib_tier = yes
		}
	}
}

			limit = {
				global_var:oe_vs2_mal_crisis ?= 1
				c:MAL = { stability < 10 }
			}
			oe_vs2_shift_mal_authority = { amount = -2 }

			limit = { c:MAL = { in_civil_war = no stability >= 15 } }
			oe_vs2_shift_mal_authority = { amount = 1 }
		}
		else_if = {
			limit = { c:MAL = { OR = { in_civil_war = yes stability < 0 } } }
			oe_vs2_shift_mal_authority = { amount = -2 }
		}
		else = {
			oe_vs2_sync_mal_tier = yes
		}
	}
}

		limit = { country_exists = c:KHM c:KHM = { is_rebel_country = no } has_global_variable = oe_vs2_khm_mandala_power }
		if = {
			limit = {
				OR = {
					global_var:oe_vs2_khm_centralized ?= 1
					global_var:oe_vs2_khm_hydraulic_restored ?= 1
				}
			}
			oe_vs2_shift_khm_mandala = { amount = 2 }
		}
		else_if = {
			limit = { c:KHM = { in_civil_war = no stability >= 10 owns = location:angkor } }
			oe_vs2_shift_khm_mandala = { amount = 1 }
		}
		else_if = {
			limit = { c:KHM = { OR = { in_civil_war = yes stability < 0 NOT = { owns = location:angkor } } } }
			oe_vs2_shift_khm_mandala = { amount = -2 }
		}
		else = {
			oe_vs2_sync_khm_tier = yes
		}
	}
}
