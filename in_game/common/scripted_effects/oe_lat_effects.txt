	if = {
		limit = {
			owns = location:thessaloniki

	}
	else_if = {
		limit = {
			owns = location:serres
			location:serres = { oe_lat_loc_is_protected_core = no }
		}
		location:serres = { save_scope_as = oe_lat_seat_1 }
	}
	else_if = {

	}
	else_if = {

		}
		location:veria = { save_scope_as = oe_lat_seat_1 }
	}
	else = {
		ordered_owned_location = {
			limit = {
				oe_lat_loc_is_protected_core = no
				oe_lat_loc_in_tier_1 = yes
			}
			order_by = population
			max = 1
			save_scope_as = oe_lat_seat_1
		}
	}
	if = {
		limit = { exists = scope:oe_lat_seat_1 }
		scope:oe_lat_seat_1 = {
			create_country_from_location = {
				overlord = scope:oe_lat_overlord
				subject_type = subject_type:oe_lat_fief
				change_government_type = government_type:monarchy
				change_country_type = location
				set_capital = scope:oe_lat_seat_1
				change_culture = scope:oe_lat_overlord.culture
				change_religion = scope:oe_lat_overlord.religion
				save_scope_as = oe_lat_new_fief_1
			}
		}
		if = {
			limit = { exists = scope:oe_lat_new_fief_1 }
			scope:oe_lat_seat_1 = {
				add_core = scope:oe_lat_new_fief_1
			}
			every_in_list = {
				list = oe_lat_tier_locs_1
				limit = {
					NOT = { this = scope:oe_lat_seat_1 }
					owner ?= scope:oe_lat_overlord
					oe_lat_loc_is_protected_core = no
				}
				change_location_owner = scope:oe_lat_new_fief_1
				change_location_controller = scope:oe_lat_new_fief_1
				add_core = scope:oe_lat_new_fief_1
			}
			scope:oe_lat_new_fief_1 = {
				if = {
					limit = { NOT = { is_subject_of = scope:oe_lat_overlord } }
					make_subject_of = {
						target = scope:oe_lat_overlord
						type = subject_type:oe_lat_fief
					}
				}
				else_if = {
					limit = { NOT = { is_subject_type = oe_lat_fief } }
					change_subject_type = subject_type:oe_lat_fief
				}
				set_variable = { name = oe_lat_tier_id value = 1 }
				set_variable = { name = oe_lat_is_partitio_fief value = 1 }
				set_variable = { name = oe_name_full_title value = 1 }
				change_country_name = OE_LAT_FIEF_THESSALONICA
				change_country_adjective = OE_LAT_FIEF_THESSALONICA_ADJ
				if = {
					limit = { NOT = { has_reform = government_reform:oe_lat_fief_reform } }
					add_reform = government_reform:oe_lat_fief_reform
				}
				add_country_modifier = { modifier = oe_lat_mod_fief_foundation years = 25 mode = replace }
				add_opinion = { target = scope:oe_lat_overlord modifier = oe_lat_op_enfeoffed }
				add_liberty_desire = -20
			}
			set_variable = { name = oe_lat_tier_1_done value = 1 }
			oe_lat_sync_fiefs = yes
			oe_lat_sit_sync = yes
			trigger_event_non_silently = oe_lat_emp.5
		}
	}
}

# Native Situation Lifecycle (oe_sit_frankokratia)
# ==============================================================================

oe_lat_sit_init = {
	set_global_variable = { name = oe_lat_sit_started value = 1 }
	situation:oe_sit_frankokratia = {
		set_variable = { name = oe_lat_sit_latin_index value = 45 }
		set_variable = { name = oe_lat_sit_pressure value = 35 }
		set_variable = { name = oe_lat_sit_fiefs value = 0 }
		set_variable = { name = oe_lat_sit_months value = 0 }
		set_variable = { name = oe_lat_sit_phase value = 1 }
	}
	oe_lat_sit_sync = yes
}

oe_lat_sit_sync = {
	if = {
		limit = { is_situation_active = situation:oe_sit_frankokratia }
		situation:oe_sit_frankokratia = {
			if = { limit = { NOT = { has_variable = oe_lat_sit_months } } set_variable = { name = oe_lat_sit_months value = 0 } }
			set_variable = { name = oe_lat_sit_latin_index value = oe_lat_latin_index_val }
			set_variable = { name = oe_lat_sit_pressure value = oe_lat_eastern_pressure_val }
			if = {
				limit = { country_exists = c:LAT }
				c:LAT = {
					situation:oe_sit_frankokratia = {
						set_variable = { name = oe_lat_sit_fiefs value = prev.oe_lat_active_fiefs_val }
					}
				}
			}
			else_if = {
				limit = { country_exists = c:ACH }
				c:ACH = {
					situation:oe_sit_frankokratia = {
						set_variable = { name = oe_lat_sit_fiefs value = prev.oe_lat_active_fiefs_val }
					}
				}
			}
			else_if = {
				limit = { country_exists = c:ATH }
				c:ATH = {
					situation:oe_sit_frankokratia = {
						set_variable = { name = oe_lat_sit_fiefs value = prev.oe_lat_active_fiefs_val }
					}
				}
			}
			else = {
				set_variable = { name = oe_lat_sit_fiefs value = 0 }
			}
			if = {
				limit = {
					OR = {
						country_exists = c:LAT
						var:oe_lat_sit_latin_index >= 75
					}
				}
				set_variable = { name = oe_lat_sit_phase value = 4 }
			}
			else_if = {
				limit = {
					OR = {
						location:thessaloniki ?= { has_owner = yes owner = { oe_lat_sit_latin_bloc = yes } }
						location:constantinople ?= { has_owner = yes owner = { oe_lat_sit_turkish_bloc = yes } }
						current_year >= 1395
					}
				}
				set_variable = { name = oe_lat_sit_phase value = 3 }
			}
			else_if = {
				limit = { var:oe_lat_sit_latin_index >= 52 }
				set_variable = { name = oe_lat_sit_phase value = 2 }
			}
			else = {
				set_variable = { name = oe_lat_sit_phase value = 1 }
			}
			clamp_variable = { name = oe_lat_sit_latin_index max = 100 min = 0 }
			clamp_variable = { name = oe_lat_sit_pressure max = 100 min = 0 }
		}
	}
}

oe_lat_sit_monthly = {
	if = {
		limit = { is_situation_active = situation:oe_sit_frankokratia }
		situation:oe_sit_frankokratia = {
			change_variable = { name = oe_lat_sit_months add = 1 }
		}
		oe_lat_sit_sync = yes
	}
}

# Country scope. Annual Latin Greece & Crusader Empire milestone dispatcher (oe_lat_emp.4, 10..13).
oe_lat_yearly_realm_tick = {
	oe_lat_sync_fiefs = yes
	# oe_lat_emp.4: Restoration of the Latin Empire at Hagia Sophia
	if = {
		limit = {
			NOT = { has_variable = oe_lat_ev4_done }
			owns = location:constantinople
			OR = {
				tag = LAT
				has_variable = oe_lat_empire_proclaimed
			}
		}
		set_variable = { name = oe_lat_ev4_done value = 1 }
		trigger_event_non_silently = { id = oe_lat_emp.4 days = 10 }
	}
	# oe_lat_emp.10: 1344+ Assizes of Romania Parliament
	else_if = {
		limit = {
			current_year >= 1344
			NOT = { has_variable = oe_lat_ev10_done }
		}
		set_variable = { name = oe_lat_ev10_done value = 1 }
		trigger_event_non_silently = { id = oe_lat_emp.10 days = 20 }
	}
	# oe_lat_emp.11: 1350+ Venetian-Latin Trade Accord
	else_if = {
		limit = {
			current_year >= 1350
			NOT = { has_variable = oe_lat_ev11_done }
			country_exists = c:VEN
		}
		set_variable = { name = oe_lat_ev11_done value = 1 }
		trigger_event_non_silently = { id = oe_lat_emp.11 days = 25 }
	}
	# oe_lat_emp.12: 1356+ Greek Archon Conciliation
	else_if = {
		limit = {
			current_year >= 1356
			NOT = { has_variable = oe_lat_ev12_done }
		}
		set_variable = { name = oe_lat_ev12_done value = 1 }
		trigger_event_non_silently = { id = oe_lat_emp.12 days = 30 }
	}
	# oe_lat_emp.13: Fourth Crusade Avenged or Healed (once Latin Empire restored)
	else_if = {
		limit = {
			NOT = { has_variable = oe_lat_ev13_done }
			owns = location:constantinople
			has_variable = oe_lat_ev4_done
			OR = {
				tag = LAT
				has_variable = oe_lat_empire_proclaimed
			}
		}
		set_variable = { name = oe_lat_ev13_done value = 1 }
		trigger_event_non_silently = { id = oe_lat_emp.13 days = 40 }
	}
}

# O(1) self-deduplicated global yearly kernel (exported as oe_lat_kernel_yearly).
oe_lat_kernel_yearly = {
	if = {
		limit = { NOT = { global_var:oe_lat_ran_y ?= current_year } }
		set_global_variable = { name = oe_lat_ran_y value = current_year }
		if = {
			limit = {
				current_date < 1338.1.1
				NOT = { has_global_variable = oe_lat_day1_init_done }
			}
			oe_lat_day1_init = yes
		}
		c:LAT ?= { if = { limit = { is_rebel_country = no num_locations > 0 } oe_lat_yearly_realm_tick = yes } }
		c:ACH ?= { if = { limit = { is_rebel_country = no num_locations > 0 } oe_lat_yearly_realm_tick = yes } }
		c:ATH ?= { if = { limit = { is_rebel_country = no num_locations > 0 } oe_lat_yearly_realm_tick = yes } }
