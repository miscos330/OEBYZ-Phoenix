oe_np_paused_cleanup = {
    if = {
        limit = { OR = { has_country_modifier = oe_npi_court_honor has_variable = oe_npi_honor_cd } }
        if = {
            limit = { has_country_modifier = oe_npi_court_honor }
            remove_country_modifier = oe_npi_court_honor
        }
        if = {
            limit = { country_exists = c:BYZ }
            remove_opinion = { modifier = oe_npi_court_honor_opinion target = c:BYZ }
        }
    }
    remove_variable = oe_npi_support_cd
    remove_variable = oe_npi_honor_cd
    remove_variable = oe_npi_recommend_cd
}
