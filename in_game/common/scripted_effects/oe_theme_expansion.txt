# Only called when changing another eligible theme type into a naval theme.
# Location scope; explicit owner keeps the fleet with the building country.
oe_theme_create_ten_current_age_galleys = {
    while = {
        count = 10
        if = {
            limit = { current_age = age_1_traditions }
            create_sub_unit_with_owner = { type = n_traditional_galley owner = $owner$ }
        }
        else_if = {
            limit = { current_age = age_2_renaissance }
            create_sub_unit_with_owner = { type = n_mediterrannean_galley owner = $owner$ }
        }
        else_if = {
            limit = { current_age = age_3_discovery }
            create_sub_unit_with_owner = { type = n_war_galley owner = $owner$ }
        }
        else_if = {
            limit = { current_age = age_4_reformation }
            create_sub_unit_with_owner = { type = n_galleass owner = $owner$ }
        }
        else_if = {
            limit = { current_age = age_5_absolutism }
            create_sub_unit_with_owner = { type = n_xebec owner = $owner$ }
        }
        else_if = {
            limit = { current_age = age_6_revolutions }
            create_sub_unit_with_owner = { type = n_archipelago_frigate owner = $owner$ }
        }
    }
}
