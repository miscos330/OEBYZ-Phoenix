# Country scope. Remove every current domestic house modifier, including any missing from the cached lists.
oe_ir_reclaim_privileges_effect = {
    if = {
        limit = { oe_ir_reclaim_privileges_ready = yes }
        international_organization:oe_byz_empire_pillars = { change_variable = { name = oe_pillars_authority subtract = 50 } }
        set_variable = oe_ir_reclaim_privileges_done
        every_owned_location = {
            oe_nf_remove_control = { slot = 1 }
            oe_nf_remove_control = { slot = 2 }
            oe_nf_remove_control = { slot = 3 }
            oe_nf_remove_control = { slot = 4 }
        }
        oe_nf_refresh = yes
    }
}
# IO scope, selected house head remains in scope:target_head.
oe_bi_reduce_selected_house = {
    if = { limit = { has_variable = oe_nf_dynasty_1 var:oe_nf_head_1 = scope:target_head } oe_nf_change_influence = { slot = 1 amount = $amount$ } }
    if = { limit = { has_variable = oe_nf_dynasty_2 var:oe_nf_head_2 = scope:target_head } oe_nf_change_influence = { slot = 2 amount = $amount$ } }
    if = { limit = { has_variable = oe_nf_dynasty_3 var:oe_nf_head_3 = scope:target_head } oe_nf_change_influence = { slot = 3 amount = $amount$ } }
    if = { limit = { has_variable = oe_nf_dynasty_4 var:oe_nf_head_4 = scope:target_head } oe_nf_change_influence = { slot = 4 amount = $amount$ } }
}
