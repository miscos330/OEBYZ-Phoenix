# Adjusts global_var:oe_chg2_authority cleanly within [0, 100].
oe_chg2_shift_authority = {
	if = {
		limit = { NOT = { has_global_variable = oe_chg2_authority } }
		set_global_variable = { name = oe_chg2_authority value = 70 }
	}
	change_global_variable = { name = oe_chg2_authority add = $amount$ }
	if = {
		limit = { global_var:oe_chg2_authority < 0 }
		set_global_variable = { name = oe_chg2_authority value = 0 }
	}
	if = {
		limit = { global_var:oe_chg2_authority > 100 }
		set_global_variable = { name = oe_chg2_authority value = 100 }
	}
}

# Refreshes authority tier modifier on c:CHG (15m replace, zero stacking).
oe_chg2_sync_authority_tier = {
	if = {
		limit = { country_exists = c:CHG has_global_variable = oe_chg2_authority }
		c:CHG = {
			if = {
				limit = { global_var:oe_chg2_authority >= 70 }
				if = { limit = { has_country_modifier = oe_chg2_mod_auth_fractured } remove_country_modifier = oe_chg2_mod_auth_fractured }
				add_country_modifier = { modifier = oe_chg2_mod_auth_supreme months = 15 mode = replace }
			}
			else_if = {
				limit = { global_var:oe_chg2_authority < 35 }
				if = { limit = { has_country_modifier = oe_chg2_mod_auth_supreme } remove_country_modifier = oe_chg2_mod_auth_supreme }
				add_country_modifier = { modifier = oe_chg2_mod_auth_fractured months = 15 mode = replace }
			}
			else = {
				if = { limit = { has_country_modifier = oe_chg2_mod_auth_supreme } remove_country_modifier = oe_chg2_mod_auth_supreme }
				if = { limit = { has_country_modifier = oe_chg2_mod_auth_fractured } remove_country_modifier = oe_chg2_mod_auth_fractured }
			}
		}
	}
}

# Yearly Chagatai authority drift with read-only awareness of oe_eug_chg_split (zero conflict with oe_eug).
oe_chg2_yearly_authority_pulse = {
	if = {
		limit = { country_exists = c:CHG has_global_variable = oe_chg2_authority }
		if = {
			limit = {
				OR = {
					global_var:oe_chg2_centralized ?= 1
					global_var:oe_eug_chg_split ?= 2
				}
			}
			oe_chg2_shift_authority = { amount = 2 }
		}
		else_if = {
			limit = { global_var:oe_eug_chg_split ?= 1 }
			oe_chg2_shift_authority = { amount = -3 }
		}
		else_if = {
			limit = { c:CHG = { stability >= 15 prestige >= 25 in_civil_war = no } }
			oe_chg2_shift_authority = { amount = 1 }
		}
		else_if = {
			limit = { c:CHG = { OR = { stability < 0 in_civil_war = yes } } }
			oe_chg2_shift_authority = { amount = -2 }
		}
		oe_chg2_sync_authority_tier = yes
	}
}
