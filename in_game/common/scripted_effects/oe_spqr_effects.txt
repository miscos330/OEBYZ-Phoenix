oe_spqr_sit_init = {
	set_global_variable = { name = oe_spqr_sit_started value = 1 }
	set_global_variable = { name = oe_spqr_sit_momentum_shift value = 0 }
	situation:oe_sit_spqr_revival = {
		set_variable = { name = oe_spqr_sit_momentum value = 38 }
		set_variable = { name = oe_spqr_sit_mare_nostrum value = 20 }
		set_variable = { name = oe_spqr_sit_proconsuls value = 0 }
		set_variable = { name = oe_spqr_sit_months value = 0 }
		set_variable = { name = oe_spqr_sit_phase value = 1 }
	}
	oe_spqr_sit_sync = yes
}

oe_spqr_sit_sync = {
	if = {
		limit = { is_situation_active = situation:oe_sit_spqr_revival }
		if = {
			limit = { NOT = { has_global_variable = oe_spqr_sit_momentum_shift } }
			set_global_variable = { name = oe_spqr_sit_momentum_shift value = 0 }
		}
		clamp_global_variable = { name = oe_spqr_sit_momentum_shift max = 25 min = -25 }
		situation:oe_sit_spqr_revival = {
			if = { limit = { NOT = { has_variable = oe_spqr_sit_months } } set_variable = { name = oe_spqr_sit_months value = 0 } }
			set_variable = { name = oe_spqr_sit_momentum value = oe_spqr_revival_momentum_val }
			set_variable = { name = oe_spqr_sit_mare_nostrum value = oe_spqr_mare_nostrum_val }
			# Live Republican champion for two_countries_header_template (clean fallback chain)
			if = { limit = { location:rome = { has_owner = yes owner = { oe_spqr_is_republic_host = yes } } } set_variable = { name = oe_spqr_sit_rep_lead value = location:rome.owner } }
			else_if = { limit = { country_exists = c:VEN c:VEN = { oe_spqr_is_republic_host = yes } } set_variable = { name = oe_spqr_sit_rep_lead value = c:VEN } }
			else_if = { limit = { country_exists = c:GEN c:GEN = { oe_spqr_is_republic_host = yes } } set_variable = { name = oe_spqr_sit_rep_lead value = c:GEN } }
			else_if = { limit = { country_exists = c:FLO c:FLO = { oe_spqr_is_republic_host = yes } } set_variable = { name = oe_spqr_sit_rep_lead value = c:FLO } }
			else_if = { limit = { country_exists = c:PIS c:PIS = { oe_spqr_is_republic_host = yes } } set_variable = { name = oe_spqr_sit_rep_lead value = c:PIS } }
			else_if = { limit = { country_exists = c:VEN } set_variable = { name = oe_spqr_sit_rep_lead value = c:VEN } }
			else_if = { limit = { country_exists = c:GEN } set_variable = { name = oe_spqr_sit_rep_lead value = c:GEN } }
			else_if = { limit = { country_exists = c:FLO } set_variable = { name = oe_spqr_sit_rep_lead value = c:FLO } }
			# Live Papal / Monarchist champion (clean fallback chain if PAP fell)
			if = { limit = { country_exists = c:PAP } set_variable = { name = oe_spqr_sit_pap_lead value = c:PAP } }
			else_if = { limit = { country_exists = c:NAP } set_variable = { name = oe_spqr_sit_pap_lead value = c:NAP } }
			else_if = { limit = { country_exists = c:MLO } set_variable = { name = oe_spqr_sit_pap_lead value = c:MLO } }
			else_if = { limit = { country_exists = c:SIC } set_variable = { name = oe_spqr_sit_pap_lead value = c:SIC } }
			else_if = { limit = { country_exists = c:ARA } set_variable = { name = oe_spqr_sit_pap_lead value = c:ARA } }
			else_if = { limit = { location:rome = { has_owner = yes } } set_variable = { name = oe_spqr_sit_pap_lead value = location:rome.owner } }
			if = {
				limit = { location:rome = { has_owner = yes owner = { oe_spqr_is_republic_host = yes } } }
				location:rome.owner = {
					situation:oe_sit_spqr_revival = {
						set_variable = { name = oe_spqr_sit_proconsuls value = prev.oe_spqr_active_proconsuls_val }
					}
				}
			}
			else = {
				set_variable = { name = oe_spqr_sit_proconsuls value = 0 }
				if = { limit = { has_global_variable = oe_spqr_prov_1_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
				if = { limit = { has_global_variable = oe_spqr_prov_2_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
				if = { limit = { has_global_variable = oe_spqr_prov_3_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
				if = { limit = { has_global_variable = oe_spqr_prov_4_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
				if = { limit = { has_global_variable = oe_spqr_prov_5_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
				if = { limit = { has_global_variable = oe_spqr_prov_6_global } change_variable = { name = oe_spqr_sit_proconsuls add = 1 } }
			}
			if = {
				limit = { has_global_variable = oe_spqr_proclaimed_global }
				set_variable = { name = oe_spqr_sit_phase value = 4 }
			}
			else_if = {
				limit = { location:rome = { has_owner = yes owner = { oe_spqr_is_republic_host = yes } } }
				set_variable = { name = oe_spqr_sit_phase value = 3 }
			}
			else_if = {
				limit = { var:oe_spqr_sit_momentum >= 55 }
				set_variable = { name = oe_spqr_sit_phase value = 2 }
			}
			else = {
				set_variable = { name = oe_spqr_sit_phase value = 1 }
			}
			clamp_variable = { name = oe_spqr_sit_momentum max = 100 min = 0 }
			clamp_variable = { name = oe_spqr_sit_mare_nostrum max = 100 min = 0 }
		}
	}
}

	set_variable = { name = oe_spqr_ticked_y value = current_year }
	oe_spqr_sync_proconsuls = yes
	# 1. 1347 Cola di Rienzo Tribune Uprising (oe_spqr.1)
	if = {
		limit = {
			NOT = { has_variable = oe_spqr_ev1_done }
			current_year >= 1347
			current_year <= 1420
			OR = { is_ai = no tag = VEN tag = GEN tag = FLO tag = PIS }
		}
		set_variable = { name = oe_spqr_ev1_done value = 1 }
		set_global_variable = { name = oe_spqr_rienzo_ignited value = 1 }
		trigger_event_non_silently = oe_spqr.1
	}

	if = {
		limit = {
			tag = VEN
			NOT = { has_variable = oe_spqr_ev2_done }

		}
		set_variable = { name = oe_spqr_ev2_done value = 1 }
		trigger_event_non_silently = oe_spqr.2
	}
	# 3. Genoese Casa di San Giorgio & Black Sea Gazaria Charter (oe_spqr.3)

		}
		set_variable = { name = oe_spqr_ev3_done value = 1 }
		trigger_event_non_silently = oe_spqr.3
	}
	# 4. Pisa Vengeance for Meloria & Tuscan/Adriatic Revival (oe_spqr.4)
	if = {
		limit = {
			OR = { tag = PIS tag = FLO tag = SIE tag = LUC tag = RAG tag = ANC }

			current_year >= 1350
		}
		set_variable = { name = oe_spqr_ev4_done value = 1 }
		trigger_event_non_silently = oe_spqr.4
	}
	# 5. Dialogue of Two Romes (oe_spqr.9 -> reciprocal oe_spqr.10 for BYZ)
	if = {
		limit = {
			NOT = { has_variable = oe_spqr_ev9_done }
			country_exists = c:BYZ
			NOT = { tag = BYZ }
			OR = {
				has_variable = oe_spqr_proclaimed
				AND = {
					owns_or_non_sovereign_subject_owns = location:rome
					OR = {
						owns_or_non_sovereign_subject_owns = location:athens
						owns_or_non_sovereign_subject_owns = location:thessaloniki
						owns_or_non_sovereign_subject_owns = location:candia
					}
				}
			}
		}
		set_variable = { name = oe_spqr_ev9_done value = 1 }
		trigger_event_non_silently = oe_spqr.9
	}
	# 6. Black Sea-Mediterranean Grain & Silk Pact (oe_spqr.12, TEU Pontus synergy!)
	if = {
		limit = {
			NOT = { has_variable = oe_spqr_ev12_done }
			OR = {
				AND = {
					has_global_variable = oe_tom_pontus_established
					OR = { has_variable = oe_spqr_proclaimed has_variable = oe_spqr_bank_merged tag = GEN tag = VEN }
				}
				AND = {
					OR = { owns_or_non_sovereign_subject_owns = location:venice owns_or_non_sovereign_subject_owns = location:genoa }

						owns_or_non_sovereign_subject_owns = location:kaffa

			}
		}

		trigger_event_non_silently = oe_spqr.12
	}
	# 7. Civitas Romana Golden Age (oe_spqr.13)
	if = {
		limit = {
			NOT = { has_variable = oe_spqr_ev13_done }
			has_variable = oe_spqr_proclaimed

			}

	}
