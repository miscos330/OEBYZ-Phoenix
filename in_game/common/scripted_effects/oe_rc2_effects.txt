	if = {
		limit = {
			current_year >= 1200
			NOT = { global_var:oe_rc2_ran_y ?= current_year }
		}
		set_global_variable = { name = oe_rc2_ran_y value = current_year }
		if = {
			limit = { NOT = { has_global_variable = oe_rc2_init } }
			set_global_variable = { name = oe_rc2_init value = 1 }
		}
		if = {
			limit = { NOT = { has_global_variable = oe_rc2_yearly_runs } }
			set_global_variable = { name = oe_rc2_yearly_runs value = 1 }
		}
		else = {
			change_global_variable = { name = oe_rc2_yearly_runs add = 1 }
		}
		if = {
			limit = {
				country_exists = c:BYZ
				c:BYZ = { is_rebel_country = no }
			}
			c:BYZ = {
				oe_rc2_byz_init = yes
				oe_rc2_yearly_supply_tick = yes
				oe_rc2_yearly_overextension_tick = yes
				oe_rc2_yearly_theater_events_tick = yes
				oe_rc2_yearly_digestion_tick = yes
			}
		}
	}
}

oe_rc2_yearly_supply_tick = {
	if = {
		limit = {
			OR = {
				oe_rc2_has_any_active_campaign = yes
				oe_rc2_has_active_expedition_subject = yes
			}
		}
		if = {
			limit = { is_ai = yes }
			if = { limit = { gold >= 1200 } set_variable = { name = oe_rc2_supply_tier value = 1 } }
			else_if = { limit = { gold >= 400 } set_variable = { name = oe_rc2_supply_tier value = 2 } }
			else = { set_variable = { name = oe_rc2_supply_tier value = 3 } }
		}
		oe_rc2_sync_supply_modifier = yes
		if = {
			limit = { var:oe_rc2_supply_tier ?= 1 }
			set_variable = { name = oe_rc2_austerity_years value = 0 }
			if = {
				limit = { oe_rc2_has_any_active_campaign = yes }
				oe_rc2_add_active_pressure = { amount = -5 }
			}
		}
		else_if = {
			limit = { var:oe_rc2_supply_tier ?= 3 }
			change_variable = { name = oe_rc2_austerity_years add = 1 }
			if = {
				limit = {
					has_variable = oe_rc2_austerity_years
					var:oe_rc2_austerity_years >= 2
					NOT = { has_variable = oe_rc2_belisarius_cd }
				}
				set_variable = { name = oe_rc2_belisarius_cd years = 5 }
				trigger_event_non_silently = oe_rc2.3
			}
		}
		else = {
			if = {
				limit = { has_variable = oe_rc2_austerity_years var:oe_rc2_austerity_years > 0 }
				change_variable = { name = oe_rc2_austerity_years add = -1 }
			}
		}
	}
	else = {
		oe_rc2_clear_supply_modifiers = yes
		if = {
			limit = { has_variable = oe_rc2_austerity_years var:oe_rc2_austerity_years > 0 }
			change_variable = { name = oe_rc2_austerity_years add = -1 }
		}
	}
}

oe_rc2_sync_overextension = {
	if = {
		limit = {
			has_variable = oe_rc2_active_fronts
			has_variable = oe_rc2_campaign_years
			OR = {
				AND = { var:oe_rc2_active_fronts >= 2 var:oe_rc2_campaign_years >= 5 }
				var:oe_rc2_campaign_years >= 9
			}
		}
		if = { limit = { has_country_modifier = oe_rc2_overextension_t1_mod } remove_country_modifier = oe_rc2_overextension_t1_mod }
		add_country_modifier = { modifier = oe_rc2_overextension_t2_mod years = 3 mode = replace }
	}
	else_if = {
		limit = {
			has_variable = oe_rc2_active_fronts
			has_variable = oe_rc2_campaign_years
			OR = {
				var:oe_rc2_active_fronts >= 2
				var:oe_rc2_campaign_years >= 5
			}
		}
		if = { limit = { has_country_modifier = oe_rc2_overextension_t2_mod } remove_country_modifier = oe_rc2_overextension_t2_mod }
		add_country_modifier = { modifier = oe_rc2_overextension_t1_mod years = 3 mode = replace }
	}
	else = {
		if = { limit = { has_country_modifier = oe_rc2_overextension_t1_mod } remove_country_modifier = oe_rc2_overextension_t1_mod }
		if = { limit = { has_country_modifier = oe_rc2_overextension_t2_mod } remove_country_modifier = oe_rc2_overextension_t2_mod }
	}
}

oe_rc2_yearly_overextension_tick = {
	set_variable = { name = oe_rc2_active_fronts value = 0 }
	if = { limit = { has_variable = oe_hc_egypt_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_africa_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_italy_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_france_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_iberia_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_britain_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_persia_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_blacksea_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { has_variable = oe_hc_arabia_active } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { oe_rc2_has_active_expedition_subject = yes } change_variable = { name = oe_rc2_active_fronts add = 1 } }
	if = { limit = { at_war = yes war_exhaustion >= 8 } change_variable = { name = oe_rc2_active_fronts add = 1 } }

	if = {
		limit = { has_variable = oe_rc2_active_fronts var:oe_rc2_active_fronts >= 1 }
		change_variable = { name = oe_rc2_campaign_years add = 1 }
		if = {
			limit = { has_variable = oe_rc2_active_fronts var:oe_rc2_active_fronts >= 2 }
			change_variable = { name = oe_rc2_campaign_years add = 1 }
		}
		clamp_variable = { name = oe_rc2_campaign_years min = 0 max = 25 }
	}
	else_if = {
		limit = { at_war = no has_variable = oe_rc2_campaign_years var:oe_rc2_campaign_years > 0 }
		change_variable = { name = oe_rc2_campaign_years add = -1 }
	}

	oe_rc2_sync_overextension = yes

	# AI consolidation when at peace with overextension fatigue.
	if = {
		limit = {
			is_ai = yes
			at_war = no
			NOT = { has_global_variable = oe_hc_any_active }
			NOT = { oe_rc2_has_active_expedition_subject = yes }
			NOT = { has_variable = oe_rc2_consolidate_cd }
			oe_rc2_has_overextension = yes
			gold >= 200
		}
		oe_rc2_do_consolidate_frontiers = yes
	}
}

	trigger_event_non_silently = oe_rc2.20
}

oe_rc2_apply_stage3_loc_buff = {
	if = {
		limit = { has_variable = oe_rc2_digest_theater }
		every_owned_location = {
			limit = { var:oe_hc_gain ?= root.var:oe_rc2_digest_theater }
			add_location_modifier = { modifier = oe_rc2_incorporated_province_loc_mod years = 25 mode = replace }
		}
	}
}

oe_rc2_yearly_digestion_tick = {
	if = {
		limit = { var:oe_rc2_digest_stage ?= 1 }
		change_variable = { name = oe_rc2_digest_years add = 1 }
		if = {
			limit = {
				has_variable = oe_rc2_digest_years
				OR = {
					var:oe_rc2_digest_years >= 10
					AND = { var:oe_rc2_digest_years >= 8 at_war = no stability >= 5 }
				}
			}
			trigger_event_non_silently = oe_rc2.21
		}
	}
	else_if = {
		limit = { var:oe_rc2_digest_stage ?= 2 }
		change_variable = { name = oe_rc2_digest_years add = 1 }
		if = {
			limit = {
				has_variable = oe_rc2_digest_years
				OR = {
					var:oe_rc2_digest_years >= 10
					AND = { var:oe_rc2_digest_years >= 8 at_war = no stability >= 10 }
				}
			}
			trigger_event_non_silently = oe_rc2.22
		}
	}
}

# =============================================================================
# 8. P19 LIVE ENGINE PROBE HOOK (STRICT CMDLOG=0, ZERO SIDE EFFECTS)
# Codes in var:oe_rc2_probe_p19 / global_var:oe_rc2_probe_p19_status:
#   1 = PASS (yearly hook, supply sync, overextension sync & CB dedup verified)
#   2 = SKIPPED (BYZ does not exist or is rebel)
#   3 = FAIL
# =============================================================================
oe_rc2_probe_p19 = {
	if = {
		limit = {
			country_exists = c:BYZ
			c:BYZ = { is_rebel_country = no }
		}
		oe_rc2_kernel_yearly = yes
		c:BYZ = {
			oe_rc2_byz_init = yes
			set_variable = { name = oe_rc2_p19_orig_tier value = var:oe_rc2_supply_tier }
			set_variable = { name = oe_rc2_p19_orig_years value = var:oe_rc2_campaign_years }
			set_variable = { name = oe_rc2_p19_orig_fronts value = var:oe_rc2_active_fronts }
			set_variable = { name = oe_rc2_p19_ok value = 0 }

			# Test 1: Supply doctrine tier 1 sync
			set_variable = { name = oe_rc2_supply_tier value = 1 }
			oe_rc2_sync_supply_modifier = yes
			if = {
				limit = { has_country_modifier = oe_rc2_supply_generous_overlord_mod }
				change_variable = { name = oe_rc2_p19_ok add = 1 }
			}

			# Test 2: Overextension tier 1 sync
			set_variable = { name = oe_rc2_active_fronts value = 2 }
			set_variable = { name = oe_rc2_campaign_years value = 4 }
			oe_rc2_sync_overextension = yes
			if = {
				limit = { has_country_modifier = oe_rc2_overextension_t1_mod }
				change_variable = { name = oe_rc2_p19_ok add = 1 }
			}

			# Test 3: Overextension tier 2 escalation (> 8 years)
			set_variable = { name = oe_rc2_campaign_years value = 9 }
			oe_rc2_sync_overextension = yes
			if = {
				limit = { has_country_modifier = oe_rc2_overextension_t2_mod }
				change_variable = { name = oe_rc2_p19_ok add = 1 }
			}

			# Restore original state cleanly
			set_variable = { name = oe_rc2_supply_tier value = var:oe_rc2_p19_orig_tier }
			set_variable = { name = oe_rc2_campaign_years value = var:oe_rc2_p19_orig_years }
			set_variable = { name = oe_rc2_active_fronts value = var:oe_rc2_p19_orig_fronts }
			oe_rc2_sync_overextension = yes
			if = {
				limit = {
					NOT = { oe_rc2_has_any_active_campaign = yes }
					NOT = { oe_rc2_has_active_expedition_subject = yes }
				}
				oe_rc2_clear_supply_modifiers = yes
			}
			else = {
				oe_rc2_sync_supply_modifier = yes
			}

			if = {
				limit = {
					has_global_variable = oe_rc2_init
					var:oe_rc2_p19_ok >= 3
				}
				set_global_variable = { name = oe_rc2_probe_p19_status value = 1 }
			}
			else = {
				set_global_variable = { name = oe_rc2_probe_p19_status value = 3 }
			}
		}
		if = {
			limit = { global_var:oe_rc2_probe_p19_status ?= 1 }
			set_variable = { name = oe_rc2_probe_p19 value = 1 }
		}
		else = {
			set_variable = { name = oe_rc2_probe_p19 value = 3 }
		}
	}
	else = {
		set_global_variable = { name = oe_rc2_probe_p19_status value = 2 }
		set_variable = { name = oe_rc2_probe_p19 value = 2 }
	}
}
