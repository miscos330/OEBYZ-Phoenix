# Same action context; new offices exist before any old office is destroyed.
oe_theme_convert_to_ordinary_effect = {
    scope:oe_theme_converted_country = {
        add_reform = government_reform:oe_byz_theme_system_subject
        set_capital = scope:oe_theme_new_capital
    }
    scope:oe_theme_new_capital = {
        construct_building = { building_type = building_type:oe_byz_theme_system_noble owner = scope:oe_theme_converted_country instant = yes cost_multiplier = 0 cost_multiplier_reason = "game_concept_event" }
    }
    every_in_list = {
        list = oe_theme_conversion_locations
        construct_building = { building_type = building_type:oe_byz_theme_system owner = scope:oe_theme_converted_country instant = yes cost_multiplier = 0 cost_multiplier_reason = "game_concept_event" }
    }
    scope:oe_theme_converted_country = {
        every_owned_building = {
            limit = {
                oe_theme_building_trigger = yes
                NOT = { OR = { building_type = building_type:oe_byz_theme_system_noble building_type = building_type:oe_byz_theme_system } }
            }
            location = { destroy_building = prev }
        }
    }
}
