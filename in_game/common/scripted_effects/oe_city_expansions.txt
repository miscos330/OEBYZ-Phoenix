# Location scope; iterate every owner, including Genoa and Venice.
oe_city_remove_foreign_quarters = {
    every_buildings_in_location = {
        limit = { OR = { building_type = building_type:galata building_type = building_type:venetian_quarter } }
        location = { destroy_building = prev }
    }
    set_variable = oe_city_foreign_quarters_removed
}
