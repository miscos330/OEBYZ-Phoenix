# Country scope. Advances Romanization milestone meter (0..100) and fires oe_as2.2 / .3 / .4.

	if = {
		limit = { oe_as2_has_any_romanization_active = yes }
		if = {
			limit = { NOT = { has_variable = oe_as2_roman_progress } }
			set_variable = { name = oe_as2_roman_progress value = 0 }
		}
		change_variable = { name = oe_as2_roman_progress add = 3 }
		if = {
			limit = { has_country_modifier = oe_as2_track_assimilation_mod }
			change_variable = { name = oe_as2_roman_progress add = 2 }
		}
		else_if = {
			limit = { has_country_modifier = oe_as2_track_foederati_mod }
			change_variable = { name = oe_as2_roman_progress add = -2 }
		}
		if = {
			limit = { has_country_modifier = oe_as2_romanization_extension_mod }
			change_variable = { name = oe_as2_roman_progress add = 2 }
		}
		if = {
			limit = { stability >= 25 }
			change_variable = { name = oe_as2_roman_progress add = 1 }
		}
		else_if = {
			limit = { stability < 0 }
			change_variable = { name = oe_as2_roman_progress add = -1 }
		}
		clamp_variable = { name = oe_as2_roman_progress min = 0 max = 100 }

		if = {
			limit = {
				NOT = { has_variable = oe_as2_ms_25_done }
				var:oe_as2_roman_progress >= 25
			}
			set_variable = { name = oe_as2_ms_25_done value = 1 }
			trigger_event_non_silently = oe_as2.2
		}
		else_if = {
			limit = {

# Country scope. CMDLOG=0 P20 Live Engine Probe Hook for Module D (Assimilation 2.0).
# Records integer pass bitmask in var:oe_probe2_p20:
#   +1    oe_as2_kernel_yearly executed and set global_var:oe_as2_ran_y
#   +10   global_var:oe_as2_init initialized & tick counter alive
#   +100  doctrine modifier round-trip verified (apply & clean restore if not already held)
# Full pass = 111.
oe_as2_probe_p20 = {
	set_variable = { name = oe_probe2_p20 value = 0 }
	oe_as2_kernel_yearly = yes
	if = {
		limit = { has_global_variable = oe_as2_ran_y }
		change_variable = { name = oe_probe2_p20 add = 1 }
	}
	if = {
		limit = {
			has_global_variable = oe_as2_init
			has_global_variable = oe_as2_tick_count
			global_var:oe_as2_tick_count >= 1
		}
		change_variable = { name = oe_probe2_p20 add = 10 }
	}
	if = {
		limit = {
			OR = {
				has_country_modifier = oe_as2_track_assimilation_mod
				has_country_modifier = oe_as2_track_foederati_mod
			}
		}
		change_variable = { name = oe_probe2_p20 add = 100 }
	}
	else = {
		add_country_modifier = { modifier = oe_as2_track_assimilation_mod years = 1 mode = replace }
		if = {
			limit = { has_country_modifier = oe_as2_track_assimilation_mod }
			remove_country_modifier = oe_as2_track_assimilation_mod
			change_variable = { name = oe_probe2_p20 add = 100 }
		}
	}
}
