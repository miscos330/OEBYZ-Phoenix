# IO scope. Removal reuses normal house cleanup, including holdings and any political alliance.
oe_nf_remove_ruling_house = {
    if = {
        limit = { c:BYZ = { NOT = { has_variable = oe_bi_coup_in_progress } } has_variable = oe_nf_dynasty_$slot$ var:oe_nf_dynasty_$slot$ = { oe_nf_is_ruling_dynasty = yes } }
        oe_nf_unregister_house = { slot = $slot$ }
    }
}
# BYZ scope. Opening friendship is granted directly, with no legitimacy charge.
oe_nf_opening_kantakouzenos_ally = {
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    oe_nf_initialize = yes
	    oe_nf_refresh = yes
	    international_organization:oe_byz_empire_pillars = {
	        if = {
	            limit = { dynasty:kantakouzenos_dynasty = { oe_nf_registered_family = yes } }
	            oe_nf_clear_political_ally = yes
	            set_variable = { name = oe_nf_political_ally value = dynasty:kantakouzenos_dynasty }
	            if = { limit = { has_variable = oe_nf_dynasty_1 var:oe_nf_dynasty_1 = dynasty:kantakouzenos_dynasty } set_variable = oe_nf_ally_1 }
	            if = { limit = { has_variable = oe_nf_dynasty_2 var:oe_nf_dynasty_2 = dynasty:kantakouzenos_dynasty } set_variable = oe_nf_ally_2 }
	            if = { limit = { has_variable = oe_nf_dynasty_3 var:oe_nf_dynasty_3 = dynasty:kantakouzenos_dynasty } set_variable = oe_nf_ally_3 }
	            if = { limit = { has_variable = oe_nf_dynasty_4 var:oe_nf_dynasty_4 = dynasty:kantakouzenos_dynasty } set_variable = oe_nf_ally_4 }
	        }
	    }
	    oe_nf_refresh = yes
	}
}
# BYZ scope. The delayed on_action checks regency after the death/succession callbacks have settled.
oe_nf_note_andronikos_death = {
    if = {
        limit = { NOT = { has_variable = oe_nf_andronikos_iii_died } }
        set_variable = oe_nf_andronikos_iii_died
        trigger_event_non_silently = { on_action = oe_nf_after_andronikos_death days = 1 }
    }
}
