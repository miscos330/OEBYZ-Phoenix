oe_rw_start_georgia_romanization = {
    if = {
        limit = { oe_rw_start_georgia_romanization_allowed = yes }
        area:georgia_area = {
            ordered_location_in_area = {
                limit = { oe_rw_georgia_origin_ge25 = yes }
                max = 1
                check_range_bounds = no
                order_by = {
                    value = population
                    multiply = oe_rw_assim_georgia_source_share
                    if = { limit = { oe_rw_georgia_origin_ge50 = yes } add = 10000000 }
                    if = { limit = { this = location:tbilisi } add = 1000000 }
                    if = { limit = { owner = c:BYZ } add = 100000000 }
                }
                save_scope_as = oe_rw_origin_loc
                spawn_movement = {
                    movement_definition = movement_definition:oe_georgia_identity_movement
                    supporters = { value = population multiply = oe_rw_assim_georgia_source_share multiply = 0.1 }
                }
                if = {
                    limit = {
                        culture:oe_georgian_greek_culture = {
                            any_movement_in_culture = {
                                movement_type = movement_definition:oe_georgia_identity_movement
                                num_locations_affected > 0
                            }
                        }
                    }
                    oe_identity_confirm_start = { holder = culture:oe_georgian_greek_culture movement = oe_georgia_identity_movement event = oe_eastern_identity.1 }
                    c:BYZ = { set_variable = { name = oe_rw_origin_georgia value = scope:oe_rw_origin_loc } }
                }
            }
        }
    }
}
oe_rw_start_arabia_romanization = {
    if = {
        limit = { NOT = { oe_rw_arabia_romanization_started = yes } }
        region:arabia_region = {
            ordered_location_in_region = {
                limit = { oe_rw_arabia_spread_location = yes }
                order_by = { value = population multiply = oe_rw_assim_arabia_source_share }
                max = 1 check_range_bounds = no
                spawn_movement = {
                    movement_definition = movement_definition:oe_arabia_identity_movement
                    supporters = { value = population multiply = oe_rw_assim_arabia_source_share multiply = 0.1 }
                }
                oe_identity_confirm_start = { holder = culture:oe_arabian_greek_culture movement = oe_arabia_identity_movement event = oe_eastern_identity.2 }
            }
        }
    }
}
