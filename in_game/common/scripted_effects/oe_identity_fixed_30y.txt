# A one-shot 30-year timer for each unique target culture/religion.
# Scheduled on a permanent location, independent of country survival and movement ticks.
# The marker only prevents duplicate scheduling; there are no early-end conditions.
oe_identity_schedule_30_years = {
    $holder$ = {
        if = {
            limit = { NOT = { has_variable = oe_identity_fixed_30y_scheduled } }
            set_variable = oe_identity_fixed_30y_scheduled
            location:constantinople = {
                trigger_event_silently = { id = $event$ years = 30 }
            }
        }
    }
}
