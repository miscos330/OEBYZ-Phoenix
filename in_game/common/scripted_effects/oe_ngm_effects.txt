# Napoleonic Anti-Hegemon Coalition & Vienna Rebalance Yearly Engine (1750-1836)

oe_ngm_coalition_yearly_tick = {
	# Step 1 (1750+): Identify dominant Great Power & spark Anti-Hegemon Coalition Loop (oe_ngm.1)
	if = {
		limit = {
			current_year >= 1750
			NOT = { has_global_variable = oe_ngm_ev1_done }
			any_great_power = { oe_ngm_is_dominant_hegemon_candidate = yes }
		}
		set_global_variable = { name = oe_ngm_ev1_done value = 1 }
		set_global_variable = { name = oe_ngm_coalition_active value = 1 }
		set_global_variable = { name = oe_ngm_coalition_wave value = 1 }
		if = {
			limit = { NOT = { has_global_variable = oe_ngm_coalition_wars_count } }
			set_global_variable = { name = oe_ngm_coalition_wars_count value = 0 }
		}
		ordered_great_power = {
			limit = { oe_ngm_is_dominant_hegemon_candidate = yes }
			order_by = great_power_score
			max = 1
			check_range_bounds = no
			save_scope_as = oe_ngm_tmp_heg
			set_global_variable = { name = oe_ngm_hegemon_target value = scope:oe_ngm_tmp_heg }
		}
		if = {
			limit = { has_global_variable = oe_heg2_tension }
			change_global_variable = { name = oe_heg2_tension add = 15 }
			clamp_global_variable = { name = oe_heg2_tension min = 0 max = 100 }
		}
		every_great_power = {
			limit = { is_rebel_country = no is_subject = no }
			trigger_event_non_silently = { id = oe_ngm.1 days = 5 }
		}
	}

	# Step 2 (1755+): Continental Blockade proclaimed by dominant Great Power (oe_ngm.2)
	if = {
		limit = {
			current_year >= 1755
			has_global_variable = oe_ngm_coalition_active
			NOT = { has_global_variable = oe_ngm_ev2_done }
			has_global_variable = oe_ngm_hegemon_target
			country_exists = global_var:oe_ngm_hegemon_target
		}
		set_global_variable = { name = oe_ngm_ev2_done value = 1 }
		set_global_variable = { name = oe_ngm_blockade_active value = 1 }
		global_var:oe_ngm_hegemon_target = {
			trigger_event_non_silently = { id = oe_ngm.2 days = 10 }
		}
	}

	# Step 3 (1760+): Smuggling Networks & Neutral Maritime Rights (oe_ngm.3)
	if = {
		limit = {
			current_year >= 1760
			has_global_variable = oe_ngm_blockade_active
			NOT = { has_global_variable = oe_ngm_ev3_done }
		}
		set_global_variable = { name = oe_ngm_ev3_done value = 1 }
		every_great_power = {
			limit = { oe_ngm_is_coalition_contender = yes has_ports = yes }
			trigger_event_non_silently = { id = oe_ngm.3 days = 12 }
		}
	}

	# Step 4 (1765+): First Anti-Hegemon Coalition Wave (oe_ngm.4) + native Pact creation
	if = {
		limit = {
			current_year >= 1765
			has_global_variable = oe_ngm_coalition_active
			NOT = { has_global_variable = oe_ngm_ev4_done }
		}
		set_global_variable = { name = oe_ngm_ev4_done value = 1 }
		set_global_variable = { name = oe_ngm_coalition_wave value = 1 }
		change_global_variable = { name = oe_ngm_coalition_wars_count add = 1 }
		ordered_great_power = {
			limit = { oe_ngm_is_coalition_contender = yes }
			order_by = great_power_score
			max = 1
			check_range_bounds = no
			save_scope_as = oe_ngm_coal_lead
			set_global_variable = { name = oe_ngm_coalition_leader value = scope:oe_ngm_coal_lead }
		}
		if = {
			limit = { has_global_variable = oe_ngm_coalition_leader country_exists = global_var:oe_ngm_coalition_leader }
			ordered_great_power = {
				limit = {
					oe_ngm_is_coalition_contender = yes
					NOT = { global_var:oe_ngm_coalition_leader ?= this }
					NOT = { is_at_war_with = global_var:oe_ngm_coalition_leader }
				}
				order_by = great_power_score
				max = 1
				check_range_bounds = no
				save_scope_as = oe_ngm_coal_partner
				create_relation = {
					first = global_var:oe_ngm_coalition_leader
					second = scope:oe_ngm_coal_partner
					type = relation_type:oe_ngm_anti_hegemon_pact
				}
			}
		}
		every_great_power = {
			limit = { oe_ngm_is_coalition_contender = yes }
			trigger_event_non_silently = { id = oe_ngm.4 days = 15 }
		}
	}

		}
		set_global_variable = { name = oe_ngm_ev5_done value = 1 }
