					NOT = { this = root }
					NOT = { is_subject_or_below_of = root }
				}
			}
		}
		location:$LOC$ = {
			owner = { save_scope_as = oe_el2_cb_tgt }

		if = {
			limit = {

			}
			add_casus_belli = {
				target = scope:oe_el2_cb_tgt
				type = casus_belli:oe_elr_cb_nostos_reclamation

	}

# Unified Yearly Tick Dispatcher (called by Stage Integrator's oe_v136_pulse.txt)
# =============================================================================
oe_el2_yearly_tick = {
	if = {
		limit = {
			is_ai = no
			is_rebel_country = no
			has_global_variable = oe_ely_settled
			global_var:oe_ely_settled ?= 1
			oe_ely_is_elysian_host = yes
		}
		oe_el2_sync_exarchate_counts = yes

		# At most one narrative event per yearly tick (zero spam, O(1) prioritized chain).
		if = {
			limit = { oe_el2_reunion_ready = yes }
			set_variable = { name = oe_el2_reunion_done value = 1 }
			trigger_event_non_silently = oe_el2.2
		}
		else_if = {
			limit = {
				NOT = { has_variable = oe_el2_silk_unlocked }
				OR = {
					num_locations >= 6
					current_year >= 1400
				}
			}
			set_variable = { name = oe_el2_silk_unlocked value = 1 }
			trigger_event_non_silently = oe_el2.13
		}
		else_if = {
			limit = {
				NOT = { has_variable = oe_el2_arsenal_unlocked }
				OR = {
					has_variable = oe_eln_pyrophoroi_ready
					has_variable = oe_eln_copper_galleon_ready
					current_year >= 1430
				}
			}
			set_variable = { name = oe_el2_arsenal_unlocked value = 1 }
			trigger_event_non_silently = oe_el2.14
		}
		else_if = {
			limit = { oe_el2_vinland_spawn_ready = yes }
			oe_el2_spawn_vinland = yes
			trigger_event_non_silently = oe_el2.5
		}
		else_if = {
			limit = {
				oe_el2_vinland_alive = yes
