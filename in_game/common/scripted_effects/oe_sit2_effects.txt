oe_sit2_yr_init = {
	set_global_variable = { name = oe_sit2_yr_started value = 1 }
	# Bridge with the inner Zhizheng Restoration track (oe_yr_*) so both panels stay consistent (zero debug_log).
	if = {
		limit = {
			country_exists = c:CHI
			NOT = { has_global_variable = oe_yr_end }
		}
		if = {
			limit = { NOT = { has_global_variable = oe_yr_on } }
			set_global_variable = oe_yr_init
			set_global_variable = oe_yr_on
			set_global_variable = { name = oe_yr_path value = 0 }
			set_global_variable = { name = oe_yr_stage value = 0 }
			set_global_variable = { name = oe_yr_strain value = 0 }
			set_global_variable = { name = oe_yr_hold value = 0 }
		}
		if = {
			limit = { NOT = { has_global_variable = oe_yr_opened } }
			set_global_variable = oe_yr_opened
			if = { limit = { NOT = { has_global_variable = oe_yr_how } } set_global_variable = { name = oe_yr_how value = 4 } }
			if = { limit = { NOT = { has_global_variable = oe_yr_open_m } } set_global_variable = { name = oe_yr_open_m value = oe_sit2_m_now } }
			if = { limit = { NOT = { has_global_variable = oe_yr_lit0 } } c:CHI = { set_global_variable = { name = oe_yr_lit0 value = average_country_literacy } } }
		}
	}
	situation:oe_sit_yuan_restoration = {
		set_variable = { name = oe_sit2_yr_progress value = 25 }
		set_variable = { name = oe_sit2_yr_balance value = 50 }
		set_variable = { name = oe_sit2_yr_months value = 0 }
		set_variable = { name = oe_sit2_yr_strain value = 0 }
		set_variable = { name = oe_sit2_yr_stage_cache value = 0 }
	}
	oe_sit2_yr_sync = yes
}

oe_sit2_yr_sync = {
	if = {
		limit = { is_situation_active = situation:oe_sit_yuan_restoration }
		situation:oe_sit_yuan_restoration = {
			if = { limit = { NOT = { has_variable = oe_sit2_yr_progress } } set_variable = { name = oe_sit2_yr_progress value = 25 } }
			if = { limit = { NOT = { has_variable = oe_sit2_yr_balance } } set_variable = { name = oe_sit2_yr_balance value = 50 } }
			if = { limit = { NOT = { has_variable = oe_sit2_yr_months } } set_variable = { name = oe_sit2_yr_months value = 0 } }
			if = { limit = { NOT = { has_variable = oe_sit2_yr_strain } } set_variable = { name = oe_sit2_yr_strain value = 0 } }
			if = { limit = { NOT = { has_variable = oe_sit2_yr_stage_cache } } set_variable = { name = oe_sit2_yr_stage_cache value = 0 } }

			# Mirror live oe_yr_* global state when Great Yuan Restoration line is active.
			if = {
				limit = { has_global_variable = oe_yr_strain }
				set_variable = { name = oe_sit2_yr_strain value = global_var:oe_yr_strain }
			}
			if = {
				limit = { has_global_variable = oe_yr_stage }
				set_variable = { name = oe_sit2_yr_stage_cache value = global_var:oe_yr_stage }
				if = {
					limit = { global_var:oe_yr_stage >= 3 var:oe_sit2_yr_progress < 75 }
					set_variable = { name = oe_sit2_yr_progress value = 75 }
				}
				else_if = {
					limit = { global_var:oe_yr_stage >= 2 var:oe_sit2_yr_progress < 50 }
					set_variable = { name = oe_sit2_yr_progress value = 50 }
				}
				else_if = {
					limit = { global_var:oe_yr_stage >= 1 var:oe_sit2_yr_progress < 30 }
					set_variable = { name = oe_sit2_yr_progress value = 30 }
				}
			}
			if = {
				limit = { has_global_variable = oe_yr_path }
				if = {
					limit = { global_var:oe_yr_path = 1 var:oe_sit2_yr_balance > 45 }
					change_variable = { name = oe_sit2_yr_balance add = -2 }
				}
				else_if = {
					limit = { global_var:oe_yr_path = 2 var:oe_sit2_yr_balance < 55 }
					change_variable = { name = oe_sit2_yr_balance add = 2 }
				}
			}
			clamp_variable = { name = oe_sit2_yr_progress max = 100 min = 0 }
			clamp_variable = { name = oe_sit2_yr_balance max = 100 min = 0 }
		}
	}
}

oe_sit2_check_bridges = {
	# Bridge 1: Vanilla Red Turban Rebellions <-> Great Yuan Restoration (v14.5: delayed 6 months / 180d into RTR)
	if = {
		limit = {
			NOT = { has_global_variable = oe_sit2_br_rtr }
			oe_sit2_bridge_rtr_active = yes
			situation:red_turban_rebellions = { days_since_situation_start >= 180 }
			country_exists = c:CHI
		}
		set_global_variable = { name = oe_sit2_br_rtr value = 1 }
		c:CHI = { trigger_event_non_silently = oe_sit2.21 }
	}
	# Bridge 2: Vanilla Rise of Timur <-> BYZ / GLH / CHI cross-line shock
	if = {
		limit = {
			NOT = { has_global_variable = oe_sit2_br_timur }
			oe_sit2_bridge_timur_active = yes
		}
		set_global_variable = { name = oe_sit2_br_timur value = 1 }
		if = { limit = { country_exists = c:BYZ } c:BYZ = { trigger_event_non_silently = oe_sit2.22 } }
		if = { limit = { country_exists = c:GLH } c:GLH = { trigger_event_non_silently = oe_sit2.22 } }
		if = { limit = { country_exists = c:CHI } c:CHI = { trigger_event_non_silently = oe_sit2.22 } }
	}
	# Bridge 3: Vanilla Western Schism <-> BYZ / ROM Papal Diplomacy
	if = {
		limit = {
			NOT = { has_global_variable = oe_sit2_br_schism }
			oe_sit2_bridge_schism_active = yes
			OR = { country_exists = c:BYZ country_exists = c:ROM }
		}
		set_global_variable = { name = oe_sit2_br_schism value = 1 }
		if = { limit = { country_exists = c:ROM } c:ROM = { trigger_event_non_silently = oe_sit2.23 } }
		else_if = { limit = { country_exists = c:BYZ } c:BYZ = { trigger_event_non_silently = oe_sit2.23 } }
	}
	# Bridge 4: Vanilla Italian Wars <-> BYZ / ROM Italian Theatre Presence
	if = {
		limit = {
			NOT = { has_global_variable = oe_sit2_br_iw }
			oe_sit2_bridge_iw_active = yes
			OR = { country_exists = c:BYZ country_exists = c:ROM }
		}
		set_global_variable = { name = oe_sit2_br_iw value = 1 }
		if = { limit = { country_exists = c:ROM } c:ROM = { trigger_event_non_silently = oe_sit2.24 } }
		else_if = { limit = { country_exists = c:BYZ } c:BYZ = { trigger_event_non_silently = oe_sit2.24 } }
	}
}
