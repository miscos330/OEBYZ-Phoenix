# Location scope. Clears Bailiwick seat location modifiers cleanly.
oe_thh_clear_seat_modifiers = {
	if = { limit = { has_location_modifier = oe_thh_seat_loyal_mod } remove_location_modifier = oe_thh_seat_loyal_mod }
	if = { limit = { has_location_modifier = oe_thh_seat_restive_mod } remove_location_modifier = oe_thh_seat_restive_mod }
}

# Bailiwick country scope. Clears subject loyalty tier modifiers cleanly.
oe_thh_clear_ballei_modifiers = {
	if = { limit = { has_country_modifier = oe_thh_ballei_compliant_mod } remove_country_modifier = oe_thh_ballei_compliant_mod }
	if = { limit = { has_country_modifier = oe_thh_ballei_restive_mod } remove_country_modifier = oe_thh_ballei_restive_mod }
}

# Bailiwick country scope. Shifts oe_thh_loyalty by $amount$ (clamped 0..100).
oe_thh_shift_loyalty = {
	if = { limit = { NOT = { has_variable = oe_thh_loyalty } } set_variable = { name = oe_thh_loyalty value = 60 } }
	change_variable = { name = oe_thh_loyalty add = $amount$ }
	clamp_variable = { name = oe_thh_loyalty min = 0 max = 100 }
}

# Bailiwick country scope. Syncs oe_thh_tier (3>=70, 2=40..69, 1<40) and country tier modifiers.
oe_thh_sync_self_tier = {
	if = { limit = { NOT = { has_variable = oe_thh_loyalty } } set_variable = { name = oe_thh_loyalty value = 60 } }
	if = {
		limit = { has_variable = oe_thh_loyalty var:oe_thh_loyalty >= 70 }
		set_variable = { name = oe_thh_tier value = 3 }
		oe_thh_clear_ballei_modifiers = yes
		add_country_modifier = { modifier = oe_thh_ballei_compliant_mod months = 14 mode = replace }
	}
	else_if = {
		limit = { has_variable = oe_thh_loyalty var:oe_thh_loyalty >= 40 }
		set_variable = { name = oe_thh_tier value = 2 }
		oe_thh_clear_ballei_modifiers = yes
	}
	else = {
		set_variable = { name = oe_thh_tier value = 1 }
		oe_thh_clear_ballei_modifiers = yes
		add_country_modifier = { modifier = oe_thh_ballei_restive_mod months = 14 mode = replace }
	}
}

# Any scope. F6-style clean dissolution of OTB1 (Ballei Bohmen): annex via overlord or destroy_building_country, then clear seat.
oe_thh_end_OTB1 = {
	set_global_variable = { name = oe_thh_ended_otb1 value = 1 }
	if = { limit = { exists = scope:oe_thh_end_lord_OTB1 } clear_saved_scope = oe_thh_end_lord_OTB1 }
	if = {
		limit = { country_exists = c:OTB1 }
		c:OTB1 = {
			set_variable = { name = oe_thh_ended value = 1 }
			if = { limit = { is_subject = yes } overlord = { save_scope_as = oe_thh_end_lord_OTB1 } }
		}
		if = {
			limit = { exists = scope:oe_thh_end_lord_OTB1 }
			scope:oe_thh_end_lord_OTB1 = { annex_country = { country = c:OTB1 reason = Diplomatic transfer = yes } }
		}
		else = {
			destroy_building_country = c:OTB1
		}
	}
	location:prague = {
		if = { limit = { has_building = building_type:oe_thh_ballei_seat_1 } destroy_all_buildings_of_type = building_type:oe_thh_ballei_seat_1 }
		oe_thh_clear_seat_modifiers = yes
	}
}

			if = {

# TEU country scope. Shifts loyalty of all active Bailiwick subjects by $amount$.

		oe_thh_sync_self_tier = yes
	}
}

# Any scope. Paced yearly event dispatcher for Teutonic HRE Statecraft (oe_thh.1..8; .9/.10 via decisions, 0 orphans).
oe_thh_dispatch_events = {
	if = {
		limit = { country_exists = c:TEU c:TEU = { oe_thh_is_teutonic_host = yes NOT = { has_variable = oe_thh_ev_cd } } }
		c:TEU = {
			if = {
				limit = { NOT = { has_global_variable = oe_thh_ev1_done } current_year >= 1340 exists = international_organization:hre international_organization:hre = { international_organization_has_leader = yes } }
				set_global_variable = { name = oe_thh_ev1_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.1
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev3_done } current_year >= 1342 country_exists = c:OTB4 c:OTB4 = { is_subject_of = c:TEU } }
				set_global_variable = { name = oe_thh_ev3_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.3
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev4_done } current_year >= 1345 OR = { AND = { country_exists = c:OTB1 c:OTB1 = { is_subject_of = c:TEU } } AND = { country_exists = c:OTB3 c:OTB3 = { is_subject_of = c:TEU } } } }
				set_global_variable = { name = oe_thh_ev4_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.4
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev2_done } current_year >= 1348 }
				set_global_variable = { name = oe_thh_ev2_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.2
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev6_done } current_year >= 1352 oe_thh_has_any_ballei = yes }
				set_global_variable = { name = oe_thh_ev6_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.6
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev5_done } current_year >= 1355 }
				set_global_variable = { name = oe_thh_ev5_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }
				trigger_event_non_silently = oe_thh.5
			}
			else_if = {
				limit = { NOT = { has_global_variable = oe_thh_ev7_done } current_year >= 1362 }
				set_global_variable = { name = oe_thh_ev7_done value = 1 }
				set_variable = { name = oe_thh_ev_cd value = 1 years = 3 }

# Any scope. Self-deduplicating yearly entrypoint (called via oe_thh_on_actions and safe for global kernel).
oe_thh_kernel_yearly = {
	if = {
		limit = { NOT = { global_var:oe_thh_ran_y ?= current_year } }
		set_global_variable = { name = oe_thh_ran_y value = current_year }
		if = {
			limit = { NOT = { has_global_variable = oe_thh_init_done } }
			oe_thh_init_once = yes
			set_global_variable = { name = oe_thh_init_done value = 1 }
		}
		oe_thh_cleanup_check = yes
		oe_thh_sync_all_balleien = yes
		oe_thh_dispatch_events = yes
	}
}
