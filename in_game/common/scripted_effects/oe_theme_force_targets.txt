oe_tfr_spawn_li = {
    if = {
        limit = { has_variable = oe_tfr_li_type }
        set_variable = { name = oe_tfr_step_size value = { value = var:oe_tfr_li_resource multiply = 1000 } }
        if = {
            limit = {
                gold >= 1000
                gold >= var:oe_tfr_li_gold
                manpower >= var:oe_tfr_li_resource
                var:oe_tfr_current <= { value = var:oe_tfr_target subtract = var:oe_tfr_step_size }
            }
            add_gold = { value = var:oe_tfr_li_gold multiply = -1 }
            add_manpower = { value = var:oe_tfr_li_resource multiply = -1 }
            capital = {
                if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 1 } }
                    create_sub_unit_with_owner = { type = a_sharpshooters owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 2 } }
                    create_sub_unit_with_owner = { type = a_hunters owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 3 } }
                    create_sub_unit_with_owner = { type = a_arquebusiers owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 4 } }
                    create_sub_unit_with_owner = { type = a_early_arquebusiers owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 5 } }
                    create_sub_unit_with_owner = { type = a_crossbowmen owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 6 } }
                    create_sub_unit_with_owner = { type = a_handgonners owner = scope:oe_tfr_owner }
                }
                else_if = {
                    limit = { scope:oe_tfr_owner = { var:oe_tfr_li_type = 7 } }
                    create_sub_unit_with_owner = { type = a_archers owner = scope:oe_tfr_owner }
                }
            }
            change_variable = { name = oe_tfr_current add = var:oe_tfr_step_size }
            set_variable = { name = oe_tfr_failures value = 0 }
            change_variable = { name = oe_tfr_cursor add = 1 }
        }
        else = { set_variable = { name = oe_tfr_stop value = 1 } }
    }
    else = {
        change_variable = { name = oe_tfr_failures add = 1 }
        change_variable = { name = oe_tfr_cursor add = 1 }
    }
}

oe_tfr_reinforce_country = {
    if = {
        limit = { oe_tfr_eligible_theme = yes gold >= 1000 oe_tfr_target_display > 0 capital = { owner = c:BYZ } }
        save_scope_as = oe_tfr_owner
        set_variable = { name = oe_tfr_target value = oe_tfr_target_display }
        # Clear transient selection values before each yearly attempt.
        remove_variable = oe_tfr_li_type
        remove_variable = oe_tfr_li_resource
        remove_variable = oe_tfr_li_gold
        remove_variable = oe_tfr_hi_type
        remove_variable = oe_tfr_hi_resource
        remove_variable = oe_tfr_hi_gold
        remove_variable = oe_tfr_lc_type
        remove_variable = oe_tfr_lc_resource
        remove_variable = oe_tfr_lc_gold
        remove_variable = oe_tfr_hc_type
        remove_variable = oe_tfr_hc_resource
        remove_variable = oe_tfr_hc_gold
        remove_variable = oe_tfr_ar_type
        remove_variable = oe_tfr_ar_resource
        remove_variable = oe_tfr_ar_gold
        remove_variable = oe_tfr_hs_type
        remove_variable = oe_tfr_hs_resource
        remove_variable = oe_tfr_hs_gold
        remove_variable = oe_tfr_ga_type
        remove_variable = oe_tfr_ga_resource
        remove_variable = oe_tfr_ga_gold
        set_variable = { name = oe_tfr_current value = oe_tfr_current_display }
        set_variable = { name = oe_tfr_failures value = 0 }
        set_variable = { name = oe_tfr_stop value = 0 }
        if = { limit = { NOT = { has_variable = oe_tfr_land_cursor } } set_variable = { name = oe_tfr_land_cursor value = 0 } }
        if = { limit = { NOT = { has_variable = oe_tfr_naval_cursor } } set_variable = { name = oe_tfr_naval_cursor value = 0 } }
        if = {
            limit = { oe_tfr_land_theme = yes var:oe_tfr_current < var:oe_tfr_target }
            if = { limit = { var:oe_tfr_target > 100000 } set_variable = { name = oe_tfr_target value = 100000 } }
            set_variable = { name = oe_tfr_cursor value = var:oe_tfr_land_cursor }
            oe_tfr_pick_li = yes
            oe_tfr_pick_hi = yes
            oe_tfr_pick_lc = yes
            oe_tfr_pick_hc = yes
            oe_tfr_pick_ar = yes
            while = {
                limit = { var:oe_tfr_current < var:oe_tfr_target gold >= 1000 var:oe_tfr_stop = 0 var:oe_tfr_failures < 5 }
                if = { limit = { var:oe_tfr_cursor >= 5 } set_variable = { name = oe_tfr_cursor value = 0 } }
                if = { limit = { var:oe_tfr_cursor = 0 } oe_tfr_spawn_li = yes }
                else_if = { limit = { var:oe_tfr_cursor = 1 } oe_tfr_spawn_hi = yes }
                else_if = { limit = { var:oe_tfr_cursor = 2 } oe_tfr_spawn_lc = yes }
                else_if = { limit = { var:oe_tfr_cursor = 3 } oe_tfr_spawn_hc = yes }
                else_if = { limit = { var:oe_tfr_cursor = 4 } oe_tfr_spawn_ar = yes }
            }
            set_variable = { name = oe_tfr_land_cursor value = var:oe_tfr_cursor }
        }
        else_if = {
            limit = { oe_tfr_naval_theme = yes var:oe_tfr_current < var:oe_tfr_target capital = { is_coastal = yes } }
            set_variable = { name = oe_tfr_cursor value = var:oe_tfr_naval_cursor }
            oe_tfr_pick_hs = yes
            oe_tfr_pick_ga = yes
            while = {
                limit = { var:oe_tfr_current < var:oe_tfr_target gold >= 1000 var:oe_tfr_stop = 0 var:oe_tfr_failures < 4 }
                if = { limit = { var:oe_tfr_cursor >= 4 } set_variable = { name = oe_tfr_cursor value = 0 } }
                if = { limit = { var:oe_tfr_cursor = 0 } oe_tfr_spawn_hs = yes }
                else = { oe_tfr_spawn_ga = yes }
            }
            set_variable = { name = oe_tfr_naval_cursor value = var:oe_tfr_cursor }
        }
        remove_variable = oe_tfr_li_type
        remove_variable = oe_tfr_li_resource
        remove_variable = oe_tfr_li_gold
        remove_variable = oe_tfr_hi_type
        remove_variable = oe_tfr_hi_resource
        remove_variable = oe_tfr_hi_gold
        remove_variable = oe_tfr_lc_type
        remove_variable = oe_tfr_lc_resource
        remove_variable = oe_tfr_lc_gold
        remove_variable = oe_tfr_hc_type
        remove_variable = oe_tfr_hc_resource
        remove_variable = oe_tfr_hc_gold
        remove_variable = oe_tfr_ar_type
        remove_variable = oe_tfr_ar_resource
        remove_variable = oe_tfr_ar_gold
        remove_variable = oe_tfr_hs_type
        remove_variable = oe_tfr_hs_resource
        remove_variable = oe_tfr_hs_gold
        remove_variable = oe_tfr_ga_type
        remove_variable = oe_tfr_ga_resource
        remove_variable = oe_tfr_ga_gold
        remove_variable = oe_tfr_current
        remove_variable = oe_tfr_failures
        remove_variable = oe_tfr_stop
        remove_variable = oe_tfr_cursor
        remove_variable = oe_tfr_step_size
        remove_variable = oe_tfr_target
    }
}
