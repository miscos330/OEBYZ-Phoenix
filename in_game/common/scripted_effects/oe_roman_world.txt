# International organization scope. Never recreate a defeated member or reset resources.
oe_roman_world_sync_effect = {
    save_scope_as = oe_roman_world_org
    # The owner, not the occupier, of Constantinople carries the Roman claim.
    # can_lead_trigger independently prevents any other country from leading.
    location:constantinople.owner ?= {
        if = {
            limit = { oe_roman_world_territorial_country_trigger = yes }
            save_scope_as = oe_roman_world_capital_owner
            if = {
                limit = { NOT = { is_member_of_international_organization = scope:oe_roman_world_org } }
                scope:oe_roman_world_org = { add_country_to_international_organization = scope:oe_roman_world_capital_owner }
            }
            scope:oe_roman_world_org = {
                if = {
                    limit = { OR = { international_organization_has_leader = no leader_country ?= { NOT = { owns = location:constantinople } } } }
                    set_leader_country = scope:oe_roman_world_capital_owner
                }
            }
        }
    }
    # Rescind the unique claim before assigning it, so the one-country cap never blocks succession.
    every_international_organization_member = {
        if = {
            limit = { NOT = { oe_roman_world_byz_trigger = yes } }
            save_scope_as = oe_roman_world_member
            scope:oe_roman_world_org = {
                if = {
                    limit = { country_has_special_status = { type = special_status:oe_byz_roman_world_byz country = scope:oe_roman_world_member } }
                    international_organization_remove_special_status = { type = special_status:oe_byz_roman_world_byz country = scope:oe_roman_world_member }
                }
            }
        }
    }
    every_international_organization_member = {
        if = {
            limit = { NOT = { oe_roman_world_territorial_country_trigger = yes } }
            save_scope_as = oe_roman_world_member
            scope:oe_roman_world_org = { remove_country_from_international_organization = scope:oe_roman_world_member }
        }
        else = { oe_roman_world_classify_effect = yes }
    }
    oe_roman_world_recalculate_contest = yes
    oe_roman_world_ensure_contest_tick = yes

    oe_heg_sync_campaign_territory = yes
    oe_city_initialize_catalog = yes
}
