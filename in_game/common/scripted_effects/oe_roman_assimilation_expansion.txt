oe_rw_start_egypt_romanization = {
    if = {
        limit = { oe_rw_start_egypt_romanization_allowed = yes }
        set_variable = oe_rw_egypt_romanization_started
        ordered_ownable_location = {
            limit = { oe_rw_assim_egypt_location = yes is_ownable = yes has_owner = yes owner = { oe_heg_byz_bloc = yes } }
            max = 1
            order_by = {
                value = population
                # Strongly prefer the historical centre, if it passes the ownership/region limit.
                if = { limit = { this = location:alexandria } add = 1000000 }
                # Preserve the existing direct-BYZ origin preference; subjects remain eligible to spread.
                if = { limit = { owner = c:BYZ } add = 100000000 }
            }
            spawn_movement = {
                movement_definition = movement_definition:oe_egypt_identity_movement
                supporters = { value = population multiply = 0.1 }
            }
            oe_identity_schedule_30_years = { holder = culture:oe_egypt_greek_culture event = oe_identity_expiry.6 }
        }
    }
}
