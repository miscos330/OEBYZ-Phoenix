# IO scope; event identity and family references stay in this event's saved scopes.
oe_daily_capture_house = {
    save_scope_value_as = { name = oe_daily_house_slot value = $slot$ }
    var:oe_nf_dynasty_$slot$ = { save_scope_as = oe_daily_family }
    var:oe_nf_head_$slot$ = { save_scope_as = oe_daily_head }
    set_variable = { name = oe_daily_house_$event$_cooldown_$slot$ value = 1 months = 36 }
    oe_daily_pick_location_$event$ = { slot = $slot$ }
}
# Country scope; exactly one candidate per house, so large families receive no selection advantage.
oe_daily_choose_house = {
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    clear_saved_scope = oe_daily_family
	    clear_saved_scope = oe_daily_head
	    clear_saved_scope = oe_daily_location
	    clear_saved_scope = oe_daily_house_slot
	    international_organization:oe_byz_empire_pillars = {
	        if = { limit = { oe_daily_house_candidate = { slot = 1 event = $event$ } } var:oe_nf_head_1 = { add_to_list = oe_daily_candidate_heads } }
	        if = { limit = { oe_daily_house_candidate = { slot = 2 event = $event$ } } var:oe_nf_head_2 = { add_to_list = oe_daily_candidate_heads } }
	        if = { limit = { oe_daily_house_candidate = { slot = 3 event = $event$ } } var:oe_nf_head_3 = { add_to_list = oe_daily_candidate_heads } }
	        if = { limit = { oe_daily_house_candidate = { slot = 4 event = $event$ } } var:oe_nf_head_4 = { add_to_list = oe_daily_candidate_heads } }
	    }
	    random_in_list = {
	        list = oe_daily_candidate_heads
	        save_scope_as = oe_daily_chosen_head
	        international_organization:oe_byz_empire_pillars = {
	            if = { limit = { has_variable = oe_nf_dynasty_1 var:oe_nf_head_1 = scope:oe_daily_chosen_head } oe_daily_capture_house = { slot = 1 event = $event$ } }
	            if = { limit = { has_variable = oe_nf_dynasty_2 var:oe_nf_head_2 = scope:oe_daily_chosen_head } oe_daily_capture_house = { slot = 2 event = $event$ } }
	            if = { limit = { has_variable = oe_nf_dynasty_3 var:oe_nf_head_3 = scope:oe_daily_chosen_head } oe_daily_capture_house = { slot = 3 event = $event$ } }
	            if = { limit = { has_variable = oe_nf_dynasty_4 var:oe_nf_head_4 = scope:oe_daily_chosen_head } oe_daily_capture_house = { slot = 4 event = $event$ } }
	        }
	    }
	}
}
