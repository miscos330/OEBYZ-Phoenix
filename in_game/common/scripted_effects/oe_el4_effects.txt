			remove_country_modifier = oe_el4_culture_s1_mod
		}
		if = {
			limit = { has_country_modifier = oe_el4_culture_s2_mod }

		if = {

			add_country_modifier = {
				modifier = oe_el4_culture_s3_mod
				years = -1
				mode = replace
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				current_year >= 1380
				oe_el4_years_since_settled >= 35

		if = {

	else = {
