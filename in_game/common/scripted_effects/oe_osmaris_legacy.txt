# BYZ scope; requests persist until the city can receive the first free level.
oe_osmaris_grant_legacy_buildings = {
    if = {
        limit = { owns = location:constantinople }
        if = {
            limit = { has_variable = oe_osmaris_academy_requested NOT = { has_variable = oe_osmaris_academy_granted } }
            location:constantinople = {
                set_variable = oe_city_manuel_academy_unlocked
                if = {
                    limit = { oe_city_manuel_academy_completed = 0 }
                    set_variable = { name = oe_city_manuel_academy_orders value = 1 }
                    construct_building = { building_type = building_type:oe_byz_constantinople_manuel_academy owner = c:BYZ instant = yes cost_multiplier = 0 cost_multiplier_reason = "game_concept_event" }
                }
                if = { limit = { oe_city_manuel_academy_completed > 0 } c:BYZ = { set_variable = oe_osmaris_academy_granted } }
            }
        }
        if = {
            limit = { has_variable = oe_osmaris_company_requested NOT = { has_variable = oe_osmaris_company_granted } }
            location:constantinople = {
                set_variable = oe_city_osman_company_unlocked
                if = {
                    limit = { oe_city_osman_company_completed = 0 }
                    set_variable = { name = oe_city_osman_company_orders value = 1 }
                    construct_building = { building_type = building_type:oe_byz_constantinople_osman_company owner = c:BYZ instant = yes cost_multiplier = 0 cost_multiplier_reason = "game_concept_event" }
                }
                if = { limit = { oe_city_osman_company_completed > 0 } c:BYZ = { set_variable = oe_osmaris_company_granted } }
            }
        }
    }
}
oe_osmaris_memorial_accept = {
    if = {
        limit = { has_variable = oe_osmaris_manuel_death_seen NOT = { has_variable = oe_osmaris_academy_requested } }
        set_variable = oe_osmaris_academy_requested
        oe_osmaris_grant_legacy_buildings = yes
    }
}
oe_osmaris_introduce_constantine = {
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    if = {
	        limit = { NOT = { exists = character:oe_osmaris_constantine } NOT = { has_variable = oe_osmaris_constantine_introduced } }
	        set_variable = oe_osmaris_constantine_introduced
	        create_character = {
	            first_name = oe_osmaris_constantine_name
	            birth_date = 1387.1.1
	            birth_location = location:bursa
	            dynasty = dynasty:ottomans_dynasty
	            culture = culture:oe_tur_greek_culture
	            religion = religion:orthodox
	            estate = estate_type:burghers_estate
	            female = no
	            adm = 85 dip = 75 mil = 60
	            script = oe_osmaris_constantine
	            save_scope_as = oe_osmaris_constantine_person
	        }
	        character:oe_osmaris_constantine ?= {
	            set_nickname = oe_osmaris_good
	            character:oe_osmaris_basil ?= { save_scope_as = oe_osmaris_constantine_father }
	            if = { limit = { exists = scope:oe_osmaris_constantine_father } change_father = scope:oe_osmaris_constantine_father }
	        }
	    }
	    character:oe_osmaris_constantine ?= { save_scope_as = oe_osmaris_constantine_person }
	    international_organization:oe_byz_empire_pillars = {
	        save_scope_value_as = { name = oe_osmaris_aid_cost value = { value = oe_pillars_purse_monthly_income min = 0 multiply = 6 } }
	    }
	}
}
oe_osmaris_aid_constantine = {
    if = {
        limit = { oe_osmaris_constantine_can_aid = yes }
        set_variable = oe_osmaris_constantine_resolved
        set_variable = oe_osmaris_investment_made
        international_organization:oe_byz_empire_pillars = {
            add_gold = { value = scope:oe_osmaris_aid_cost multiply = -1 }
            change_variable = { name = oe_pillars_authority add = 2 }
            clamp_variable = { name = oe_pillars_authority min = 0 max = 100 }
        }
        oe_osmaris_reward_house = { member = character:oe_osmaris_constantine }
        trigger_event_non_silently = { id = oe_flavor_byz.1011 years = 5 }
    }
}
oe_osmaris_decline_constantine = {
    if = {
        limit = { oe_osmaris_constantine_can_choose = yes }
        set_variable = oe_osmaris_constantine_resolved
        oe_osmaris_reward_house = { member = character:oe_osmaris_constantine }
    }
}
oe_osmaris_collect_return = {
    if = {
        limit = { has_variable = oe_osmaris_investment_made NOT = { has_variable = oe_osmaris_investment_returned } exists = international_organization:oe_byz_empire_pillars }
        set_variable = oe_osmaris_investment_returned
        international_organization:oe_byz_empire_pillars = { add_gold = scope:oe_osmaris_return_amount }
    }
}
