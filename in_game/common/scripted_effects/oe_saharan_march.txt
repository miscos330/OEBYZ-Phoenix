# Country scope. Snapshot both areas BEFORE creation changes the seed's owner.
oe_found_saharan_march_effect = {
    area:desert_area = {
        every_location_in_area = {
            limit = { oe_saharan_march_grant_location = yes }
            add_to_list = oe_saharan_march_locations
        }
    }
    area:western_sahara_area = {
        every_location_in_area = {
            limit = { oe_saharan_march_grant_location = yes }
            add_to_list = oe_saharan_march_locations
        }
    }
    ordered_in_list = {
        list = oe_saharan_march_locations
        order_by = population
        max = 1
        check_range_bounds = no
        save_scope_as = oe_saharan_march_capital
    }
    if = {
        limit = { exists = scope:oe_saharan_march_capital }
        scope:oe_saharan_march_capital = {
            if = {
            	limit = { has_owner = yes }
	            create_country_from_location = {
	                overlord = c:BYZ
	                subject_type = subject_type:oe_roman_march
	                save_scope_as = oe_new_saharan_march
	                change_government_type = government_type:monarchy
	                change_country_type = location
	                change_heir_selection = heir_selection:cognatic_primogeniture
	                change_culture = culture:greek_culture
	                change_religion = religion:orthodox
	                change_country_name = OE_SAHARAN_MARCH
	                change_country_adjective = OE_SAHARAN_MARCH_ADJ
	                change_country_flag = BYZ
	                set_capital = scope:oe_saharan_march_capital
	                if = {
	                    limit = { has_ruler = no }
	                    create_character = {
	                        age = 35 culture = culture:greek_culture religion = religion:orthodox
	                        estate = estate_type:nobles_estate save_scope_as = oe_saharan_march_ruler
	                    }
	                    set_new_ruler = scope:oe_saharan_march_ruler
	                }
	            }
            }
        }
        if = {
            limit = { exists = scope:oe_new_saharan_march }
            every_in_list = {
                list = oe_saharan_march_locations
                change_location_owner = scope:oe_new_saharan_march
            }
            set_variable = oe_saharan_march_founded
            international_organization:oe_byz_roman_world ?= {
                add_country_to_international_organization = scope:oe_new_saharan_march
            }
        }
    }
}
