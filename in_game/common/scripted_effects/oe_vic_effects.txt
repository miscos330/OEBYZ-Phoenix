# Location scope. Removes all oe_vic_* seat loyalty tier modifiers cleanly.
oe_vic_clear_seat_modifiers = {
	if = { limit = { has_location_modifier = oe_vic_seat_loyal_mod } remove_location_modifier = oe_vic_seat_loyal_mod }
	if = { limit = { has_location_modifier = oe_vic_seat_restive_mod } remove_location_modifier = oe_vic_seat_restive_mod }
	if = { limit = { has_location_modifier = oe_vic_seat_defiant_mod } remove_location_modifier = oe_vic_seat_defiant_mod }
}

# Viceroyalty country scope. Removes all oe_vic_subj_* tier modifiers cleanly.
oe_vic_clear_subj_modifiers = {
	if = { limit = { has_country_modifier = oe_vic_subj_compliant_mod } remove_country_modifier = oe_vic_subj_compliant_mod }
	if = { limit = { has_country_modifier = oe_vic_subj_restive_mod } remove_country_modifier = oe_vic_subj_restive_mod }
	if = { limit = { has_country_modifier = oe_vic_subj_defiant_mod } remove_country_modifier = oe_vic_subj_defiant_mod }
}

# Viceroyalty country scope. Adjusts oe_vic_loyalty by $amount$ (clamped 0..100).
oe_vic_shift_loyalty = {
	if = { limit = { NOT = { has_variable = oe_vic_loyalty } } set_variable = { name = oe_vic_loyalty value = 55 } }
	change_variable = { name = oe_vic_loyalty add = $amount$ }
	clamp_variable = { name = oe_vic_loyalty min = 0 max = 100 }
}

# Viceroyalty country scope. Syncs oe_vic_tier (3>=70, 2=40..69, 1=20..39, 0<20) and country tier modifiers.
oe_vic_sync_self_tier = {
	if = { limit = { NOT = { has_variable = oe_vic_loyalty } } set_variable = { name = oe_vic_loyalty value = 55 } }
	if = {
		limit = { has_variable = oe_vic_loyalty var:oe_vic_loyalty >= 70 }
		set_variable = { name = oe_vic_tier value = 3 }
		oe_vic_clear_subj_modifiers = yes
		add_country_modifier = { modifier = oe_vic_subj_compliant_mod months = 14 mode = replace }
	}
	else_if = {
		limit = { has_variable = oe_vic_loyalty var:oe_vic_loyalty >= 40 }
		set_variable = { name = oe_vic_tier value = 2 }
		oe_vic_clear_subj_modifiers = yes
	}
	else_if = {
		limit = { has_variable = oe_vic_loyalty var:oe_vic_loyalty >= 20 }
		set_variable = { name = oe_vic_tier value = 1 }
		oe_vic_clear_subj_modifiers = yes
		add_country_modifier = { modifier = oe_vic_subj_restive_mod months = 14 mode = replace }
	}
	else = {
		set_variable = { name = oe_vic_tier value = 0 }
		oe_vic_clear_subj_modifiers = yes
		add_country_modifier = { modifier = oe_vic_subj_defiant_mod months = 14 mode = replace }
	}
}

# Any scope. F6-style clean dissolution of DLD (Deccan Iqta Viceroyalty): annex via overlord or destroy_building_country, then clear seat.
oe_vic_end_DLD = {
	set_global_variable = { name = oe_vic_ended_dld value = 1 }
	if = { limit = { exists = scope:oe_vic_end_lord_DLD } clear_saved_scope = oe_vic_end_lord_DLD }
	if = {
		limit = { country_exists = c:DLD }
		c:DLD = {
			set_variable = { name = oe_vic_ended value = 1 }
			if = { limit = { is_subject = yes } overlord = { save_scope_as = oe_vic_end_lord_DLD } }
		}
		if = {
			limit = { exists = scope:oe_vic_end_lord_DLD }
			scope:oe_vic_end_lord_DLD = { annex_country = { country = c:DLD reason = Diplomatic transfer = yes } }
		}
		else = {
			destroy_building_country = c:DLD
		}
	}
	location:daulatabad = {
		if = { limit = { has_building = building_type:oe_vic_seat_dld } destroy_all_buildings_of_type = building_type:oe_vic_seat_dld }
		oe_vic_clear_seat_modifiers = yes
	}
}

# Any scope. Yearly loyalty drift & seat-location modifier synchronization across all active viceroyalties.
oe_vic_sync_all_viceroys = {
	if = { # DLD under DLH
		limit = { country_exists = c:DLD country_exists = c:DLH c:DLD = { is_subject_of = c:DLH is_subject_type = oe_dlh_iqta } }
		c:DLD = {
			if = { limit = { NOT = { has_variable = oe_vic_loyalty } } set_variable = { name = oe_vic_loyalty value = 48 } }
			if = {
				limit = { c:DLH = { oe_vic_dlh_in_crisis = yes } }
				change_variable = { name = oe_vic_loyalty add = -5 }
			}
			else_if = {
				limit = { c:DLH = { stability >= 15 legitimacy >= 70 } }
				change_variable = { name = oe_vic_loyalty add = 3 }
			}
			else_if = {
				limit = { c:DLH = { stability < -15 } }
				change_variable = { name = oe_vic_loyalty add = -4 }
			}
			if = {
				limit = { c:DLH = { has_country_modifier = oe_vic_audit_harmonized_mod } }
				change_variable = { name = oe_vic_loyalty add = 3 }
			}
			clamp_variable = { name = oe_vic_loyalty min = 0 max = 100 }
			oe_vic_sync_self_tier = yes
		}
		location:daulatabad = {
			if = {
				limit = { owner ?= c:DLH }
				oe_vic_clear_seat_modifiers = yes
				if = { limit = { c:DLD = { var:oe_vic_tier ?= 3 } } add_location_modifier = { modifier = oe_vic_seat_loyal_mod months = 14 mode = replace } }
				else_if = { limit = { c:DLD = { var:oe_vic_tier ?= 1 } } add_location_modifier = { modifier = oe_vic_seat_restive_mod months = 14 mode = replace } }
				else_if = { limit = { c:DLD = { var:oe_vic_tier ?= 0 } } add_location_modifier = { modifier = oe_vic_seat_defiant_mod months = 14 mode = replace } }
			}
		}
	}
	if = { # DLG under DLH
		limit = { country_exists = c:DLG country_exists = c:DLH c:DLG = { is_subject_of = c:DLH is_subject_type = oe_dlh_iqta } }
		c:DLG = {
			if = { limit = { NOT = { has_variable = oe_vic_loyalty } } set_variable = { name = oe_vic_loyalty value = 52 } }

# Any scope. Dynamically awakens vanilla dormant TRA (Transylvania) and/or SVN (Slavonia) if Hungary critically weakens.
oe_vic_hun_awaken_tra_svn = {
	if = {
		limit = {
			country_exists = c:HUN
			NOT = { has_global_variable = oe_vic_hun_diet_reformed }
		}
		if = {
			limit = {
				NOT = { country_exists = c:TRA }
				c:HUN = { any_owned_location = { is_core_of = c:TRA } }
			}
			c:HUN = { create_country_from_cores_in_our_locations = c:TRA }
			if = {
				limit = { country_exists = c:TRA }
				c:TRA = {
					add_country_modifier = { modifier = oe_vic_hun_voivode_charter_mod years = 25 mode = replace }
					if = {
						limit = { c:HUN = { owns = location:buda } }
						make_subject_of = { target = c:HUN type = subject_type:vassal }
					}
				}
			}
			oe_vic_end_HNT = yes
		}
		if = {
			limit = {
				NOT = { country_exists = c:SVN }
				c:HUN = {
					any_owned_location = { is_core_of = c:SVN }
					OR = {
						NOT = { owns = location:buda }
						stability < -15
						in_civil_war = yes
					}
				}
			}
			c:HUN = { create_country_from_cores_in_our_locations = c:SVN }
			if = {
				limit = { country_exists = c:SVN }
				c:SVN = {
					add_country_modifier = { modifier = oe_vic_hun_voivode_charter_mod years = 20 mode = replace }
					if = {
						limit = { c:HUN = { owns = location:buda } }
						make_subject_of = { target = c:HUN type = subject_type:vassal }
					}
				}
			}
		}
	}
}

# Any scope. Yearly historical event dispatcher for all 12 oe_vic events (once per campaign each, zero orphans).
oe_vic_dispatch_events = {
	# --- Delhi Sultanate (DLH): oe_vic.1, oe_vic.2, oe_vic.3 ---
	if = {
		limit = { country_exists = c:DLH c:DLH = { is_rebel_country = no } }
		if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev1_done }
				current_year >= 1345
				country_exists = c:DLD
				c:DLD = { is_subject_of = c:DLH }
			}
			set_global_variable = { name = oe_vic_ev1_done value = 1 }
			c:DLH = { trigger_event_non_silently = oe_vic.1 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev2_done }
				current_year >= 1341
				OR = {
					AND = { country_exists = c:DLG c:DLG = { is_subject_of = c:DLH } }
					AND = { country_exists = c:DLM c:DLM = { is_subject_of = c:DLH } }
				}
			}
			set_global_variable = { name = oe_vic_ev2_done value = 1 }
			c:DLH = { trigger_event_non_silently = oe_vic.2 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev3_done }
				current_year >= 1349
				OR = {
					AND = { country_exists = c:DLP c:DLP = { is_subject_of = c:DLH } }
					AND = { country_exists = c:DLJ c:DLJ = { is_subject_of = c:DLH } }
				}
			}
			set_global_variable = { name = oe_vic_ev3_done value = 1 }
			c:DLH = { trigger_event_non_silently = oe_vic.3 }
		}
	}

	# --- Mamluk Sultanate (MAM): oe_vic.4, oe_vic.5, oe_vic.6 ---
	if = {
		limit = { country_exists = c:MAM c:MAM = { is_rebel_country = no } }
		if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev4_done }
				current_year >= 1340
				country_exists = c:MMD
				c:MMD = { is_subject_of = c:MAM }
			}
			set_global_variable = { name = oe_vic_ev4_done value = 1 }
			c:MAM = { trigger_event_non_silently = oe_vic.4 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev5_done }
				current_year >= 1382
				OR = {
					AND = { country_exists = c:MMD c:MMD = { is_subject_of = c:MAM } }
					AND = { country_exists = c:MMH c:MMH = { is_subject_of = c:MAM } }
					AND = { country_exists = c:MMK c:MMK = { is_subject_of = c:MAM } }
				}
			}
			set_global_variable = { name = oe_vic_ev5_done value = 1 }
			c:MAM = { trigger_event_non_silently = oe_vic.5 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev6_done }
				current_year >= 1344
				OR = {
					AND = { country_exists = c:MEC c:MEC = { is_subject_of = c:MAM is_subject_type = oe_mam_sharifate } }
					AND = { country_exists = c:MDA c:MDA = { is_subject_of = c:MAM is_subject_type = oe_mam_sharifate } }
				}
			}
			set_global_variable = { name = oe_vic_ev6_done value = 1 }
			c:MAM = { trigger_event_non_silently = oe_vic.6 }
		}
	}

	# --- Kingdom of Hungary (HUN): oe_vic.8 (1339+), oe_vic.7 (1382+ crisis) ---
	if = {
		limit = { country_exists = c:HUN c:HUN = { is_rebel_country = no } }
		if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev8_done }
				current_year >= 1339
				country_exists = c:HNT
				c:HNT = { is_subject_of = c:HUN }
			}
			set_global_variable = { name = oe_vic_ev8_done value = 1 }
			c:HUN = { trigger_event_non_silently = oe_vic.8 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev7_done }
				NOT = { has_global_variable = oe_vic_hun_diet_reformed }
				current_year >= 1382
				c:HUN = { oe_vic_hun_in_crisis = yes }
				OR = {
					AND = { country_exists = c:HNT c:HNT = { is_subject_of = c:HUN } }
					c:HUN = { any_owned_location = { is_core_of = c:TRA } }
				}
			}
			set_global_variable = { name = oe_vic_ev7_done value = 1 }
			c:HUN = { trigger_event_non_silently = oe_vic.7 }
		}
	}

	# --- Crown of Castile (CAS): oe_vic.10 (1340+), oe_vic.9 (1351+ Trastamara) ---
	if = {
		limit = { country_exists = c:CAS c:CAS = { is_rebel_country = no } }
		if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev9_done }
				current_year >= 1351
				OR = {
					c:CAS = { oe_vic_cas_in_crisis = yes }
					current_year >= 1358
				}
				OR = {
					AND = { country_exists = c:CST c:CST = { is_subject_of = c:CAS } }
					AND = { country_exists = c:CSG c:CSG = { is_subject_of = c:CAS } }
				}
			}
			set_global_variable = { name = oe_vic_ev9_done value = 1 }
			c:CAS = { trigger_event_non_silently = oe_vic.9 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev10_done }
				current_year >= 1340
				country_exists = c:CSG
				c:CSG = { is_subject_of = c:CAS }
			}
			set_global_variable = { name = oe_vic_ev10_done value = 1 }
			c:CAS = { trigger_event_non_silently = oe_vic.10 }
		}
	}

	# --- Solomonic Ethiopia (ETH): oe_vic.11 (1342+), oe_vic.12 (1356+) ---
	if = {
		limit = { country_exists = c:ETH c:ETH = { is_rebel_country = no } }
		if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev11_done }
				current_year >= 1342
				country_exists = c:TIG
				c:TIG = { is_subject_of = c:ETH }
			}
			set_global_variable = { name = oe_vic_ev11_done value = 1 }
			c:ETH = { trigger_event_non_silently = oe_vic.11 }
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_vic_ev12_done }
				current_year >= 1356
				country_exists = c:TIG
				c:TIG = { is_subject_of = c:ETH }
			}
			set_global_variable = { name = oe_vic_ev12_done value = 1 }
			c:ETH = { trigger_event_non_silently = oe_vic.12 }
		}
	}
}

# Any scope. Exported zero-slot yearly dispatcher called from oe_gk_yearly_dispatch (self-deduplicating).
oe_vic_kernel_yearly = {
	if = {
		limit = { NOT = { global_var:oe_vic_ran_y ?= current_year } }
		set_global_variable = { name = oe_vic_ran_y value = current_year }
		if = {
			limit = { NOT = { has_global_variable = oe_vic_init_done } }
			set_global_variable = { name = oe_vic_init_done value = 1 }
			oe_vic_init_once = yes
		}
		oe_vic_sync_sharifates = yes
		oe_vic_cleanup_check = yes
		oe_vic_sync_all_viceroys = yes
		oe_vic_dispatch_events = yes
	}
}
