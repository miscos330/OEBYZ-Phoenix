# Country scope. Snapshot every eligible holding, then use a seed in the fully owned prerequisite area.
oe_found_caucasian_march_effect = {
    region:caucasus_region = {
        every_location_in_region = { limit = { oe_caucasian_march_grant_location = yes } add_to_list = oe_caucasian_march_locations }
    }
    province_definition:tuapse_province = {
        every_location_in_province_definition = { limit = { oe_caucasian_march_grant_location = yes } add_to_list = oe_caucasian_march_locations }
    }
    area:georgia_area = {
        ordered_location_in_area = {
            limit = { oe_caucasian_march_grant_location = yes }
            order_by = population
            max = 1
            check_range_bounds = no
            save_scope_as = oe_caucasian_march_capital
        }
    }
    scope:oe_caucasian_march_capital = {
        create_country_from_location = {
            overlord = c:BYZ
            subject_type = subject_type:oe_roman_march
            save_scope_as = oe_new_caucasian_march
            change_government_type = government_type:monarchy
            change_country_type = location
            change_heir_selection = heir_selection:cognatic_primogeniture
            change_culture = culture:greek_culture
            change_religion = religion:orthodox
            change_country_name = OE_CAUCASIAN_MARCH
            change_country_adjective = OE_CAUCASIAN_MARCH_ADJ
            change_country_flag = GEO
            set_capital = scope:oe_caucasian_march_capital
            add_country_modifier = { modifier = oe_caucasian_march_modifier years = -1 mode = replace }
            if = {
                limit = { has_ruler = no }
                create_character = {
                    age = 35 culture = culture:greek_culture religion = religion:orthodox
                    estate = estate_type:nobles_estate save_scope_as = oe_caucasian_march_ruler
                }
                set_new_ruler = scope:oe_caucasian_march_ruler
            }
        }
    }
    every_in_list = { list = oe_caucasian_march_locations change_location_owner = scope:oe_new_caucasian_march }
    set_variable = oe_caucasian_march_founded
    international_organization:oe_byz_roman_world ?= { add_country_to_international_organization = scope:oe_new_caucasian_march }
}

# Snapshot all eligible holdings before the new country changes ownership.
oe_found_nubian_march_effect = {
    region:nubia_region = {
        every_location_in_region = {
            limit = { oe_nubian_march_grant_location = yes }
            add_to_list = oe_nubian_march_locations
        }
    }
    ordered_in_list = {
        list = oe_nubian_march_locations
        order_by = population
        max = 1
        check_range_bounds = no
        save_scope_as = oe_nubian_march_capital
    }
    if = {
        limit = { exists = scope:oe_nubian_march_capital }
        scope:oe_nubian_march_capital = {
            create_country_from_location = {
                overlord = c:BYZ
                subject_type = subject_type:oe_roman_march

                change_government_type = government_type:monarchy
                change_country_type = location

                change_religion = religion:orthodox
                change_country_name = OE_NUBIAN_MARCH
                change_country_adjective = OE_NUBIAN_MARCH_ADJ
                change_country_flag = MAK
                set_capital = scope:oe_nubian_march_capital
                if = {
                    limit = { has_ruler = no }
                    create_character = {
                        age = 35
                        culture = culture:nubian
                        religion = religion:orthodox
                        estate = estate_type:nobles_estate
                        save_scope_as = oe_nubian_march_ruler
                    }
                    set_new_ruler = scope:oe_nubian_march_ruler

            }
        }
        if = {
            limit = { exists = scope:oe_new_nubian_march }
            every_in_list = {
                list = oe_nubian_march_locations

                add_country_to_international_organization = scope:oe_new_nubian_march
            }
        }
    }
