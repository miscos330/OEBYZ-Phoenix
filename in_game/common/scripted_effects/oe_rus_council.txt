oe_rsc_depart_to_venue = {
    scope:expedition = { set_variable = { name = oe_rsc_stage value = 6 } }
    if = {
        limit = { scope:expedition = { expedition_current_location = var:oe_rsc_venue } }
        oe_rsc_open_council = yes
    }
    else = {
        scope:expedition = { add_new_waypoint = var:oe_rsc_venue }
        resume_expedition = scope:expedition
    }
}
oe_rsc_open_council = {
    if = {
        limit = { scope:expedition.var:oe_rsc_venue = { oe_rsc_venue_eligible = yes } }
        scope:expedition = {
            set_variable = { name = oe_rsc_stage value = 3 }
            set_variable = { name = oe_rsc_host value = var:oe_rsc_venue.owner }
            pause_expedition = 200
        }
        # Count from arrival. Pending player choices delay only final settlement,
        # not the 60/120/180-day agenda or the 200-day completion deadline.
        trigger_event_non_silently = oe_rus_council.4
        trigger_event_silently = { id = oe_rus_council.10 days = 60 }
        trigger_event_silently = { id = oe_rus_council.10 days = 120 }
        trigger_event_silently = { id = oe_rus_council.10 days = 180 }
        trigger_event_silently = { id = oe_rus_council.20 days = 200 }
    }
    else = { fail_expedition = scope:expedition }
}
oe_rsc_finish_if_ready = {
    if = {
        limit = {
            oe_rsc_live = yes
            scope:expedition = {
                var:oe_rsc_stage = 3
                has_variable = oe_rsc_due
                var:oe_rsc_pending = 0
            }
        }
        scope:expedition = { set_variable = { name = oe_rsc_stage value = 4 } }
        if = {
            limit = { scope:expedition = { var:oe_rsc_disagreements >= 3 } }
            trigger_event_non_silently = oe_rus_council.21
        }
        else_if = {
            limit = { scope:expedition = { var:oe_rsc_disagreements = 0 } }
            trigger_event_non_silently = oe_rus_council.22
        }
        else = { trigger_event_non_silently = oe_rus_council.23 }
    }
}
