# ---------------------------------------------------------------- Cohesion adjustment & clamping
oe_ghh_shift_cohesion = {
	if = {
		limit = { has_global_variable = oe_ghh_cohesion }
		change_global_variable = { name = oe_ghh_cohesion add = $amount$ }
		if = {
			limit = { global_var:oe_ghh_cohesion < 0 }
			set_global_variable = { name = oe_ghh_cohesion value = 0 }
		}
		if = {
			limit = { has_global_variable = oe_ghh_cohesion_cap global_var:oe_ghh_cohesion > global_var:oe_ghh_cohesion_cap }
			set_global_variable = { name = oe_ghh_cohesion value = global_var:oe_ghh_cohesion_cap }
		}
		if = {
			limit = { global_var:oe_ghh_cohesion > 100 }
			set_global_variable = { name = oe_ghh_cohesion value = 100 }
		}
		oe_ghh_update_tier = yes
	}
}

		}

		}
	}
}

# ---------------------------------------------------------------- Annual January cohesion tick
oe_ghh_yearly_cohesion = {
	if = {
		limit = { country_exists = c:GLH has_global_variable = oe_ghh_cohesion }
		c:GLH = {
			# Stabilizing factors
			if = { limit = { horde_unity >= 80 } oe_ghh_shift_cohesion = { amount = 2 } }
			if = { limit = { stability >= 25 } oe_ghh_shift_cohesion = { amount = 2 } }
			if = { limit = { owns = location:sarai_al_jadid owns = location:astrakhan } oe_ghh_shift_cohesion = { amount = 1 } }
			if = { limit = { has_global_variable = oe_ghh_revival_stage global_var:oe_ghh_revival_stage = 1 } oe_ghh_shift_cohesion = { amount = 3 } }
			if = { limit = { has_global_variable = oe_ghh_revival_stage global_var:oe_ghh_revival_stage >= 2 } oe_ghh_shift_cohesion = { amount = 5 } }
			if = { limit = { has_global_variable = oe_ghh_phase global_var:oe_ghh_phase = 4 } oe_ghh_shift_cohesion = { amount = 6 } }
			# Destabilizing factors
			if = { limit = { has_active_disaster = disaster_type:horde_civil_war } oe_ghh_shift_cohesion = { amount = -6 } }
			if = { limit = { horde_unity < 50 } oe_ghh_shift_cohesion = { amount = -3 } }
			if = { limit = { horde_unity < 30 } oe_ghh_shift_cohesion = { amount = -3 } }
			if = { limit = { stability < 0 } oe_ghh_shift_cohesion = { amount = -3 } }
			if = { limit = { has_regent = yes } oe_ghh_shift_cohesion = { amount = -2 } }
			if = { limit = { in_civil_war = yes } oe_ghh_shift_cohesion = { amount = -4 } }
			if = { limit = { war_exhaustion >= 5 } oe_ghh_shift_cohesion = { amount = -2 } }
			if = { limit = { NOT = { location:sarai_al_jadid = { owner ?= c:GLH controller ?= c:GLH } } } oe_ghh_shift_cohesion = { amount = -5 } }
			# Gentle AI historical decline drift (never applied to human GLH player!)
			if = {
				limit = { is_ai = yes current_date >= 1359.1.1 has_global_variable = oe_ghh_phase global_var:oe_ghh_phase < 4 }
				oe_ghh_shift_cohesion = { amount = -2 }
			}
			if = {
				limit = { is_ai = yes current_date >= 1395.1.1 has_global_variable = oe_ghh_phase global_var:oe_ghh_phase = 3 }
				oe_ghh_shift_cohesion = { amount = -3 }
			}
		}
		oe_ghh_sync_modifiers = yes
	}
}

# ---------------------------------------------------------------- Phase 2 inter-ulus raiding friction (every 36m while cohesion < 50)
oe_ghh_phase2_friction = {
	if = {
		limit = {
			has_global_variable = oe_ghh_phase global_var:oe_ghh_phase = 2
			has_global_variable = oe_ghh_cohesion global_var:oe_ghh_cohesion < 50
			OR = {
				NOT = { has_global_variable = oe_ghh_last_raid_m }
				global_var:oe_ghh_last_raid_m <= { value = oe_ml_month_now subtract = 36 }
			}
		}
		set_global_variable = { name = oe_ghh_last_raid_m value = oe_ml_month_now }
		if = {
			limit = { country_exists = c:WHO country_exists = c:SBR c:WHO = { is_subject_of = c:GLH } c:SBR = { is_subject_of = c:GLH } }
			c:WHO = { add_opinion = { modifier = oe_ghh_b_ulus_raid target = c:SBR } }
			c:SBR = { add_opinion = { modifier = oe_ghh_b_ulus_raid target = c:WHO } }
		}
		if = {
			limit = { country_exists = c:CRI country_exists = c:KZN c:CRI = { is_subject_of = c:GLH } c:KZN = { is_subject_of = c:GLH } }

		oe_ghh_shift_cohesion = { amount = -2 }
	}
}

	}
	if = {

		# Stop cleanly if GLH ceases to exist or after 1520
		if = {
			limit = { OR = { NOT = { country_exists = c:GLH } current_date >= 1520.1.1 } }
			remove_global_variable = oe_ghh_on
			oe_ghh_sync_modifiers = yes
		}
		else = {
			# 1. Human GLH 1337 intro event (.1)
			if = {
				limit = {
					NOT = { has_global_variable = oe_ghh_ev_1 }
					current_date >= 1337.5.1
					oe_ghh_ev_month_free = yes
					c:GLH = { is_ai = no }
				}
				set_global_variable = { name = oe_ghh_ev_1 value = 1 }
				oe_ghh_send_ev = { n = 1 }
			}
			# 2. Annual January cohesion drift & modifier sync
			if = {
				limit = { current_month = 1 }
				oe_ghh_yearly_cohesion = yes
			}
			# 3. Black Death shock (1346-1354)
			if = {
				limit = {
					NOT = { has_global_variable = oe_ghh_plague_hit }
					has_global_variable = oe_ghh_phase global_var:oe_ghh_phase < 4
					current_date >= 1346.1.1 current_date < 1358.1.1
					OR = {
						disease_is_active = disease:bubonic_plague
						current_date >= 1348.6.1
					}
				}
				set_global_variable = { name = oe_ghh_plague_hit value = 1 }
				if = {
					limit = { c:GLH = { is_ai = no } oe_ghh_ev_month_free = yes }
					set_global_variable = { name = oe_ghh_ev_2 value = 1 }
					oe_ghh_send_ev = { n = 2 }
				}
				else = {
					oe_ghh_shift_cohesion = { amount = -15 }
					oe_ghh_sync_modifiers = yes
				}
			}
			# 4. Phase 2 trigger — Great Troubles (1359+)
			if = {
				limit = { oe_ghh_should_start_phase2 = yes }
				set_global_variable = { name = oe_ghh_phase value = 2 }
				oe_ghh_shift_cohesion = { amount = -25 }
				oe_ghh_sync_modifiers = yes
				if = {
					limit = { NOT = { has_global_variable = oe_ghh_ev_3 } c:GLH = { is_ai = no } oe_ghh_ev_month_free = yes }
					set_global_variable = { name = oe_ghh_ev_3 value = 1 }
					oe_ghh_send_ev = { n = 3 }
				}
			}
			# Track months spent in Phase 2 + Tokhtamysh brief resurgence (sensing oe_glh_troubles = 2)
			if = {
				limit = { has_global_variable = oe_ghh_phase global_var:oe_ghh_phase = 2 }
				change_global_variable = { name = oe_ghh_p2_months add = 1 }
				oe_ghh_phase2_friction = yes
				if = {
					limit = {
						NOT = { has_global_variable = oe_ghh_tokhtamysh_bump }
						has_global_variable = oe_glh_troubles global_var:oe_glh_troubles = 2
					}
					set_global_variable = { name = oe_ghh_tokhtamysh_bump value = 1 }
					oe_ghh_shift_cohesion = { amount = 20 }
					oe_ghh_sync_modifiers = yes
				}
				# Early Phase 2 breakaway: Khwarazm (KHW, Sufi dynasty, 1361+)

					limit = {
						current_date >= 1361.1.1
						NOT = { has_global_variable = oe_ghh_free_KHW }
						NOT = { has_global_variable = oe_ghh_ev_4 }
						has_global_variable = oe_ghh_cohesion global_var:oe_ghh_cohesion < 50
						country_exists = c:KHW c:KHW = { is_subject_of = c:GLH is_subject_type = oe_glh_ulus }
					}
					if = {
						limit = { c:GLH = { is_ai = no } oe_ghh_ev_month_free = yes }
						set_global_variable = { name = oe_ghh_ev_4 value = 1 }
						oe_ghh_send_ev = { n = 4 }
					}
					else_if = {
						limit = { c:GLH = { is_ai = yes } }
						oe_ghh_break_wing = { tag = KHW ind_name = OE_GHH_NAME_KHW_KHANATE }
					}
				}
			}
			# 5. Phase 3 trigger — Timur Shock & Great Fragmentation (1395+)
			if = {
				limit = { oe_ghh_should_start_phase3 = yes }
				set_global_variable = { name = oe_ghh_phase value = 3 }
				set_global_variable = { name = oe_ghh_timur_hit value = 1 }
				set_global_variable = { name = oe_ghh_cohesion_cap value = 55 }
				oe_ghh_shift_cohesion = { amount = -30 }
				oe_ghh_sync_modifiers = yes

					oe_ghh_break_cd_ready = yes
				}

				else_if = {

						NOT = { has_global_variable = oe_ghh_free_CRI }
						country_exists = c:CRI c:CRI = { is_subject_of = c:GLH is_subject_type = oe_glh_ulus }

					}
					oe_ghh_break_wing = { tag = KZN ind_name = OE_GHH_NAME_KZN_KHANATE }
					if = { limit = { c:GLH = { is_ai = no } oe_ghh_ev_month_free = yes } set_global_variable = { name = oe_ghh_ev_7 value = 1 } oe_ghh_send_ev = { n = 7 } }

							current_date >= 1450.1.1
							AND = { has_global_variable = oe_ghh_rel_NOG global_var:oe_ghh_cohesion < 15 }

		}

oe_ghh_convert_wings_to_vassal = {
	if = {
		limit = { country_exists = c:GLH }
		c:GLH = {
			every_subject = {
				limit = { is_subject_type = oe_glh_ulus }
				change_subject_type = subject_type:vassal
				if = { limit = { has_country_modifier = oe_ghh_ulus_loyal } remove_country_modifier = oe_ghh_ulus_loyal }
				if = { limit = { has_country_modifier = oe_ghh_ulus_restless } remove_country_modifier = oe_ghh_ulus_restless }

		}
	}
}
