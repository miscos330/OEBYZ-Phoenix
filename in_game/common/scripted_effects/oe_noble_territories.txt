# Location scope. No character/dynasty variables; one unique pair of modifiers per house slot.
oe_nf_apply_control_modifier = {
    if = {
        limit = { location_rank = location_rank:rural_settlement }
        if = { limit = { has_location_modifier = oe_nf_industry_$slot$ } remove_location_modifier = oe_nf_industry_$slot$ }
        if = { limit = { NOT = { has_location_modifier = oe_nf_land_$slot$ } } add_location_modifier = { modifier = oe_nf_land_$slot$ years = -1 mode = replace } }
    }
    else = {
        if = { limit = { has_location_modifier = oe_nf_land_$slot$ } remove_location_modifier = oe_nf_land_$slot$ }
        if = { limit = { NOT = { has_location_modifier = oe_nf_industry_$slot$ } } add_location_modifier = { modifier = oe_nf_industry_$slot$ years = -1 mode = replace } }
    }
}
oe_nf_claim_location = {
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    if = {
	        limit = { oe_nf_can_control = { slot = $slot$ } }
	        save_scope_as = oe_nf_claimed_location
	        international_organization:oe_byz_empire_pillars = { add_to_variable_list = { name = oe_nf_locations_$slot$ target = scope:oe_nf_claimed_location } }
	        oe_nf_apply_control_modifier = { slot = $slot$ }
	    }
	}
}
# international_organization:oe_byz_empire_pillars scope, called before releasing an extinct family slot.
oe_nf_release_territories = {
    hidden_effect = {
        every_location_in_the_world = {
            limit = { OR = { has_location_modifier = oe_nf_land_$slot$ has_location_modifier = oe_nf_industry_$slot$ } }
            if = { limit = { has_location_modifier = oe_nf_land_$slot$ } remove_location_modifier = oe_nf_land_$slot$ }
            if = { limit = { has_location_modifier = oe_nf_industry_$slot$ } remove_location_modifier = oe_nf_industry_$slot$ }
        }
    }
    clear_variable_list = oe_nf_locations_$slot$
    remove_variable = oe_nf_expansion_wait_$slot$
    remove_variable = oe_nf_recommend_cooldown_$slot$
    remove_variable = oe_daily_house_1_cooldown_$slot$
    remove_variable = oe_daily_house_2_cooldown_$slot$
    remove_variable = oe_daily_house_3_cooldown_$slot$
    remove_variable = oe_daily_house_4_cooldown_$slot$
    remove_variable = oe_daily_house_9_cooldown_$slot$
    remove_variable = oe_daily_house_11_cooldown_$slot$
    if = { limit = { has_variable = oe_nf_map_focus var:oe_nf_map_focus = $slot$ } remove_variable = oe_nf_map_focus }
}
# Rank changes: retain the earliest slot if a shared town becomes rural.
# Existing holdings persist through ownership changes; acquisition is BYZ-only.
oe_nf_clean_territory_slot = {
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    clear_variable_list = oe_nf_kept_locations
	    every_in_list = {
	        variable = oe_nf_locations_$slot$
	        if = {
	            limit = {
	                this != location:mount_athos
	                NOT = { oe_nf_province_protected = yes }
	                OR = {
	                    NOT = { location_rank = location_rank:rural_settlement }
	                    NOT = { oe_nf_earlier_control_$slot$ = yes }
	                }
	            }
	            oe_nf_apply_control_modifier = { slot = $slot$ }
	            save_scope_as = oe_nf_kept_location
	            international_organization:oe_byz_empire_pillars = { add_to_variable_list = { name = oe_nf_kept_locations target = scope:oe_nf_kept_location } }
	        }
	        else = {
	            if = { limit = { has_location_modifier = oe_nf_land_$slot$ } remove_location_modifier = oe_nf_land_$slot$ }
	            if = { limit = { has_location_modifier = oe_nf_industry_$slot$ } remove_location_modifier = oe_nf_industry_$slot$ }
	        }
	    }
	    clear_variable_list = oe_nf_locations_$slot$
	    every_in_list = {
	        variable = oe_nf_kept_locations
	        save_scope_as = oe_nf_kept_location
	        international_organization:oe_byz_empire_pillars = { add_to_variable_list = { name = oe_nf_locations_$slot$ target = scope:oe_nf_kept_location } }
	    }
	    clear_variable_list = oe_nf_kept_locations
	}
}

# international_organization:oe_byz_empire_pillars scope. Exposes the successful target so costs are charged only for real expansion.
oe_nf_expand_once = {
    clear_saved_scope = oe_nf_expanded_location
    c:BYZ = {
        random_owned_location = {
            limit = { oe_nf_expansion_target = { slot = $slot$ } }
            save_scope_as = oe_nf_expanded_location
            oe_nf_claim_location = { slot = $slot$ }
        }
    }
}
oe_nf_expand_five_years = {
    if = {
        limit = { c:BYZ = { NOT = { has_variable = oe_ir_reclaim_privileges_done } } has_variable = oe_nf_dynasty_$slot$ NOT = { has_variable = oe_nf_expansion_wait_$slot$ } }
        set_variable = { name = oe_nf_expansion_wait_$slot$ value = 1 months = 60 }
        oe_nf_expand_once = { slot = $slot$ }
        if = {
            limit = { var:oe_nf_influence_$slot$ > 400 }
            # Plan both connected extra targets before spending 150 or applying either claim.
            clear_saved_scope = oe_nf_extra_first
            clear_saved_scope = oe_nf_extra_second
            c:BYZ = {
                random_owned_location = {
                    limit = { oe_nf_expansion_target = { slot = $slot$ } }
                    save_scope_as = oe_nf_extra_first
                }
                if = {
                    limit = { exists = scope:oe_nf_extra_first }
                    random_owned_location = {
                        limit = {
                            oe_nf_can_control = { slot = $slot$ }
                            this != scope:oe_nf_extra_first
                            OR = {
                                any_neighbor_location = { oe_nf_controls_slot = { slot = $slot$ } }
                                is_neighbor_of_location = scope:oe_nf_extra_first
                            }
                        }
                        save_scope_as = oe_nf_extra_second
                    }
                }
            }
            if = {
                limit = { exists = scope:oe_nf_extra_first exists = scope:oe_nf_extra_second }
                scope:oe_nf_extra_first = { oe_nf_claim_location = { slot = $slot$ } }
                scope:oe_nf_extra_second = { oe_nf_claim_location = { slot = $slot$ } }
                oe_nf_change_influence = { slot = $slot$ amount = -150 }
            }
        }
        else_if = {
            limit = { var:oe_nf_influence_$slot$ > 100 var:oe_nf_influence_$slot$ <= 400 }
            oe_nf_expand_once = { slot = $slot$ }
            if = { limit = { exists = scope:oe_nf_expanded_location } oe_nf_change_influence = { slot = $slot$ amount = -50 } }
        }
    }
}
