oe_rw_start_griko_romanization = {
    if = {
        limit = { oe_rw_start_griko_romanization_allowed = yes }
        if = { limit = { area:abruzzo_area = { any_location_in_area = { oe_rw_griko_spread_location = yes } } } area:abruzzo_area = { oe_rw_spawn_griko_in_area = yes } }
        else_if = { limit = { area:apulia_area = { any_location_in_area = { oe_rw_griko_spread_location = yes } } } area:apulia_area = { oe_rw_spawn_griko_in_area = yes } }
        else_if = { limit = { area:basilicata_area = { any_location_in_area = { oe_rw_griko_spread_location = yes } } } area:basilicata_area = { oe_rw_spawn_griko_in_area = yes } }
        else_if = { limit = { area:campania_area = { any_location_in_area = { oe_rw_griko_spread_location = yes } } } area:campania_area = { oe_rw_spawn_griko_in_area = yes } }
        else_if = { limit = { area:sicily_area = { any_location_in_area = { oe_rw_griko_spread_location = yes } } } area:sicily_area = { oe_rw_spawn_griko_in_area = yes } }
    }
}
oe_rw_spawn_griko_in_area = {
    ordered_location_in_area = {
        limit = { oe_rw_griko_spread_location = yes }
        max = 1
        check_range_bounds = no
        order_by = { value = population multiply = oe_rw_assim_griko_source_share }
        spawn_movement = {
            movement_definition = movement_definition:oe_griko_identity_movement
            supporters = { value = population multiply = oe_rw_assim_griko_source_share multiply = 0.1 }
        }
        oe_identity_confirm_start = { holder = culture:griko_culture movement = oe_griko_identity_movement event = oe_identity_start_repair.18 }
    }
}
