# Upgraded by fx23 (Decision 306 R3B) to two-tier unified origin selection.
oe_rw_start_griko_romanization = {
    if = {
        limit = { oe_rw_start_griko_romanization_allowed = yes }
        region:italy_region = {
            ordered_location_in_region = {
                limit = { oe_rw_griko_origin_ge25 = yes }
                max = 1
                check_range_bounds = no
                order_by = {
                    value = population
                    multiply = oe_rw_assim_griko_source_share
                    if = { limit = { oe_rw_griko_origin_ge50 = yes } add = 10000000 }
                    if = { limit = { OR = { this = location:palermo this = location:catania this = location:syracuse } } add = 1000000 }
                    if = { limit = { owner = c:BYZ } add = 100000000 }
                }
                save_scope_as = oe_rw_origin_loc
                spawn_movement = {
                    movement_definition = movement_definition:oe_griko_identity_movement
                    supporters = { value = population multiply = oe_rw_assim_griko_source_share multiply = 0.1 }
                }
                if = {
                    limit = {
                        culture:griko_culture = {
                            any_movement_in_culture = {
                                movement_type = movement_definition:oe_griko_identity_movement
                                num_locations_affected > 0
                            }
                        }
                    }
                    oe_identity_confirm_start = { holder = culture:griko_culture movement = oe_griko_identity_movement event = oe_identity_start_repair.18 }
                    c:BYZ = { set_variable = { name = oe_rw_origin_griko value = scope:oe_rw_origin_loc } }
                }
            }
        }
    }
}
