oe_heg_plan_persia_effect = {
	if = {
		limit = { exists = international_organization:oe_byz_roman_world }
	    add_gold = -1000
	    set_variable = oe_heg_persia_plan_done
	    international_organization:oe_byz_roman_world = {
	        set_variable = oe_rw_persia_frontier_included
	        region:caucasus_region = { every_location_in_region = { oe_heg_add_roman_location = yes } }
	        region:crescent_region = { every_location_in_region = { oe_heg_add_roman_location = yes } }
	    }
	}
}
oe_heg_invade_persia_effect = {
    add_gold = -200
    add_political_influence = -20
    add_stability = -3
    add_casus_belli = { target = scope:target_country type = casus_belli:oe_heg_capital_invasion }
}
oe_heg_restore_persia_effect = {
	if = {
		limit = { exists = international_organization:oe_byz_roman_world }
	    # Freeze the entire Byzantine bloc's Persian holdings before any country can disappear.
	    region:persia_region = {
	        every_location_in_region = { limit = { oe_heg_persia_grant_location = yes } add_to_list = oe_persia_grant_locations }
	        ordered_location_in_region = {
	            limit = { oe_heg_persia_grant_location = yes }
	            order_by = population
	            max = 1
	            check_range_bounds = no
	            save_scope_as = oe_persia_capital
	        }
	    }
	    scope:oe_persia_capital = {
	        # Creation always uses a Byzantine-owned seed, even when all eligible land is held by subjects.
	        if = { limit = { NOT = { owner = c:BYZ } } change_location_owner = c:BYZ }
	        if = {
	        	limit = { has_owner = yes }
		        create_country_from_location = {
		            overlord = c:BYZ
		            subject_type = subject_type:oe_persian_empire
		            save_scope_as = oe_new_persian_empire
		            change_government_type = government_type:monarchy
		            change_country_type = location
		            change_heir_selection = heir_selection:cognatic_primogeniture
		            change_culture = culture:farsi_culture
		            change_religion = religion:zoroastrian
		            set_country_rank = country_rank:rank_empire
		            change_country_name = OE_RESTORED_PERSIAN_EMPIRE
		            change_country_adjective = OE_RESTORED_PERSIAN_EMPIRE_ADJ
		            change_country_flag = IRA
		            set_capital = scope:oe_persia_capital
		            oe_heg_stabilize_africa_effect = yes
		            set_variable = oe_restored_persian_empire
		            add_reform = government_reform:oe_persian_restoration
		            if = {
		                limit = { has_ruler = no }
		                create_character = {
		                    age = 35 culture = culture:farsi_culture religion = religion:zoroastrian
		                    estate = estate_type:nobles_estate save_scope_as = oe_persian_ruler
		                }
		                set_new_ruler = scope:oe_persian_ruler
		            }
		        }
	        }
	    }
	    every_in_list = { list = oe_persia_grant_locations change_location_owner = scope:oe_new_persian_empire }
	    set_variable = oe_heg_persia_restored
	    international_organization:oe_byz_roman_world = { add_country_to_international_organization = scope:oe_new_persian_empire }
	    oe_start_persian_religious_revival = yes
	}
}
oe_start_persian_religious_revival = {
    region:persia_region = {
        ordered_location_in_region = {
            limit = { oe_rw_zoroastrian_spread_location = yes }
            order_by = population
            max = 1
            check_range_bounds = no
            spawn_movement = {
                movement_definition = movement_definition:oe_persian_religious_revival
                supporters = {
                    value = "religion_percentage(religion:sunni)"
                    add = "religion_percentage(religion:shia)"
                    multiply = population
                    multiply = 0.1
                }
            }
            oe_identity_schedule_30_years = { holder = religion:zoroastrian event = oe_identity_expiry.22 }
        }
    }
}
