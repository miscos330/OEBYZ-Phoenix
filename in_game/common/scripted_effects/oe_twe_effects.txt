oe_twe_tick = {
	if = {
		limit = { is_situation_active = situation:oe_sit_two_emperors }
		oe_twe_sync = yes
		situation:oe_sit_two_emperors = {
			# Hub advantage gently shifts balance each month
			if = {
				limit = { var:oe_twe_hub_east > var:oe_twe_hub_west }
				change_variable = { name = oe_twe_balance add = 1 }
			}
			else_if = {
				limit = { var:oe_twe_hub_west > var:oe_twe_hub_east }
				change_variable = { name = oe_twe_balance add = -1 }
			}

			# HRE Imperial Authority influence on balance
			if = {
				limit = {
					exists = international_organization:hre
					international_organization:hre = {
						has_variable = imperial_authority
						var:imperial_authority >= 70
					}
				}
				change_variable = { name = oe_twe_balance add = -1 }
			}
			else_if = {
				limit = {
					exists = international_organization:hre
					international_organization:hre = {
						has_variable = imperial_authority
						var:imperial_authority <= 15
					}
				}
				change_variable = { name = oe_twe_balance add = 1 }
			}

			# War & War of Religions pressure on tension & concord
			if = {
				limit = { is_situation_active = situation:war_of_religions }
				change_variable = { name = oe_twe_tension add = 1 }
			}
			if = {
				limit = { oe_twe_emperors_at_war = yes }
				change_variable = { name = oe_twe_tension add = 2 }
				change_variable = { name = oe_twe_concord add = -2 }
			}
			else_if = {
				limit = { var:oe_twe_tension > 25 }
				change_variable = { name = oe_twe_tension add = -1 }
			}

			# Track 12-month Concord window (Outcome 2)
			if = {
				limit = {
					var:oe_twe_concord >= 80
					var:oe_twe_balance >= 35
					var:oe_twe_balance <= 65
					oe_twe_emperors_at_war = no
				}
				change_variable = { name = oe_twe_concord_months add = 1 }
			}
			else = {
				set_variable = { name = oe_twe_concord_months value = 0 }
			}

			# Track 12-month Direct Hegemony window (Outcome 1 alt path)
			if = {
				limit = { oe_twe_east_direct_hegemony = yes }
				change_variable = { name = oe_twe_hegemony_months add = 1 }
			}
			else = {
				set_variable = { name = oe_twe_hegemony_months value = 0 }
			}

			clamp_variable = { name = oe_twe_balance max = 100 min = 0 }
			clamp_variable = { name = oe_twe_concord max = 100 min = 0 }
			clamp_variable = { name = oe_twe_tension max = 100 min = 0 }
		}

		# Re-sync posture code after monthly tick
		oe_twe_sync = yes

		# Milestone multi-country narrative events (oe_twe.2 - oe_twe.6), max one per month via else_if chain
		if = {
			limit = {
				NOT = { has_global_variable = oe_twe_ev2_done }
				situation:oe_sit_two_emperors = {
					has_variable = oe_twe_tension
					var:oe_twe_tension >= 35
				}
			}
			set_global_variable = { name = oe_twe_ev2_done value = 1 }
			oe_twe_notify_both_emperors_2 = yes
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_twe_ev3_done }
				situation:oe_sit_two_emperors = {
					has_variable = oe_twe_tension
					var:oe_twe_tension >= 50
					has_variable = oe_twe_hub_east
					var:oe_twe_hub_east >= 1
				}
			}
			set_global_variable = { name = oe_twe_ev3_done value = 1 }
			oe_twe_notify_both_emperors_3 = yes
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_twe_ev4_done }
				country_exists = c:FRA
				situation:oe_sit_two_emperors = {
					has_variable = oe_twe_tension
					var:oe_twe_tension >= 40
				}
			}
			set_global_variable = { name = oe_twe_ev4_done value = 1 }
			oe_twe_notify_french_4 = yes
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_twe_ev5_done }
				is_situation_active = situation:war_of_religions
			}
			set_global_variable = { name = oe_twe_ev5_done value = 1 }
			oe_twe_notify_both_emperors_5 = yes
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = oe_twe_ev6_done }
				oe_twe_emperors_at_war = no
				situation:oe_sit_two_emperors = {

			oe_twe_notify_both_emperors_6 = yes
		}
	}

			limit = { has_global_variable = oe_twe_ended has_global_variable = oe_twe_on }
			remove_global_variable = oe_twe_on
		}
	}
}
