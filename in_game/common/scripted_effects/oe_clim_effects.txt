# Country scope. Buys genuine staple food from the realm's capital market using real gold
# via `buy_goods_from_market`, then distributes grain into the most food-stressed provinces.
oe_clim_granary_buy_food = {
	if = {
		limit = {
			exists = capital
			capital = { exists = market }
		}
		if = {
			limit = { oe_clim_is_east_asia_crop_zone = yes }
			buy_goods_from_market = {
				market = capital.market
				goods = goods:rice
				amount = market_goods_mild_gain
			}
		}
		else_if = {
			limit = { capital = { continent = continent:america } }
			buy_goods_from_market = {
				market = capital.market
				goods = goods:maize
				amount = market_goods_mild_gain
			}
		}
		else_if = {
			limit = {
				capital = {
					continent = continent:africa
					NOT = { sub_continent = sub_continent:north_africa }
				}
			}
			buy_goods_from_market = {
				market = capital.market
				goods = goods:millet
				amount = market_goods_mild_gain
			}
		}
		else = {
			buy_goods_from_market = {
				market = capital.market
				goods = goods:wheat
				amount = market_goods_mild_gain
			}
		}
	}
	ordered_province = {
		limit = {
			OR = {
				is_starving = yes
				province_food_percentage < 0.65
			}
		}
		order_by = {
			value = 1
			subtract = province_food_percentage
		}
		max = 4
		check_range_bounds = no
		change_province_food_percentage = 0.35
	}
	custom_tooltip = oe_clim_granary_buy_food_tt
}

			}
			else_if = {
				limit = {
					current_year >= 1820
					has_variable = oe_clim_had_cholera
					NOT = { has_variable = oe_clim_ev18_done }
				}
				set_variable = { name = oe_clim_ev18_done value = 1 }
				set_variable = { name = oe_clim_ev_y value = current_year }
				trigger_event_non_silently = oe_clim.18
			}
		}
	}
}
