# Country scope. Disabled countries still have no bureaucracy expenses or effects.
oe_remove_disabled_bureaucracies = {
    if = {
        limit = { NOT = { oe_settings_bureaucracy_enabled = yes } }

    every_current_bureaucracy = {
        save_scope_as = oe_disabled_bureaucracy
        prev = { remove_bureaucracy = scope:oe_disabled_bureaucracy }
    }
    }
}
