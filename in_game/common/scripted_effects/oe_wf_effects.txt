# Safely adjust tributary Royal Favor (oe_wf_huangen) clamped to [0, 100].
oe_wf_add_huangen = {
	if = {
		limit = { NOT = { has_variable = oe_wf_huangen } }
		set_variable = { name = oe_wf_huangen value = 0 }
	}
	change_variable = { name = oe_wf_huangen add = $amount$ }
	if = {
		limit = { var:oe_wf_huangen > 100 }
		set_variable = { name = oe_wf_huangen value = 100 }
	}
	if = {
		limit = { var:oe_wf_huangen < 0 }
		set_variable = { name = oe_wf_huangen value = 0 }
	}
}

# Propagate lightweight shared Celestial Calendar aura ("奉正朔") to tributaries/subjects (O(1) bounded, zero location loops).
oe_wf_propagate_era_to_tributaries = {
	if = {
		limit = {
			exists = international_organization:middle_kingdom
			is_leader_of_international_organization = international_organization:middle_kingdom
		}
		international_organization:middle_kingdom = {
			every_international_organization_member = {
				limit = {
					NOT = { this = root }
					oe_main_mod_old_world_event_allowed = yes
				}
				add_country_modifier = { modifier = oe_wf_era_tributary_calendar_mod years = 15 mode = replace }
			}
		}
	}
	every_subject = {
		limit = {
			oe_main_mod_old_world_event_allowed = yes
			NOT = { has_country_modifier = oe_wf_era_tributary_calendar_mod }
		}
		add_country_modifier = { modifier = oe_wf_era_tributary_calendar_mod years = 15 mode = replace }
	}
}
