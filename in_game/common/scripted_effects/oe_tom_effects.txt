# Northern Crusade Situation (oe_sit_northern_crusade) Lifecycle Effects
# ==============================================================================

oe_tom_sit_init = {
	set_global_variable = { name = oe_tom_sit_started value = 1 }
	situation:oe_sit_northern_crusade = {
		set_variable = { name = oe_tom_sit_momentum value = 50 }
		set_variable = { name = oe_tom_sit_ostsiedlung value = 25 }
		set_variable = { name = oe_tom_sit_marches value = 0 }
		set_variable = { name = oe_tom_sit_months value = 0 }
		set_variable = { name = oe_tom_sit_phase value = 1 }
	}
	oe_tom_sit_sync = yes
}

oe_tom_sit_sync = {
	if = {
		limit = { is_situation_active = situation:oe_sit_northern_crusade }
		situation:oe_sit_northern_crusade = {
			if = { limit = { NOT = { has_variable = oe_tom_sit_months } } set_variable = { name = oe_tom_sit_months value = 0 } }
			set_variable = { name = oe_tom_sit_momentum value = oe_tom_crusade_momentum_val }
			set_variable = { name = oe_tom_sit_ostsiedlung value = oe_tom_ostsiedlung_val }
			if = {
				limit = { country_exists = c:TEU }
				c:TEU = {
					situation:oe_sit_northern_crusade = {
						set_variable = { name = oe_tom_sit_marches value = prev.oe_tom_active_marches_val }
					}
				}
			}
			else = {
				set_variable = { name = oe_tom_sit_marches value = 0 }
			}
			if = {
				limit = { var:oe_tom_sit_momentum >= 75 }
				set_variable = { name = oe_tom_sit_phase value = 4 }
			}
			else_if = {
				limit = {
					country_exists = c:TEU
					OR = {
						AND = { country_exists = c:POL c:TEU = { is_at_war_with = c:POL } }
						AND = { country_exists = c:LIT c:TEU = { is_at_war_with = c:LIT } }
						current_year >= 1405
					}
				}
				set_variable = { name = oe_tom_sit_phase value = 3 }
			}
			else_if = {
				limit = { var:oe_tom_sit_ostsiedlung >= 40 }
				set_variable = { name = oe_tom_sit_phase value = 2 }
			}
			else = {
				set_variable = { name = oe_tom_sit_phase value = 1 }
			}
			clamp_variable = { name = oe_tom_sit_momentum max = 100 min = 0 }
			clamp_variable = { name = oe_tom_sit_ostsiedlung max = 100 min = 0 }
		}
	}
}

	if = {

		}
		oe_tom_sit_sync = yes
	}
}

	oe_tom_sync_marches = yes
	# Check 1242 Battle on the Ice Avenged (oe_tom.8)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev8_done }

		}
		set_variable = { name = oe_tom_ev8_done value = 1 }
		trigger_event_non_silently = oe_tom.8
	}
	# Check Tannenberg Reversed (oe_tom.9)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev9_done }
			OR = {
				owns_or_non_sovereign_subject_owns = location:vilnius
				owns_or_non_sovereign_subject_owns = location:kaunas
			}
			OR = {
				owns_or_non_sovereign_subject_owns = location:warsaw
				owns_or_non_sovereign_subject_owns = location:krakow
				owns_or_non_sovereign_subject_owns = location:poznan
			}
		}
		set_variable = { name = oe_tom_ev9_done value = 1 }
		trigger_event_non_silently = oe_tom.9
	}
	# Check Black Cross & Double-Headed Eagle Climax (oe_tom.10)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev10_done }
			has_global_variable = oe_tom_pontus_established
			oe_tom_holds_black_sea_anchor = yes
			OR = {
				country_exists = c:BYZ
				country_exists = c:ROM
			}
		}
		set_variable = { name = oe_tom_ev10_done value = 1 }
		trigger_event_non_silently = oe_tom.10
	}
	# Yearly Ordensburg Frontier Auto-Settlement Pulse (oe_tom.13, 5y cooldown)
	if = {
		limit = {
			oe_tom_has_any_march = yes
			NOT = { has_variable = oe_tom_ev13_cd }
			at_war = no
		}
		set_variable = { name = oe_tom_ev13_cd months = 60 }
		trigger_event_non_silently = oe_tom.13
	}
}
