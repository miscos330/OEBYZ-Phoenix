# ==============================================================================
# Counter-Crisis 2: Prussian Confederation (Preussischer Bund) Tension Tracker
# Unlocks 1410+ (or 1453 start initialized at 55). Mutually exclusive with flavor_teu.1!
# ==============================================================================

oe_tom_update_bund_tension = {
	if = {
		limit = {
			tag = TEU
			current_year >= 1410
			NOT = { has_variable = oe_teu_secularized_pru }
			NOT = { has_variable = oe_teu_bund_settled_perm }
			OR = {
				owns = location:gdansk
				owns = location:torun
				owns = location:elblag
			}
		}
		if = {
			limit = { NOT = { has_variable = oe_teu_bund_tension } }
			set_variable = { name = oe_teu_bund_tension value = 10 }
		}
		# +5/yr: hard state monopoly & unhappy burghers (< 50%)
		if = {
			limit = {
				country_has_estate = estate_type:burghers_estate
				estate_satisfaction:burghers_estate < 0.50
				OR = {
					has_policy = state_controlled_amber_gathering
					has_policy = oe_teu_pol_order_trade_monopoly
				}
			}
			change_variable = { name = oe_teu_bund_tension add = 5 }
		}
		# +6/yr: war exhaustion >= 4 OR recent defeat by Poland-Lithuania
		if = {
			limit = {
				OR = {
					war_exhaustion >= 4
					has_variable = oe_teu_recent_pl_defeat
				}
			}
			change_variable = { name = oe_teu_bund_tension add = 6 }
		}
		# +4/yr: vanilla extra taxes on bailiwicks OR burghers severely squeezed (< 35%)
		if = {
			limit = {
				OR = {
					has_country_modifier = teu_extra_taxes_on_bailwicks
					AND = {
						country_has_estate = estate_type:burghers_estate
						estate_satisfaction:burghers_estate < 0.35
					}
				}
			}
			change_variable = { name = oe_teu_bund_tension add = 4 }
		}
		# +3/yr: overextended eastern marches (>= 3 active Ordensmarks)
		if = {
			limit = { oe_tom_active_marches_val >= 3 }
			change_variable = { name = oe_teu_bund_tension add = 3 }
		}
		# -4/yr: peace + Hanseatic profit-sharing + satisfied burghers (>= 60%) + low war exhaustion (< 1)
		if = {
			limit = {
				at_war = no
				war_exhaustion < 1
				country_has_estate = estate_type:burghers_estate
				estate_satisfaction:burghers_estate >= 0.60
				OR = {
					has_policy = burghers_controlled_amber_gathering
					has_policy = oe_teu_pol_hanseatic_profit_sharing
					has_policy = oe_teu_pol_knight_merchant_confed
				}
			}
			change_variable = { name = oe_teu_bund_tension add = -4 }
		}
		clamp_variable = { name = oe_teu_bund_tension min = 0 max = 100 }
		# Stage 1 Warning (tension >= 55): Petition of Marienwerder (1440 historical analog)
		if = {
			limit = {
				has_variable = oe_teu_bund_tension
				var:oe_teu_bund_tension >= 55
				NOT = { has_variable = oe_teu_bund_warn_fired }
			}
			set_variable = { name = oe_teu_bund_warn_fired value = 1 }
			trigger_event_non_silently = oe_tom.17
		}
		# Stage 2 Eruption (tension >= 85): Mutual exclusion guard against vanilla flavor_teu.1!
		else_if = {
			limit = {
				has_variable = oe_teu_bund_tension
				var:oe_teu_bund_tension >= 85
				NOT = { has_variable = oe_teu_bund_revolt_fired }
			}
			set_variable = { name = oe_teu_bund_revolt_fired value = 1 }
			# If vanilla flavor_teu.1 already spawned an active rebellion, do NOT double-fire!
			if = {
				limit = {
					NOT = { any_rebel = { rebel_progress > 0 } }
				}
				trigger_event_non_silently = oe_tom.18
			}
		}
	}
}

oe_tom_sit_init = {
	set_global_variable = { name = oe_tom_sit_started value = 1 }
	set_global_variable = { name = oe_tom_sit_action_shift value = 0 }
	# 1453 start compatibility: Lithuania is already Catholic & Prussian Bund is on the eve of 1454 revolt!
	if = {
		limit = { current_year >= 1450 }
		set_global_variable = { name = oe_tom_crusade_crisis_active value = 1 }
		if = {
			limit = { country_exists = c:TEU }
			c:TEU = {
				set_variable = { name = oe_tom_ev15_done value = 1 }
				if = {
					limit = { NOT = { has_variable = oe_teu_bund_tension } }
					set_variable = { name = oe_teu_bund_tension value = 55 }
				}
				add_country_modifier = { modifier = oe_tom_mod_crusade_legitimacy_waning years = -1 mode = replace }
			}
		}
	}
	situation:oe_sit_northern_crusade = {
		set_variable = { name = oe_tom_sit_momentum value = 50 }
		set_variable = { name = oe_tom_sit_bund_tension value = 0 }
		set_variable = { name = oe_tom_sit_marches value = 0 }
		set_variable = { name = oe_tom_sit_months value = 0 }
		set_variable = { name = oe_tom_sit_phase value = 1 }
	}
	oe_tom_sit_sync = yes
}

oe_tom_sit_sync = {
	if = {
		limit = { is_situation_active = situation:oe_sit_northern_crusade }
		situation:oe_sit_northern_crusade = {
			if = { limit = { NOT = { has_variable = oe_tom_sit_months } } set_variable = { name = oe_tom_sit_months value = 0 } }
			set_variable = { name = oe_tom_sit_momentum value = oe_tom_crusade_momentum_val }
			set_variable = { name = oe_tom_sit_bund_tension value = oe_tom_bund_tension_sit_val }
			if = {
				limit = { country_exists = c:TEU }
				c:TEU = {
					situation:oe_sit_northern_crusade = {
						set_variable = { name = oe_tom_sit_marches value = prev.oe_tom_active_marches_val }
					}
				}
			}
			else_if = {
				limit = { country_exists = c:PRU c:PRU = { OR = { has_variable = oe_teu_secularized_pru has_variable = oe_teu_ordensstaat_active } } }
				c:PRU = {
					situation:oe_sit_northern_crusade = {
						set_variable = { name = oe_tom_sit_marches value = prev.oe_tom_active_marches_val }
					}
				}
			}
			else = {
				set_variable = { name = oe_tom_sit_marches value = 0 }
			}
			if = {
				limit = { has_variable = oe_tom_sit_momentum var:oe_tom_sit_momentum >= 75 }
				set_variable = { name = oe_tom_sit_phase value = 4 }
			}
			else_if = {
				limit = {
					OR = {
						has_variable = oe_tom_sit_bund_tension
						has_global_variable = oe_tom_crusade_crisis_active
					}
					OR = {
						AND = { has_variable = oe_tom_sit_bund_tension var:oe_tom_sit_bund_tension >= 50 }
						current_year >= 1440
					}
				}
				set_variable = { name = oe_tom_sit_phase value = 3 }
			}
			else_if = {
				limit = { has_global_variable = oe_tom_crusade_crisis_active }
				set_variable = { name = oe_tom_sit_phase value = 2 }
			}
			else = {
				set_variable = { name = oe_tom_sit_phase value = 1 }
			}
			clamp_variable = { name = oe_tom_sit_momentum max = 100 min = 0 }
			clamp_variable = { name = oe_tom_sit_bund_tension max = 100 min = 0 }
		}
	}
}

oe_tom_sit_monthly = {
	if = {
		limit = { is_situation_active = situation:oe_sit_northern_crusade }

	}
}

oe_tom_yearly_host_tick = {
	oe_tom_sync_marches = yes
	# Counter-Crisis 1: 1385+ Union of Krewo & Lithuanian Baptism Crusade Legitimacy Collapse
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev15_done }
			NOT = { has_variable = oe_teu_secularized_pru }
			oe_tom_krewo_condition_met = yes
		}
		set_global_variable = { name = oe_tom_crusade_crisis_active value = 1 }
		set_variable = { name = oe_tom_ev15_done value = 1 }
		add_country_modifier = { modifier = oe_tom_mod_crusade_legitimacy_waning years = -1 mode = replace }
		trigger_event_non_silently = oe_tom.15
	}
	# Counter-Crisis 2: 1410+ Prussian Confederation Tension Update
	oe_tom_update_bund_tension = yes
	# Stage III: 1525 Reformation Dual Endgame Auto-Dispatch (TEU -> PRU vs Counter-Reformation Ordensstaat)
	if = {
		limit = {
			tag = TEU
			NOT = { has_variable = oe_teu_1525_choice_made }
			OR = {
				current_year >= 1525
				AND = {
					current_year >= 1520
					religion:lutheran = { is_religion_enabled = yes }
					"religion_percentage_in_country(religion:lutheran)" >= 0.15
				}
			}
		}
		set_variable = { name = oe_teu_1525_choice_made value = 1 }
		trigger_event_non_silently = oe_tom.19
	}
	# Check 1242 Battle on the Ice Avenged (Peipus-Pskov boundary!)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev8_done }
			owns_or_non_sovereign_subject_owns = location:pskov
			owns_or_non_sovereign_subject_owns = location:izborsk
		}
		set_variable = { name = oe_tom_ev8_done value = 1 }
		trigger_event_non_silently = oe_tom.8
	}
	# Check Tannenberg Trial (defeated POL/LIT or holds Kaunas & Plock)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev9_done }
			OR = {
				has_variable = oe_tom_won_pl_war
				AND = {
					owns_or_non_sovereign_subject_owns = location:kaunas
					owns_or_non_sovereign_subject_owns = location:plock
				}
			}
		}
		set_variable = { name = oe_tom_ev9_done value = 1 }
		trigger_event_non_silently = oe_tom.9
	}
	# Hidden Cross-Mod Black Sea Easter Egg (only if sandbox player manually reaches Crimea!)
	if = {
		limit = {
			NOT = { has_variable = oe_tom_ev10_done }
			has_global_variable = oe_tom_pontus_established
			oe_tom_holds_black_sea_anchor = yes
			OR = {
				country_exists = c:BYZ
				country_exists = c:ROM
			}
		}
		set_variable = { name = oe_tom_ev10_done value = 1 }
		trigger_event_non_silently = oe_tom.10
	}
}
