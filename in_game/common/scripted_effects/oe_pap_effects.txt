# Country scope. Enfeoffs conquered non-core border lands into a loyal Cardinal Legation (oe_pap_legation).
# NEVER cedes capital, location:rome, location:avignon, area:lazio_area, or Papal cores.
oe_pap_charter_legation_effect = {
	save_scope_as = oe_pap_overlord
	ordered_owned_location = {
		limit = { oe_pap_loc_eligible_for_legation = yes }
		order_by = population
		max = 1
		save_scope_as = oe_pap_legation_seat
	}
	if = {
		limit = { exists = scope:oe_pap_legation_seat }
		scope:oe_pap_legation_seat = {
			area = { save_scope_as = oe_pap_legation_area }
			region = { save_scope_as = oe_pap_legation_region }
		}
		every_owned_location = {
			limit = {
				oe_pap_loc_eligible_for_legation = yes
				area = scope:oe_pap_legation_area
			}
			add_to_list = oe_pap_legation_locs
		}
		ordered_owned_location = {
			limit = {
				oe_pap_loc_eligible_for_legation = yes
				region = scope:oe_pap_legation_region
				NOT = { is_in_list = oe_pap_legation_locs }
			}
			order_by = population
			max = 4
			add_to_list = oe_pap_legation_locs
		}
		scope:oe_pap_legation_seat = {
			create_country_from_location = {
				overlord = scope:oe_pap_overlord
				subject_type = subject_type:oe_pap_legation
				change_government_type = government_type:theocracy
				change_country_type = location
				set_capital = scope:oe_pap_legation_seat
				change_culture = scope:oe_pap_overlord.culture
				change_religion = religion:catholic
				save_scope_as = oe_pap_new_legation
			}
		}
		if = {
			limit = { exists = scope:oe_pap_new_legation }
			scope:oe_pap_legation_seat = {
				add_core = scope:oe_pap_new_legation
			}
			every_in_list = {
				list = oe_pap_legation_locs
				limit = {
					NOT = { this = scope:oe_pap_legation_seat }
					oe_pap_loc_eligible_for_legation = yes
					has_owner = yes
					owner = scope:oe_pap_overlord
				}
				change_location_owner = scope:oe_pap_new_legation
				change_location_controller = scope:oe_pap_new_legation
				add_core = scope:oe_pap_new_legation
			}
			scope:oe_pap_new_legation = {
				if = {
					limit = { NOT = { is_subject_of = scope:oe_pap_overlord } }
					make_subject_of = {
						target = scope:oe_pap_overlord
						type = subject_type:oe_pap_legation
					}
				}
				else_if = {
					limit = { NOT = { is_subject_type = oe_pap_legation } }
					change_subject_type = subject_type:oe_pap_legation
				}
				set_variable = { name = oe_pap_is_legation value = 1 }
				if = {
					limit = { NOT = { has_reform = government_reform:oe_pap_legation_reform } }
					add_reform = government_reform:oe_pap_legation_reform
				}
				add_country_modifier = { modifier = oe_pap_mod_legation_charter years = 25 mode = replace }
				add_opinion = { target = scope:oe_pap_overlord modifier = oe_pap_op_legation_chartered }
				add_liberty_desire = -20
			}
			trigger_event_non_silently = oe_pap.7
		}
	}
}
