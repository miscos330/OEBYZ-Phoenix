oe_heg_plan_arabia_effect = {
    add_gold = -2000
    set_variable = oe_heg_arabia_plan_done
    international_organization:oe_byz_roman_world = {
        set_variable = oe_rw_arabia_included
        region:arabia_region = { every_location_in_region = { oe_heg_add_roman_location = yes } }
    }
}
oe_heg_invade_arabia_effect = {
    add_gold = -200
    add_casus_belli = { target = scope:target_country type = casus_belli:oe_heg_capital_invasion }
}
oe_heg_restore_ghassanids_effect = {
    add_gold = -1000
    add_political_influence = -25
    set_variable = oe_heg_ghassanids_pending
    trigger_event_non_silently = oe_ghassanid_restoration.1
}

# Each branch snapshots ONLY direct Byzantine holdings before country creation.
oe_heg_ghassanids_unified = {
    region:arabia_region = {
        every_location_in_region = {
            limit = { oe_heg_arabia_grant_location = yes }
            add_to_list = oe_ghassanid_grant_locations
        }
    }
    oe_heg_create_ghassanid_country = { subject = oe_arabian_kingdom }
    if = {
        limit = { exists = scope:oe_new_ghassanid_kingdom }
        scope:oe_new_ghassanid_kingdom = {
            add_country_modifier = { modifier = oe_unify_arabia years = -1 mode = replace }
        }
        oe_rw_start_arabia_romanization = yes
    }
}
oe_heg_ghassanids_frontier = {
    area:hail_area = {
        every_location_in_area = {
            limit = { oe_heg_arabia_grant_location = yes }
            add_to_list = oe_ghassanid_grant_locations
        }
    }
    oe_heg_create_ghassanid_country = { subject = oe_roman_march }
    if = {
        limit = { exists = scope:oe_new_ghassanid_kingdom }
        add_country_modifier = { modifier = oe_conquer_arabia years = -1 mode = replace }
    }
}
# Country event root remains BYZ for both options; no world-country iterator.
oe_heg_create_ghassanid_country = {
    ordered_in_list = {
        list = oe_ghassanid_grant_locations
        order_by = population
        max = 1 check_range_bounds = no
        save_scope_as = oe_ghassanid_capital
    }
    if = {
        limit = { exists = scope:oe_ghassanid_capital }
        scope:oe_ghassanid_capital = {
            create_country_from_location = {
                overlord = c:BYZ
                subject_type = subject_type:$subject$
                save_scope_as = oe_new_ghassanid_kingdom
                change_government_type = government_type:monarchy
                change_country_type = location
                change_heir_selection = heir_selection:cognatic_primogeniture
                change_culture = culture:oe_arabian_greek_culture
                change_religion = religion:orthodox
                set_country_rank = country_rank:rank_kingdom
                change_country_name = OE_GHASSANID_KINGDOM
                change_country_adjective = OE_GHASSANID_KINGDOM_ADJ
                change_country_flag = BYZ
                set_capital = scope:oe_ghassanid_capital
                oe_heg_stabilize_africa_effect = yes
                if = {
                    limit = { has_ruler = no }
                    create_character = {
                        age = 35 culture = culture:oe_arabian_greek_culture religion = religion:orthodox
                        estate = estate_type:nobles_estate save_scope_as = oe_ghassanid_ruler
                    }
                    set_new_ruler = scope:oe_ghassanid_ruler
                }
            }
        }
    }
    if = {
        limit = { exists = scope:oe_new_ghassanid_kingdom }
        every_in_list = { list = oe_ghassanid_grant_locations change_location_owner = scope:oe_new_ghassanid_kingdom }
        set_variable = oe_heg_ghassanids_restored
        international_organization:oe_byz_roman_world = { add_country_to_international_organization = scope:oe_new_ghassanid_kingdom }
    }
    else = {
        # Holdings may be lost while the event awaits a choice; do not charge for a failed creation.
        add_gold = 1000
        add_political_influence = 25
    }
    remove_variable = oe_heg_ghassanids_pending
}
