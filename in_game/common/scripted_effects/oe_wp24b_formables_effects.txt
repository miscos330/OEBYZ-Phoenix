# Country scope. Dispatches the one-time Founding Destiny event for any eligible flagship realm.
oe_wp24b_dispatch_founding_event = {
	if = {
		limit = {
			oe_wp24b_is_rom_host = yes
			NOT = { has_variable = oe_wp24b_rom_founded }
		}
		set_variable = { name = oe_wp24b_rom_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.101
	}
	else_if = {
		limit = {
			oe_wp24b_is_lat_host = yes
			NOT = { has_variable = oe_wp24b_lat_founded }
		}
		set_variable = { name = oe_wp24b_lat_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.201
	}
	else_if = {
		limit = {
			oe_wp24b_is_pru_host = yes
			NOT = { has_variable = oe_wp24b_pru_founded }
		}
		set_variable = { name = oe_wp24b_pru_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.301
	}
	else_if = {
		limit = {
			oe_wp24b_is_mng_host = yes
			NOT = { has_variable = oe_wp24b_mng_founded }
		}
		set_variable = { name = oe_wp24b_mng_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.401
	}
	else_if = {
		limit = {
			oe_wp24b_is_mch_host = yes
			NOT = { has_variable = oe_wp24b_mch_founded }
		}
		set_variable = { name = oe_wp24b_mch_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.501
	}
	else_if = {
		limit = {
			oe_wp24b_is_yua_host = yes
			NOT = { has_variable = oe_wp24b_yua_founded }
		}
		set_variable = { name = oe_wp24b_yua_founded value = 1 }
		trigger_event_non_silently = oe_wp24b.601
	}
}
