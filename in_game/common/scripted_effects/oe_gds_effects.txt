# Zero-slot yearly kernel hook (invoked once per calendar year from oe_gk_yearly_dispatch).
# Self-deduplicates via global_var:oe_gds_ran_y ?= current_year, short-circuits cleanly before 1520,
# bridges vanilla Suez/Panama canal completions into triumph events, and dispatches:
#   - 4 Strategic Trade Goods breakthroughs (oe_gds.1-4, max 1/year/country)
#   - 7 Vanilla Late-Game Situations cross-Eurasian bridges (oe_gds.10-16, max 1/year/country)
# Canal triumph events (oe_gds.5-9) are triggered directly by the 5 canal megaproject decisions.
oe_gds_kernel_yearly = {
	if = {
		limit = {
			current_year >= 1520
			NOT = { global_var:oe_gds_ran_y ?= current_year }
		}
		set_global_variable = { name = oe_gds_ran_y value = current_year }

		# 1. Vanilla Canal Completion Bridge (Suez & Panama)
		if = {
			limit = {
				NOT = { has_global_variable = oe_gds_suez_canal_built }
				location:suez = {
					has_owner = yes
					is_canal_open = yes
				}
			}
			set_global_variable = { name = oe_gds_suez_canal_built value = 1 }
			location:suez = {
				add_location_modifier = { modifier = oe_gds_loc_suez_canal_hub years = -1 mode = replace }
				owner = {
					if = {
						limit = { is_rebel_country = no }
						set_variable = { name = oe_gds_built_suez value = 1 }
						add_country_modifier = { modifier = oe_gds_mod_suez_canal_dominion years = -1 mode = replace }
						trigger_event_non_silently = oe_gds.5
					}
				}
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = oe_gds_panama_canal_built }
				location:panama = {
					has_owner = yes
					is_canal_open = yes
				}
			}
			set_global_variable = { name = oe_gds_panama_canal_built value = 1 }
			location:panama = {
				add_location_modifier = { modifier = oe_gds_loc_panama_canal_hub years = -1 mode = replace }
				owner = {
					if = {
						limit = { is_rebel_country = no }
						set_variable = { name = oe_gds_built_panama value = 1 }
						add_country_modifier = { modifier = oe_gds_mod_panama_canal_dominion years = -1 mode = replace }
						trigger_event_non_silently = oe_gds.6
					}
				}
			}
		}

		# 2. Strategic Trade Goods & Late-Game Situations Cross-Eurasian Bridges
		every_country = {
			limit = {
				oe_gds_eligible_strategic_host = yes
			}
			if = {
				limit = { current_year >= 1560 }
				oe_gds_goods_tick = yes
			}
			oe_gds_situations_tick = yes
		}
	}
}
