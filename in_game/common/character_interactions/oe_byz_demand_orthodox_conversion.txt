        looking_for_a = character
        source = actor
        target_flag = recipient

        none_available_msg_key = "OE_CONVERSION_NO_CANDIDATE"
        column = { data = name }
        visible = { oe_byz_orthodox_conversion_candidate = yes }

        if = {
            limit = {
                scope:actor = { oe_byz_orthodox_conversion_allowed = yes }

                text = OE_CONVERSION_AUTHORITY_COST

        }
    }
}
