oe_court_chancellor_filled = { scope = country is_shown = { oe_court_chancellor_filled = yes } }
oe_court_chancellor_dismiss = {
    scope = country
    is_shown = { oe_court_actor = yes oe_court_chancellor_filled = yes }
    is_valid = {
        oe_court_actor = yes
        oe_court_chancellor_filled = yes
        custom_tooltip = { text = OE_COURT_DISMISS_COST legitimacy >= 5 }
    }
    effect = {
        if = {
            limit = { oe_court_actor = yes oe_court_chancellor_filled = yes legitimacy >= 5 }
            add_legitimacy = -5
            var:oe_court_chancellor = { remove_variable = oe_court_office_held }
            remove_variable = oe_court_chancellor
        }
    }
}
