    scope = country
    is_valid = { has_variable = oe_bells_history_offset var:oe_bells_history_offset > 0 }
    effect = {
        if = {
            limit = { has_variable = oe_bells_history_offset }
            change_variable = { name = oe_bells_history_offset subtract = 100 }
            clamp_variable = { name = oe_bells_history_offset min = 0 }
        }
    }
}

        }
    }
}
