oe_court_auto_state = { scope = country is_shown = { has_variable = oe_court_auto_enabled } }
oe_court_auto_toggle = {
    scope = country
    is_shown = { oe_court_actor = yes oe_court_offices_active = yes }
    is_valid = { oe_court_actor = yes oe_court_offices_active = yes }
    effect = {
        if = {
            limit = { oe_court_actor = yes oe_court_offices_active = yes }
            if = {
                limit = { has_variable = oe_court_auto_enabled }
                remove_variable = oe_court_auto_enabled
            }
            else = { set_variable = oe_court_auto_enabled oe_court_auto_fill = yes }
        }
    }
}
