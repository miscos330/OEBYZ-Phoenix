oe_tx_remove_give = {
    scope = country
    saved_scopes = { target_location }
    is_shown = { tag = BYZ has_variable = oe_tx_partner }
    is_valid = { tag = BYZ is_target_in_variable_list = { name = oe_tx_give target = scope:target_location } }
    effect = {
        if = {
            limit = { tag = BYZ is_target_in_variable_list = { name = oe_tx_give target = scope:target_location } }
            remove_list_variable = { name = oe_tx_give target = scope:target_location }
            oe_tx_refresh = yes
        }
    }
}
