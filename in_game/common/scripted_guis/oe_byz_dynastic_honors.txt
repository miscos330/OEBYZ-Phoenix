oe_byz_co_emperor_panel = {
    scope = country
    is_shown = { tag = BYZ oe_byz_imperial_honors_allowed = yes }
}
oe_byz_co_emperor_filled = {
    scope = country
    is_shown = {
        tag = BYZ
        oe_byz_imperial_honors_allowed = yes
        var:oe_byz_co_emperor_display ?= { is_alive = yes has_character_modifier = oe_byz_select_as_second_empire_mod }
    }
}
oe_byz_refresh_co_emperor_display = {
    effect = {
        if = {
            limit = {
                country_exists = c:BYZ
                c:BYZ = {
                    is_ai = no
                    oe_byz_imperial_honors_allowed = yes
                }
            }
            c:BYZ = { oe_byz_refresh_co_emperor_display = yes }
        }
    }
}
