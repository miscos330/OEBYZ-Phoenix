# Read-only panel predicates; no character/population iteration and no click effects.
oe_gr_emperor_alive = {
    scope = country
    is_shown = { has_variable = oe_gr_emperor var:oe_gr_emperor = { is_alive = yes } }
}
oe_gr_emperor_detained = {
    scope = country
    is_shown = { has_variable = oe_gr_emperor var:oe_gr_emperor = { is_alive = yes has_character_modifier = oe_gr_imprisoned_emperor } }
}
oe_gr_emperor_free = {
    scope = country
    is_shown = { has_variable = oe_gr_emperor var:oe_gr_emperor = { is_alive = yes NOT = { has_character_modifier = oe_gr_imprisoned_emperor } } }
}
oe_gr_emperor_succession_pending = {
    scope = country
    is_shown = { has_variable = oe_gr_succession_pending }
}
oe_gr_emperor_extinct = {
    scope = country
    is_shown = { has_variable = oe_gr_royal_line_extinct }
}
oe_gr_emperor_unassigned = {
    scope = country
    is_shown = {
        NOT = { has_variable = oe_gr_emperor }
        NOT = { has_variable = oe_gr_succession_pending }
        NOT = { has_variable = oe_gr_royal_line_extinct }
    }
}
