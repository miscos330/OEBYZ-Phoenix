# Read-only inventory predicates. Construction uses the fixed-price expansion transaction.
oe_city_has_building = {
    scope = country
    saved_scopes = { city_building_type }
    is_shown = {
        oe_city_access_trigger = yes
        scope:city_building_type = { oe_city_special_building_trigger = yes }
        location:constantinople = {
            any_buildings_in_location = {
                building_type = scope:city_building_type
                OR = { building_level > 0 building_levels_under_construction > 0 }
            }
        }
    }
}
