# 1. The Golden Bull & Electoral Constitution (1345-1410)
# =============================================================================
oe_sit_v45_hre_golden_bull = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_oe_sit_v45_hre_golden_bull

	can_start = {
		current_year >= 1345
		current_year <= 1410
		NOT = { has_global_variable = oe_v45_hre_bull_ended }
		exists = international_organization:hre
		international_organization:hre = {
			international_organization_has_leader = yes
		}
	}

	visible = {
		is_rebel_country = no
		exists = international_organization:hre
		OR = {
			is_member_of_international_organization = international_organization:hre
			capital ?= { region = region:north_german_region }
			capital ?= { region = region:south_german_region }
		}
	}

	# Outcome 1: Imperial Golden Bull Supremacy (+65..+100)
	outcome = {
		name = "oe_v45_hre_bull_out_imperial"
		desc = "oe_v45_hre_bull_out_imperial_desc"
		trigger = {
			exists = international_organization:hre
			international_organization:hre = { international_organization_has_leader = yes }
			trigger_if = {
				limit = { has_variable = oe_v45_hre_constitution }
				var:oe_v45_hre_constitution >= 65
			}
			trigger_else = { always = no }
			trigger_if = {
				limit = { has_variable = oe_v45_hre_bull_codification }
				var:oe_v45_hre_bull_codification >= 70
			}
			trigger_else = { always = no }
		}
		immediate = {
			set_global_variable = { name = oe_v45_hre_bull_ended value = 1 }
			if = {
				limit = { exists = international_organization:hre }
				change_imperial_authority = { value = 8 }
				international_organization:hre = {
					leader_country ?= {
						if = {
							limit = { is_rebel_country = no }
							add_prestige = 20
							add_country_modifier = { modifier = oe_v45_hre_mod_bull_imperial_prerogative years = 35 mode = replace }
						}
					}
				}
			}
		}
	}

	# Outcome 2: Electoral Estates Constitution (-100..-65)
	outcome = {
		name = "oe_v45_hre_bull_out_electoral"
		desc = "oe_v45_hre_bull_out_electoral_desc"
		trigger = {
			exists = international_organization:hre
			trigger_if = {
				limit = { has_variable = oe_v45_hre_constitution }
				var:oe_v45_hre_constitution <= -65
			}
			trigger_else = { always = no }
			trigger_if = {
				limit = { has_variable = oe_v45_hre_bull_codification }
				var:oe_v45_hre_bull_codification >= 70
			}
			trigger_else = { always = no }
		}
		immediate = {
			set_global_variable = { name = oe_v45_hre_bull_ended value = 2 }
			if = {
				limit = { exists = international_organization:hre }
				international_organization:hre = {
					every_international_organization_member = {
						limit = {
							is_rebel_country = no
							OR = {
								has_special_status_in_international_organization = { type = special_status:elector international_organization = international_organization:hre }
								has_special_status_in_international_organization = { type = special_status:archbishop_elector international_organization = international_organization:hre }
								has_special_status_in_international_organization = { type = special_status:free_city international_organization = international_organization:hre }
							}
						}
						add_prestige = 12
						add_country_modifier = { modifier = oe_v45_hre_mod_bull_electoral_liberties years = 35 mode = replace }
					}
				}
			}
		}
	}

	# Outcome 3: Compromise of Nuremberg & Metz (Era Close or HRE Dissolved)
	outcome = {
		name = "oe_v45_hre_bull_out_compromise"
		desc = "oe_v45_hre_bull_out_compromise_desc"
		trigger = {
			OR = {
				NOT = { exists = international_organization:hre }
				current_year >= 1410
			}
		}
		immediate = {
			set_global_variable = { name = oe_v45_hre_bull_ended value = 3 }
			if = {
				limit = { exists = international_organization:hre }
				international_organization:hre = {
					leader_country ?= {
						if = {
							limit = { is_rebel_country = no }
							add_country_modifier = { modifier = oe_v45_hre_mod_bull_metz_compromise years = 30 mode = replace }
						}
					}
				}
			}
		}
	}

	on_start = {
		set_variable = { name = oe_v45_hre_constitution value = 0 }
		set_variable = { name = oe_v45_hre_constitution_bar value = 100 }
		set_variable = { name = oe_v45_hre_bull_codification value = 20 }
		if = {
			limit = {
				exists = international_organization:hre
				international_organization:hre = { international_organization_has_leader = yes }
			}
			international_organization:hre = {
				leader_country ?= {
					if = {
						limit = {
							is_rebel_country = no
							NOT = { has_variable = oe_v45_hre_ev_1_seen }
						}
						set_variable = { name = oe_v45_hre_ev_1_seen value = 1 }
						trigger_event_non_silently = oe_v45_hre.1
					}
				}
			}
		}
	}

	on_monthly = {
		if = { limit = { NOT = { has_variable = oe_v45_hre_constitution } } set_variable = { name = oe_v45_hre_constitution value = 0 } }
		if = { limit = { NOT = { has_variable = oe_v45_hre_bull_codification } } set_variable = { name = oe_v45_hre_bull_codification value = 20 } }
		if = {
			limit = { exists = international_organization:hre }
			if = {
				limit = {
					international_organization:hre = {
						international_organization_has_leader = yes
						international_organization_has_internal_peace = yes
					}
					var:oe_v45_hre_bull_codification < 75
				}
				change_variable = { name = oe_v45_hre_bull_codification add = 1 }
			}
			if = {
				limit = {
					international_organization:hre = {
						international_organization_has_leader = yes
						leader_country ?= { stability >= 15 at_war = no }
					}
					var:oe_v45_hre_constitution < 45
				}
				change_variable = { name = oe_v45_hre_constitution add = 1 }
			}
			else_if = {
				limit = {
					international_organization:hre = {
						international_organization_has_leader = no
					}
					var:oe_v45_hre_constitution > -65
				}
				change_variable = { name = oe_v45_hre_constitution add = -2 }
			}
		}
		if = { limit = { var:oe_v45_hre_constitution > 100 } set_variable = { name = oe_v45_hre_constitution value = 100 } }
		if = { limit = { var:oe_v45_hre_constitution < -100 } set_variable = { name = oe_v45_hre_constitution value = -100 } }
		set_variable = { name = oe_v45_hre_constitution_bar value = var:oe_v45_hre_constitution }
		change_variable = { name = oe_v45_hre_constitution_bar add = 100 }
		if = { limit = { var:oe_v45_hre_bull_codification > 100 } set_variable = { name = oe_v45_hre_bull_codification value = 100 } }
		if = { limit = { var:oe_v45_hre_bull_codification < 0 } set_variable = { name = oe_v45_hre_bull_codification value = 0 } }
	}

	on_ended = {
		set_global_variable = { name = oe_v45_hre_bull_ended value = 1 }
	}

	tooltip = {
		if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= { is_leader_of_international_organization = international_organization:hre }
			}
			custom_tooltip = oe_v45_hre_map_emperor_tt
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= {
					OR = {
						has_special_status_in_international_organization = { type = special_status:elector international_organization = international_organization:hre }
						has_special_status_in_international_organization = { type = special_status:archbishop_elector international_organization = international_organization:hre }
					}
				}
			}
			custom_tooltip = oe_v45_hre_map_elector_tt
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= {
					has_special_status_in_international_organization = { type = special_status:free_city international_organization = international_organization:hre }
				}
			}
			custom_tooltip = oe_v45_hre_map_free_city_tt
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= { is_member_of_international_organization = international_organization:hre }
			}
			custom_tooltip = oe_v45_hre_map_prince_tt
		}
	}

	is_data_map = yes
	map_color = {
		if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= { is_leader_of_international_organization = international_organization:hre }
			}
			value = yellow_mid
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= {
					OR = {
						has_special_status_in_international_organization = { type = special_status:elector international_organization = international_organization:hre }
						has_special_status_in_international_organization = { type = special_status:archbishop_elector international_organization = international_organization:hre }
					}
				}
			}
			value = tyrian
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= {
					has_special_status_in_international_organization = { type = special_status:free_city international_organization = international_organization:hre }
				}
			}
			value = blue_mid
		}
		else_if = {
			limit = {
				has_owner = yes
				exists = international_organization:hre
				owner ?= { is_member_of_international_organization = international_organization:hre }
			}
			value = brown_light
		}
		else = { value = define:NMapColors|DEFAULT_COLOR }
	}
}
