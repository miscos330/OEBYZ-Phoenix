phoenix_nirvana = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_phoenix_nirvana
	can_start = {
		current_year >= 1337
		always = no
	}
	can_end = {
		has_global_variable = phoenix_nirvana_is_end
	}
	visible = {
		has_or_had_tag = BYZ
	}
	on_start = {

	}
	on_monthly = {



	}
	on_ending = {}
	tooltip = {
		if = {
			limit = {
				has_owner = yes
				owner ?= { tag = BYZ }
			}
			custom_tooltip = phoenix_nirvana_byz_map_tt
		}
		if = {
			limit = {
				has_owner = yes
				country_exists = c:BYZ
				owner ?= { is_subject_of = c:BYZ }
			}
			custom_tooltip = phoenix_nirvana_byz_map_ttt
		}
		if = {
			limit = {
				has_owner = yes
				owner ?= { capital ?= { region = region:anatolia_region } culture = culture:turkish_culture }
			}
			custom_tooltip = phoenix_nirvana_byz_map_tttt
		}
	}

	map_color = {
		if = {
			limit = {
				has_owner = yes
				owner ?= { tag = BYZ }
			}
			value = owner.country_color
		}
		if = {
			limit = {
				has_owner = yes
				NOT = { owner ?= { tag = BYZ } }
				NOT = { AND = { country_exists = c:BYZ owner ?= { is_subject_of = c:BYZ } } }
				NOT = { owner ?= { religion = religion:catholic } }
				OR = {
					owner ?= { capital ?= { region = region:balkan_region } }
					owner ?= { capital ?= { region = region:anatolia_region } }
				}
			}
			value = owner.country_color
		}
		if = {
			limit = {
				has_owner = yes
				country_exists = c:BYZ
				owner ?= { is_subject_of = c:BYZ }
			}
			value = c:BYZ.country_color
		}
	}
}
