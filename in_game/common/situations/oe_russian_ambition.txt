oe_russian_ambition = {
    monthly_spawn_chance = monthly_spawn_chance_unique
    hint_tag = hint_oe_russian_ambition
    can_start = {
        country_exists = c:RUS
        has_global_variable = oe_rus_started
        NOT = { is_situation_active = situation:oe_rise_of_rus }
        NOT = { has_global_variable = oe_ra_started }
    }
    visible = { oe_ra_observer = yes }
    outcome = {
        name = oe_ra_outcome_fall
        desc = oe_ra_outcome_fall_desc
        trigger = { NOT = { country_exists = c:RUS } }
    }
    on_start = {
        set_global_variable = oe_ra_started
        c:RUS = {
            oe_ra_establish_historical_rivalries = yes
            trigger_event_non_silently = oe_russian_ambition.2
        }
        every_country = {
            limit = { oe_ra_observer = yes }
            trigger_event_non_silently = oe_russian_ambition.1
        }
    }
    map_color = {
        if = { limit = { has_owner = yes owner = c:RUS } value = owner.country_color }
        else_if = { limit = { has_owner = yes owner = { is_subject_or_below_of = c:RUS } } value = c:RUS.country_color }
        else_if = { limit = { has_owner = yes owner = { oe_ra_observer = yes } } value = owner.country_color }
        else = { value = define:NMapColors|DEFAULT_COLOR }
    }
}
