oe_byz_press_claims = {#收复帝国失地
	type = internationalorganization
	
	potential = {
		exists = international_organization:oe_byz_roman_world
	}

	allow = { scope:actor = { at_war = no has_raised_levies = no has_special_status_in_international_organization = { type = special_status:oe_byz_roman_world_byz international_organization = international_organization:oe_byz_roman_world } } }

	automation_tick = never
	automation_tick_frequency = 12
	ai_tick = never
	price = price:oe_byz_press_claims_price

	cooldown = {
		type = oe_byz_press_claims_cooldown_key
		years = 1
	}


	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		none_available_msg_key = "leave_international_organization_none"
		column = {
			data = name
		}
		visible = {
			scope:actor = {
					has_special_status_in_international_organization = {
						type = special_status:oe_byz_roman_world_byz
						international_organization = international_organization:oe_byz_roman_world
					}						
			}
			international_organization_type = international_organization_type:oe_byz_roman_world
		}
	}
	select_trigger = {
	    # Explicit 1.4 target overlay; nonselectable land remains unhighlighted.
	    only_color_selectable = yes
	    map_color = define:NMapColors|MAP_COLOR_HIGH
		looking_for_a = province
        # Include foreign provinces explicitly; the default actor source lists owned provinces.
        interaction_source_list = {
            every_country = {
                limit = { oe_roman_world_territorial_country_trigger = yes }
                every_province = {
                    limit = { OR = { region = region:anatolia_region region = region:balkan_region owner = { tag = HUN } } }
                    add_to_list = source
                }
            }
        }
		target_flag = target
		name = "choose_province"
		none_available_msg_key = "no_provinces_available"
		column = { data = owner_flag }
		column = { data = name }
		column = { data = population }
		visible = {
			scope:actor = { can_declare_war_on = root.owner }

				OR = {
					region = region:anatolia_region
					region = region:balkan_region
                    owner = { tag = HUN }
				}
		
			any_location_in_province = {
				is_neighbor_of = scope:actor
			}
			NOT = {
				owner = {
					OR = {
						this = scope:actor
						overlord ?= scope:actor
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = scope:actor
						}
					}
				}
			}
		}
	        enabled = {
            owner = {
                custom_tooltip = { text = OE_RW_CLAIM_RECOGNITION_BLOCK NOT = { has_variable = oe_rw_recognized_local_rule } }
            }
        }
    }
	effect = {
        if = {
            limit = { scope:target.owner = { tag = HUN } }
            scope:actor = { add_casus_belli = { target = scope:target.owner type = casus_belli:oe_cb_byz_vs_hun } }
        }
        else_if = {
			limit = {
				exists = scope:target
						scope:target.owner = c:SER

			}
			custom_tooltip = {
				text = oe_byz_press_claims_tt
				scope:actor = {
					trigger_event_silently = phoenix_nirvana.102
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:target
						scope:target.owner = c:BUL

			}
			custom_tooltip = {
				text = oe_byz_press_claims_tt
				scope:actor = {
					trigger_event_silently = phoenix_nirvana.103
				}
			}
		}		
		else_if = {
			limit = {
				exists = scope:target
						scope:target = { region = region:anatolia_region }

			}
			custom_tooltip = {
				text = oe_byz_press_claims_tt
				scope:actor = {
					trigger_event_silently = phoenix_nirvana.100
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:target
						scope:target = { region = region:balkan_region }
			}
			custom_tooltip = {
				text = oe_byz_press_claims_tt
				scope:actor = {
					trigger_event_silently = phoenix_nirvana.101
				}
			}
		}
		else = { custom_tooltip = oe_byz_press_claims_tt }
	}
}
