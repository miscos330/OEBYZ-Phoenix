oe_create_theme = {#创建军区
	type = internationalorganization
	show_message = no
	potential = {
		exists = international_organization:oe_byz_empire_pillars
		scope:actor = { oe_is_phoenix_empire = yes }
	}

	allow = {
		scope:actor = {
			at_war = no
			has_raised_levies = no
		}
	
        international_organization:oe_byz_empire_pillars ?= {
            custom_tooltip = { text = OE_PILLARS_AUTHORITY_REQUIRED_5 var:oe_pillars_authority >= 5 }
        }
}

	automation_tick = never
	automation_tick_frequency = 12
	ai_tick = never
	price = price:oe_byz_create_theme_price

	cooldown = {
		type = oe_create_theme_cooldown_key
		years = 0
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		none_available_msg_key = "leave_international_organization_none"
		column = {
			data = name
		}
		visible = {
			scope:actor = {
				oe_is_phoenix_empire = yes
			}
			international_organization_type = international_organization_type:oe_byz_empire_pillars
		}
	}

	select_trigger = {
	    # Explicit 1.4 target overlay; nonselectable land remains unhighlighted.
	    only_color_selectable = yes
	    map_color = define:NMapColors|MAP_COLOR_HIGH
		looking_for_a = province
		target_flag = target
		name = "choose_province"
		none_available_msg_key = "no_provinces_available"
		column = { data = owner_flag }
		column = { data = name }
		column = { data = population }
		visible = {
			# v26.0 WP11: inside the 79 Roman/historical regions a theme is founded from the
			# province holding the region centre (see scripted_triggers/oe_theme_regions.txt).
			# Outside them the original area rule still applies.
			OR = {
				any_location_in_province = { oe_theme_region_center_creatable = yes }
				AND = {
					NOT = { any_location_in_province = { has_variable = oe_theme_region } }
				oe_theme_creation_region_trigger = yes
				NOT = { area = area:thrace_area }
				any_location_in_province = {
					oe_owned_by_phoenix_empire = yes
				}
				NOT = {
					root.area = {
					any_location_in_area = {
						oe_theme_location_occupied_trigger = yes
					}
				}
				}
				}
			}
		}
	}
	effect = {
        international_organization:oe_byz_empire_pillars = { change_variable = { name = oe_pillars_authority add = -5 } }
		custom_tooltip = {
			text = oe_create_theme_tt
		# v26.0 WP11: inside the 79 Roman regions delegate to unified Option 2A Hub-deploying oe_theme_found_at_seat_empire;
		# outside them keep the legacy area fallback with Option 2A hub posts and BYZ/ROM overlord.
		if = {
			limit = { scope:target = { any_location_in_province = { oe_theme_region_center_creatable = yes } } }
			scope:target = { random_location_in_province = { limit = { oe_theme_region_center_creatable = yes } save_scope_as = oe_theme_seat } }
			scope:actor = {
				hidden_effect = { oe_theme_found_at_seat_empire = yes }
				oe_theme_empire_sync_caps = yes
			}
		}
		else = {
			scope:target = { random_location_in_province = { limit = { is_province_capital = yes } save_scope_as = oe_theme_seat } }
			scope:oe_theme_seat = {
				create_building_country_in_location = {
					religion = scope:actor.religion
					culture = culture:oe_roman_noble_culture
					subject_type = subject_type:oe_empire_theme
					overlord = scope:actor
					reforms = { oe_byz_theme_system_subject }
					hidden_effect = {
						change_government_type = government_type:monarchy
						change_heir_selection = heir_selection:oe_rome_province_elect
					}
					save_scope_as = target_building_country
				}
				construct_building = {
					building_type = building_type:oe_byz_theme_system_noble
					owner = scope:target_building_country
					cost_multiplier = 0
					cost_multiplier_reason = "game_concept_event"
				}
			}
			scope:target_building_country = {
				if = { limit = { exists = ruler } ruler = { add_trait = trait:oe_character_war_score_trait_VI } }
			}
			scope:oe_theme_seat = { save_scope_as = oe_theme_name_seat }
			scope:target_building_country = { oe_theme_apply_name = yes }
			international_organization:oe_byz_empire_pillars = {
				add_country_to_international_organization = scope:target_building_country
				international_organization_add_special_status = {
					type = special_status:oe_pillars_theme
					country = scope:target_building_country
				}
			}
			scope:target.area = {
				hidden_effect = {
					every_location_in_area = {
						limit = {
							oe_owned_by_phoenix_empire = yes
							oe_theme_is_hub_location = yes
							NOT = { this = scope:oe_theme_seat }
						}
						construct_building = {
							building_type = building_type:oe_byz_theme_system
							owner = scope:target_building_country
							cost_multiplier = 0
							cost_multiplier_reason = "game_concept_event"
						}
					}
				}
			}
			scope:oe_theme_seat = {
				while = {
					count = 3
					create_sub_unit_with_owner = { type = oe_byz_theme_subject_soldier_I owner = scope:target_building_country }
				}
				if = {
					limit = { is_coastal = yes }
					while = {
						count = 6
						create_sub_unit_with_owner = { type = n_cog owner = scope:target_building_country }
					}
				}
			}
			scope:target_building_country = {
				set_gold = 300
				add_manpower = 3
				if = { limit = { has_variable = oe_theme_named } oe_theme_admin_init = yes }
			}
			scope:actor = { oe_theme_empire_sync_caps = yes }
		}
	}}
}
