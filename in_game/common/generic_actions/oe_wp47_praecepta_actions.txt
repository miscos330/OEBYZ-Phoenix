# Calls shared oe_wp47_apply_* scripted effects synchronized with WP33 Phoenix Command Dock!

oe_wp47_act_mobilize_thematic_fleet = {
	type = owncountry
	sound = UI_action_government_generic
	show_message = no

	potential = {
		scope:actor = {
			oe_wp47_is_byz_praecepta_host = yes
		}
	}

	allow = {
		scope:actor = {
			has_ports = yes
			gold >= 75
			custom_tooltip = {
				text = oe_wp47_fleet_not_active_tt
				NOT = { has_country_modifier = oe_wp47_mod_mobilized_thematic_fleet }
				NOT = { has_variable = oe_wp47_fleet_cd }
			}
		}
	}

	cooldown = {
		type = oe_wp47_fleet_cd
		years = 5
	}

	automation_tick = never
	ai_tick = monthly
	ai_tick_frequency = 18
	ai_prerequisite = {
		is_rebel_country = no
		has_ports = yes
		at_war = yes
		gold >= 120
		NOT = { has_variable = oe_wp47_fleet_cd }
	}

	effect = {
		custom_tooltip = oe_wp47_act_mobilize_thematic_fleet_tt
		scope:actor = {
			oe_wp47_apply_mobilize_thematic_fleet = yes
		}
	}

	ai_will_do = {
		add = 0
		if = {
			limit = {
				scope:actor = {
					at_war = yes
					gold >= 150
					has_ports = yes
				}
			}
			add = 35
		}
	}
}
