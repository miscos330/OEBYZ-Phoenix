oe_bi_force_repression = {
    type = disaster
    show_message = no
    potential = { scope:actor = { tag = BYZ } }
    allow = { scope:actor = { oe_bi_force_ready = yes } }
    automation_tick = never ai_tick = never
    select_trigger = {
        looking_for_a = disaster target_flag = recipient name = "choose_disaster"
        column = { data = name }
        visible = { disaster_type = disaster_type:oe_byzantine_intrigue disaster_is_active = yes scope:actor = { tag = BYZ } }
    }
    select_trigger = {
        looking_for_a = character source = actor target_flag = target_head
        name = "OE_BI_CHOOSE_HOUSE" column = { data = character_info }
        interaction_source_list = { international_organization:oe_byz_empire_pillars = {
            if = { limit = { has_variable = oe_nf_dynasty_1 exists = var:oe_nf_head_1 } var:oe_nf_head_1 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_2 exists = var:oe_nf_head_2 } var:oe_nf_head_2 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_3 exists = var:oe_nf_head_3 } var:oe_nf_head_3 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_4 exists = var:oe_nf_head_4 } var:oe_nf_head_4 = { add_to_list = source } }
        } }
        visible = { oe_bi_rival_head = yes }
    }
    effect = {
        if = { limit = { scope:recipient = { disaster_type = disaster_type:oe_byzantine_intrigue disaster_is_active = yes } } scope:actor = { oe_bi_force_house = yes } }
    }
}
