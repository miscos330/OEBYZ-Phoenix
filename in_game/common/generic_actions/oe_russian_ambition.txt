oe_ra_expand_steppe = {
    type = situation
    show_message = no
    potential = { scope:actor = { oe_ra_actor = yes } }
    allow = { scope:actor = { oe_ra_actor = yes political_influence >= 25 gold >= oe_ra_steppe_cost } }
    cooldown = { type = oe_ra_expand_steppe_cooldown months = 6 }
    automation_tick = never
    ai_tick = monthly
    ai_tick_frequency = 1
    ai_prerequisite = { oe_ra_actor = yes political_influence >= 25 gold > oe_ra_annual_income }
    select_trigger = {
        looking_for_a = situation
        interaction_source_list = { situation:oe_russian_ambition = { add_to_list = source } }
        target_flag = recipient
        name = choose_situation
        column = { data = name }
        visible = { this = situation:oe_russian_ambition situation_is_active = yes }
    }
    select_trigger = {
        looking_for_a = province
        interaction_source_list = {
            region:steppes_region = { every_area_in_region = { every_province_in_area = { add_to_list = source } } }
            region:khorasan_region = { every_area_in_region = { every_province_in_area = { add_to_list = source } } }
        }
        target_flag = target
        name = choose_province
        none_available_msg_key = no_provinces_available
        column = { data = owner_flag }
        column = { data = name }
        column = { data = population }
        only_color_selectable = yes
        map_color = { value = owner.country_color }
        visible = { oe_ra_steppe_province = yes }
    }
    effect = {
        if = {
            limit = { exists = scope:target scope:target = { oe_ra_steppe_province = yes } }
            scope:actor = {
                add_political_influence = -25
                add_gold = { value = oe_ra_steppe_cost multiply = -1 }
                add_casus_belli = { target = scope:target.owner type = casus_belli:cb_conquer_province province = scope:target }
            }
        }
    }
    ai_will_do = { add = 80 }
}

    }
    effect = {
        if = {
            limit = { scope:actor = { oe_ra_saint_petersburg_ready = yes } }
            scope:actor = {
                add_gold = { value = oe_ra_annual_income multiply = -1 }

                set_variable = oe_ra_saint_petersburg_founded

            location:neva = {
                # Native helper upgrades rural/town ranks and preserves existing metropolises.

                set_variable = oe_ra_saint_petersburg_unlocked
                if = {
                    limit = { NOT = { has_building = building_type:oe_ra_alexander_column } }
                    construct_building = { building_type = building_type:oe_ra_alexander_column owner = scope:actor cost_multiplier = 0 cost_multiplier_reason = oe_ra_found_saint_petersburg }
                }
                if = {
                    limit = { NOT = { has_building = building_type:oe_ra_nevsky_prospect } }
                    construct_building = { building_type = building_type:oe_ra_nevsky_prospect owner = scope:actor cost_multiplier = 0 cost_multiplier_reason = oe_ra_found_saint_petersburg }
                }
            }
            scope:actor = {
                if = { limit = { NOT = { capital = location:neva } } set_capital = location:neva }
            }
        }
    }
    ai_will_do = { add = 80 }
}
