		years = 5
	}
	automation_tick = never
	ai_tick = monthly
	ai_tick_frequency = 12
	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:oe_sit_manchu_pass = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:oe_sit_manchu_pass = this
			situation_is_active = yes
		}
	}
	effect = {
		scope:actor = {
			add_gold = -100
			add_prestige = -10
			add_manpower = 1.5
			add_army_tradition = 5
			add_country_modifier = { modifier = oe_pmch_mod_act_banner_vanguard years = 5 mode = replace }
		}

			if = { limit = { NOT = { has_global_variable = oe_pmch_pass_pressure } } set_global_variable = { name = oe_pmch_pass_pressure value = 30 } }
			change_global_variable = { name = oe_pmch_pass_pressure add = 8 }

		}
	}
	ai_will_do = {
		add = 20
		if = {
			limit = { scope:actor = { at_war = yes } }
			add = 25
		}
	}
}
