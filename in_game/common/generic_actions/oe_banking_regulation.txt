oe_ir_banking = {
    type = internationalorganization
    show_message = no
    potential = { scope:actor = { tag = BYZ } }
    allow = {
        scope:actor = {
            tag = BYZ
            custom_tooltip = { text = OE_IR_NOT_DONE NOT = { has_variable = oe_ir_banking_done } }
            has_variable = oe_ir_central_done
            has_embraced_institution = institution:banking
        }
        international_organization:oe_byz_empire_pillars = {
            custom_tooltip = { text = OE_IR_AUTHORITY_50 var:oe_pillars_authority >= 50 }
        }
    }
    automation_tick = never
    ai_tick = never
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars scope:actor = { tag = BYZ } }
    }
    effect = {
        if = {
            limit = {
                scope:actor = {
                    tag = BYZ
                    NOT = { has_variable = oe_ir_banking_done }
                    has_variable = oe_ir_central_done
                    has_embraced_institution = institution:banking
                }
                scope:recipient = {
                    this = international_organization:oe_byz_empire_pillars
                    var:oe_pillars_authority >= 50
                }
            }
            scope:recipient = { change_variable = { name = oe_pillars_authority add = -50 } }
            scope:actor = {
                set_variable = oe_ir_banking_done
                add_country_modifier = { modifier = oe_ir_banking_modifier years = -1 mode = replace }
            }
            custom_tooltip = OE_BANKING_UNLOCK_FOUNDING
        }
    }
}
