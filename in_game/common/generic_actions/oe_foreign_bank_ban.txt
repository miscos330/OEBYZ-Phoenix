oe_pillars_ban_foreign_banks = {
    type = internationalorganization
    show_message = no
    potential = {
    	exists = international_organization:oe_byz_empire_pillars
        scope:actor = { tag = BYZ has_variable = oe_ir_banking_done }
    }
    allow = {
        scope:actor = {
            tag = BYZ
            has_variable = oe_ir_banking_done
            custom_tooltip = {
                text = OE_FOREIGN_BANK_BAN_NOT_DONE
                NOT = { has_variable = oe_foreign_banks_banned }
            }
        }
        international_organization:oe_byz_empire_pillars = {
            custom_tooltip = { text = OE_FOREIGN_BANK_BAN_AUTHORITY var:oe_pillars_authority >= 10 }
        }
    }
    automation_tick = never
    ai_tick = never
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = {
            this = international_organization:oe_byz_empire_pillars
            scope:actor = { tag = BYZ }
        }
    }
    effect = {
        if = {
            limit = {
                scope:actor = {
                    tag = BYZ
                    has_variable = oe_ir_banking_done
                    NOT = { has_variable = oe_foreign_banks_banned }
                }
                scope:recipient = {
                    this = international_organization:oe_byz_empire_pillars
                    var:oe_pillars_authority >= 10
                }
            }
            scope:recipient = { change_variable = { name = oe_pillars_authority add = -10 } }
            scope:actor = { set_variable = oe_foreign_banks_banned }
            custom_tooltip = OE_FOREIGN_BANK_BAN_EFFECT
            hidden_effect = {
                every_country = {
                    limit = { OR = { this = c:BYZ is_subject_or_below_of = c:BYZ } }
                    every_owned_location = {
                        every_foreign_buildings_in_location = {
                            limit = {
                                OR = {
                                    building_type = building_type:banking_office
                                    building_type = building_type:oe_byz_bank
                                }
                                exists = owner
                                owner = { NOR = { this = c:BYZ is_subject_or_below_of = c:BYZ } }
                            }
                            location = { destroy_building = prev }
                        }
                    }
                }
            }
        }
    }
}
