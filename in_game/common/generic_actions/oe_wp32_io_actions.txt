# --- Armamenton IO Interactive Actions (3) ---

oe_wp32_act_amt_equip_cataphracts = {
	type = internationalorganization
	show_in_gui_list = yes
	show_message = no
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12

	potential = { exists = international_organization:oe_wp32_armamenton_io }
	allow = {
		scope:actor = {
			oe_wp32_is_byz_host = yes
			gold >= 80
		}
	}
	cooldown = { type = oe_wp32_act_amt_equip_cd years = 8 }

	select_trigger = {
		looking_for_a = international_organization
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			this = international_organization:oe_wp32_armamenton_io
			scope:actor ?= { is_leader_of_international_organization = root }
		}
	}

	effect = {
		scope:actor = {
			add_gold = -80
			add_army_tradition = 4
			add_country_modifier = { modifier = oe_wp32_mod_amt_cataphract_issue years = 10 mode = replace }
		}
		international_organization:oe_wp32_armamenton_io ?= {
			if = { limit = { has_variable = oe_wp32_amt_techne } change_variable = { name = oe_wp32_amt_techne add = 8 } }
			else = { set_variable = { name = oe_wp32_amt_techne value = 38 } }
		}
	}

	ai_will_do = {
		value = 0
		if = { limit = { scope:actor = { gold >= 200 at_war = yes } } add = 30 }
	}
}
