# --- Four Courts Crisis: 4 Interactive Mitigation Actions ---

oe_wp32_act_fc_palace_audit = {
	type = disaster
	show_message = no
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12
	allow = {
		scope:actor = {
			gold >= 100
			NOT = { has_variable = oe_wp32_fc_palace_done }
		}
	}
	cooldown = { type = oe_wp32_fc_palace_cd years = 3 }
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:oe_wp32_four_courts_crisis
			disaster_is_active = yes
		}
	}
	effect = {
		scope:actor = {
			set_variable = { name = oe_wp32_fc_palace_done value = 1 }
			add_gold = -100
			add_stability = 6
			add_legitimacy = 5
			add_country_modifier = { modifier = oe_wp32_mod_fc_palace_reformed years = 10 mode = replace }
			custom_tooltip = oe_wp32_act_fc_palace_audit_tt
			hidden_effect = {
				if = { limit = { has_variable = oe_wp32_fc_steps } change_variable = { name = oe_wp32_fc_steps add = 1 } }
				else = { set_variable = { name = oe_wp32_fc_steps value = 1 } }
			}
		}
	}
	ai_will_do = { add = 25 }
}
