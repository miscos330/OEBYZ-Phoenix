# Explicit buttons in Empire Pillars > Edicts; no diplomacy menu registration.
oe_exchange_territory = {
    type = owncountry
    ai_tick = never
    automation_tick = never
    show_in_gui_list = no
    show_message = no
    show_message_to_target = no
    potential = { scope:actor = { oe_tx_actor_unlocked = yes } }
    allow = {
        scope:actor = {
            custom_tooltip = { text = OE_TX_COOLDOWN_READY NOT = { has_cooldown = oe_territory_exchange_cooldown } }
            num_of_diplomats >= 1
        }
    }
    select_trigger = {
        looking_for_a = country
        source = world
        target_flag = tx_partner
        name = OE_TX_CHOOSE_PARTNER
        none_available_msg_key = OE_TX_NO_PARTNERS
        column = { data = name }
        visible = {
            oe_tx_partner_allowed = yes
            trigger_if = {
                limit = { c:BYZ = { oe_tx_session = yes } }
                this = c:BYZ.var:oe_tx_partner
            }
        }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_tx_actor_unlocked = yes num_of_diplomats >= 1 NOT = { has_cooldown = oe_territory_exchange_cooldown } }
                exists = scope:tx_partner
                scope:tx_partner = { oe_tx_partner_allowed = yes }
            }
            scope:actor = {
                if = {
                    limit = { NOT = { oe_tx_session = yes } }
                    oe_tx_clear = yes
                    set_variable = { name = oe_tx_partner value = scope:tx_partner }
                    oe_tx_refresh = yes
                }
                custom_tooltip = OE_TX_DRAFT_OPENED
            }
        }
    }
}

oe_tx_add_give = {
    type = owncountry
    ai_tick = never
    automation_tick = never
    show_in_gui_list = no
    show_message = no
    show_message_to_target = no
    potential = { scope:actor = { oe_tx_session = yes } }
    allow = { scope:actor = { var:oe_tx_give_count < 50 } }
    select_trigger = {
        looking_for_a = country
        source = world
        target_flag = tx_partner
        name = choose_country
        column = { data = name }
        visible = { this = c:BYZ.var:oe_tx_partner oe_tx_partner_allowed = yes }
    }
    select_trigger = {
        looking_for_a = location
        target_flag = target
        name = OE_TX_CHOOSE_GIVE
        none_available_msg_key = OE_TX_NO_GIVE_LOCATIONS
        interaction_source_list = {
            scope:actor = {
                every_owned_location = { add_to_list = source }
                every_subject = {
                    every_owned_location = { add_to_list = source }
                    every_subject = { every_owned_location = { add_to_list = source } }
                }
            }
        }
        column = { data = name }
        column = { data = population }
        only_color_selectable = yes
        map_color = define:NMapColors|MAP_COLOR_HIGH
        visible = { oe_tx_give_candidate = yes }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_tx_session = yes }
                exists = scope:tx_partner
                scope:tx_partner = { this = c:BYZ.var:oe_tx_partner }
            }
            scope:actor = { oe_tx_add_give_effect = yes }
        }
    }
}
