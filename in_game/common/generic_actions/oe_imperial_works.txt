oe_works_thrace_food = {
    type = internationalorganization
    show_message = no
    potential = { exists = international_organization:oe_byz_roman_world scope:actor = { oe_works_actor = yes  } }
    allow = { scope:actor = { oe_works_thrace_food_allowed = yes } }
    automation_tick = never
    ai_tick = never
    
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_roman_world }
    }
    select_trigger = {
        looking_for_a = location
        source = actor
        target_flag = target_location
        name = "oe_works_thrace_food_choose"
        none_available_msg_key = "oe_works_thrace_food_none"
        column = { data = name }
        column = { data = raw_material }
        column = { data = population }
        interaction_source_list = {
            area:thrace_area = {
                every_location_in_area = { limit = { oe_works_thrace_food_target = yes } add_to_list = source }
            }
        }
        only_color_selectable = yes
        map_color = define:NMapColors|MAP_COLOR_HIGH
        visible = { oe_works_thrace_food_target = yes }
        enabled = { oe_works_thrace_food_target = yes }
    }
    effect = {
        if = {
            limit = {
                scope:recipient = { this = international_organization:oe_byz_roman_world }
                scope:actor = { oe_works_thrace_food_allowed = yes }
                scope:target_location = { oe_works_thrace_food_target = yes }
            }
            scope:actor = {
                add_gold = -1000
                if = {
                    limit = { NOT = { has_variable = oe_works_thrace_food_projects } }
                    set_variable = { name = oe_works_thrace_food_projects value = 0 }
                }
                change_variable = { name = oe_works_thrace_food_projects add = 1 }
            }
            scope:target_location = {
                add_location_modifier = { modifier = oe_supply_the_capital years = -1 mode = replace }
            }
        }
    }
}
