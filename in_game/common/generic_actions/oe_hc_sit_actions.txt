# 1. BYZ / ROM & Imperial Bloc: Imperial Offensive Supply (+6 active campaign progress, 3y cd)
oe_hc_act_imperial_supply = {
	type = situation
	show_message = no
	potential = {
		scope:actor = {
			OR = {
				tag = BYZ
				tag = ROM
				oe_heg_byz_bloc = yes
			}
		}
	}
	allow = {
		scope:actor = {
			gold >= 150
			manpower >= 0.5
		}
	}
	cooldown = {
		type = oe_hc_act_imperial_supply_cd
		years = 3
	}
	automation_tick = never
	ai_tick = monthly
	ai_tick_frequency = 12
	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:oe_hc_egypt = { add_to_list = source }
			situation:oe_hc_africa = { add_to_list = source }
			situation:oe_hc_italy = { add_to_list = source }

			situation:oe_hc_iberia = { add_to_list = source }
			situation:oe_hc_britain = { add_to_list = source }
			situation:oe_hc_persia = { add_to_list = source }
			situation:oe_hc_blacksea = { add_to_list = source }
			situation:oe_hc_arabia = { add_to_list = source }
		}
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation_is_active = yes
			OR = {
				this = situation:oe_hc_egypt
				this = situation:oe_hc_africa
				this = situation:oe_hc_italy
				this = situation:oe_hc_france
				this = situation:oe_hc_iberia
				this = situation:oe_hc_britain
				this = situation:oe_hc_persia
				this = situation:oe_hc_blacksea
				this = situation:oe_hc_arabia
			}
		}
	}
	effect = {
		scope:actor = {
			add_gold = -150
			add_army_tradition = 5
			add_country_modifier = { modifier = oe_hc_mod_act_imperial_supply years = 4 mode = replace }
		}
		custom_tooltip = oe_hc_act_imperial_supply_tt
		hidden_effect = {
			if = {
				limit = { country_exists = c:BYZ }
				c:BYZ = {
				if = { limit = { has_variable = oe_hc_egypt_active has_variable = oe_hc_egypt_progress } change_variable = { name = oe_hc_egypt_progress add = 6 } clamp_variable = { name = oe_hc_egypt_progress max = 100 min = 0 } }
				if = { limit = { has_variable = oe_hc_africa_active has_variable = oe_hc_africa_progress } change_variable = { name = oe_hc_africa_progress add = 6 } clamp_variable = { name = oe_hc_africa_progress max = 100 min = 0 } }
				if = { limit = { has_variable = oe_hc_italy_active has_variable = oe_hc_italy_progress } change_variable = { name = oe_hc_italy_progress add = 6 } clamp_variable = { name = oe_hc_italy_progress max = 100 min = 0 } }
				if = { limit = { has_variable = oe_hc_france_active has_variable = oe_hc_france_progress } change_variable = { name = oe_hc_france_progress add = 6 } clamp_variable = { name = oe_hc_france_progress max = 100 min = 0 } }
				if = { limit = { has_variable = oe_hc_iberia_active has_variable = oe_hc_iberia_progress } change_variable = { name = oe_hc_iberia_progress add = 6 } clamp_variable = { name = oe_hc_iberia_progress max = 100 min = 0 } }
				if = { limit = { has_variable = oe_hc_britain_active has_variable = oe_hc_britain_progress } change_variable = { name = oe_hc_britain_progress add = 6 } clamp_variable = { name = oe_hc_britain_progress max = 100 min = 0 } }
