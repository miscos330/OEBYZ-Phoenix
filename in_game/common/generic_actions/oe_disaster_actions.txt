oe_dis_act_overext_austerity = {
	type = disaster
	show_message = no
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never
	automation_tick_frequency = 12
	allow = {
		scope:actor = {
			gold >= 100
			prestige >= 5
		}
	}
	cooldown = {
		type = oe_dis_act_overext_austerity_cd
		years = 4
	}
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:oe_dis2_imperial_overextension
			disaster_is_active = yes
		}
	}
	effect = {
		scope:actor = {
			add_gold = -100
			add_prestige = -5
			add_stability = 4
			add_inflation = -0.015
			add_country_modifier = { modifier = oe_dis_mod_overext_austerity years = 5 mode = replace }
			custom_tooltip = oe_dis_act_overext_austerity_tt
			hidden_effect = {
				if = { limit = { has_variable = oe_dis2_overext_steps } change_variable = { name = oe_dis2_overext_steps add = 1 } }
				else = { set_variable = { name = oe_dis2_overext_steps value = 1 } }
			}
		}
	}
	ai_will_do = {
		add = 20
	}
}
