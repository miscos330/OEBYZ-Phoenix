# v26.0 WP11 Bug (b) fix: removed the absurd oe_ir_italian_theme_done hard prerequisite on Eastern/Austrian/Swiss theme expansion!
# Now independently available to BYZ/ROM and synced with the new 4-Prefecture Expansion Charters.
oe_create_eastern_themes = {
    type = internationalorganization
    show_message = no
    potential = { exists = international_organization:oe_byz_empire_pillars scope:actor = { oe_is_phoenix_empire = yes NOT = { has_variable = oe_create_eastern_themes_done } } }
    allow = {
        scope:actor = {
            oe_is_phoenix_empire = yes
            custom_tooltip = { text = OE_THEME_EXPANSION_NOT_DONE NOT = { has_variable = oe_create_eastern_themes_done } }
        }
        international_organization:oe_byz_empire_pillars ?= {
            custom_tooltip = { text = OE_IR_AUTHORITY_20 var:oe_pillars_authority >= 20 }
        }
    }
    automation_tick = never
    ai_tick = never
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        none_available_msg_key = "leave_international_organization_none"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars scope:actor = { oe_is_phoenix_empire = yes } }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_is_phoenix_empire = yes NOT = { has_variable = oe_create_eastern_themes_done } }
                scope:recipient = { this = international_organization:oe_byz_empire_pillars var:oe_pillars_authority >= 20 }
            }
            scope:recipient = { change_variable = { name = oe_pillars_authority add = -20 } }
            scope:actor = { custom_tooltip = { text = oe_create_eastern_themes_tt set_variable = oe_create_eastern_themes_done set_variable = oe_theme_pref_1_charter } }
        }
    }
}
