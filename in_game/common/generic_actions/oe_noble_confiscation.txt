oe_nf_integrate_1 = {
    type = internationalorganization
    show_in_gui_list = no
    show_message = no
    ai_tick = never
    automation_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { scope:actor = { oe_court_actor = yes } international_organization:oe_byz_empire_pillars = { oe_nf_integration_ready = { slot = 1 } } }
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    select_trigger = {
        # Explicit 1.4 target overlay; nonselectable land remains unhighlighted.
        only_color_selectable = yes
        map_color = define:NMapColors|MAP_COLOR_HIGH
        looking_for_a = province
        source = actor
        target_flag = target_province
        name = "choose_province"
        column = { data = name }
        visible = { oe_nf_integration_target = { slot = 1 } }
    }
    effect = {
        if = {
            limit = { scope:actor = { oe_court_actor = yes } scope:recipient = { this = international_organization:oe_byz_empire_pillars } }
            international_organization:oe_byz_empire_pillars = { oe_nf_assist_integration = { slot = 1 } }
        }
    }
}

oe_nf_confiscate_house = {
    type = internationalorganization
    show_in_gui_list = no
    show_message = no
    ai_tick = never
    automation_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { scope:actor = { oe_court_actor = yes }  }
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    select_trigger = {
        looking_for_a = character
        source = actor
        target_flag = target_character
        name = "OE_NF_CHOOSE_CONFISCATE"
        column = { data = character_info }
        interaction_source_list = { international_organization:oe_byz_empire_pillars = {
            if = { limit = { has_variable = oe_nf_dynasty_1 exists = var:oe_nf_head_1 } var:oe_nf_head_1 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_2 exists = var:oe_nf_head_2 } var:oe_nf_head_2 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_3 exists = var:oe_nf_head_3 } var:oe_nf_head_3 = { add_to_list = source } }
            if = { limit = { has_variable = oe_nf_dynasty_4 exists = var:oe_nf_head_4 } var:oe_nf_head_4 = { add_to_list = source } }
        } }
        visible = { oe_nf_living = yes }
        enabled = { oe_nf_confiscation_head = yes }
    }
    effect = {
        if = {
            limit = { scope:actor = { oe_court_actor = yes } scope:recipient = { this = international_organization:oe_byz_empire_pillars } }
            international_organization:oe_byz_empire_pillars = { 
                if = {
                    limit = { has_variable = oe_nf_dynasty_1 var:oe_nf_head_1 = scope:target_character scope:target_character = { oe_nf_confiscation_head = yes } }
                    oe_nf_confiscate = { slot = 1 }
                }
                if = {
                    limit = { has_variable = oe_nf_dynasty_2 var:oe_nf_head_2 = scope:target_character scope:target_character = { oe_nf_confiscation_head = yes } }
                    oe_nf_confiscate = { slot = 2 }
                }
                if = {
                    limit = { has_variable = oe_nf_dynasty_3 var:oe_nf_head_3 = scope:target_character scope:target_character = { oe_nf_confiscation_head = yes } }
                    oe_nf_confiscate = { slot = 3 }
                }
                if = {
                    limit = { has_variable = oe_nf_dynasty_4 var:oe_nf_head_4 = scope:target_character scope:target_character = { oe_nf_confiscation_head = yes } }
                    oe_nf_confiscate = { slot = 4 }
                } }
        }
    }
}
