oe_court_chancellor_appoint = {
    type = internationalorganization
    show_message = no
    ai_tick = never
    automation_tick = never
    potential = { scope:actor = { oe_court_actor = yes oe_court_offices_active = yes } }
    allow = { scope:actor = { oe_court_actor = yes oe_court_offices_active = yes NOT = { oe_court_chancellor_filled = yes } } }
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    select_trigger = {
        looking_for_a = character
        source = actor
        interaction_source_list = {
            scope:actor = {
                every_character = { limit = { oe_court_candidate = yes } add_to_list = source }
                # PAUSED 2026-09-26: noble countries replaced by domestic dynasties.
# every_subject_or_below = {
#                     limit = { is_subject_type = oe_noble }
#                     every_character = { limit = { oe_court_candidate = yes } add_to_list = source }
#                 }

            }
        }
        target_flag = target_character
        name = "OE_COURT_SELECT"
        none_available_msg_key = "OE_COURT_NONE"
        column = { data = character_info }
        visible = { oe_court_candidate = yes }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_court_actor = yes oe_court_offices_active = yes NOT = { oe_court_chancellor_filled = yes } }
                scope:recipient = { this = international_organization:oe_byz_empire_pillars }
                scope:target_character = { oe_court_candidate = yes }
            }
            scope:actor = {
                oe_court_clear_invalid = yes
                set_variable = { name = oe_court_chancellor value = scope:target_character }
            }
            scope:target_character = { set_variable = oe_court_office_held }
        }
    }
}
