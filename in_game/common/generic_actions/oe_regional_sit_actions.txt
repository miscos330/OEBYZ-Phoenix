# 1. oe_russian_ambition — 3 Rival / Observer Actions (all non-RUS visible powers)
# =============================================================================
oe_ra_act_frontier_bulwark = {
	type = situation
	show_message = no
	potential = {
		scope:actor = {
			is_rebel_country = no
			NOT = { tag = RUS }
		}
	}
	allow = {
		scope:actor = {
			gold >= 80
		}
	}
	cooldown = {
		type = oe_ra_act_frontier_bulwark_cd
		years = 5
	}
	automation_tick = never
	ai_tick = monthly
	ai_tick_frequency = 12
	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:oe_russian_ambition = { add_to_list = source }
		}
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			this = situation:oe_russian_ambition
			situation_is_active = yes
		}
	}
	effect = {
		scope:actor = {
			add_gold = -80
			add_manpower = 1.2
			add_country_modifier = { modifier = oe_reg_mod_ra_frontier_bulwark years = 5 mode = replace }
		}
	}
	ai_will_do = {
		add = 14
		if = {
			limit = { scope:actor = { country_exists = c:RUS is_neighbor_of = c:RUS } }
			add = 18
		}
	}
}
