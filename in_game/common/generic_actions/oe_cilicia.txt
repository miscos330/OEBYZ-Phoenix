oe_guarantee_cilicia = {
    type = internationalorganization
    show_message = no
    automation_tick = never
    ai_tick = never
    price = price:oe_cilicia_action_price
    potential = { scope:actor = { tag = BYZ current_date >= 1337.1.1 current_date < 1375.1.1 NOT = { has_variable = oe_cilicia_guaranteed } } }
    allow = { scope:actor = { oe_cilicia_guarantee_allowed = yes } }
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_roman_world }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_cilicia_guarantee_allowed = yes }
                scope:recipient = { this = international_organization:oe_byz_roman_world }
            }
            create_relation = { first = scope:actor second = c:CIL type = relation_type:guarantee }
            scope:actor = { set_variable = oe_cilicia_guaranteed }
        }
    }
}
