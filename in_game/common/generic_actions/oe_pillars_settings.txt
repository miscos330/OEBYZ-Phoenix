oe_pillars_settings = {
    type = internationalorganization
    show_message = no
    automation_tick = never
    ai_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { scope:actor = { oe_court_actor = yes } }
    select_trigger = {
        looking_for_a = international_organization source = actor target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    effect = {
        if = {
            limit = { scope:actor = { oe_court_actor = yes } scope:recipient = { this = international_organization:oe_byz_empire_pillars } }
            scope:actor = { trigger_event_non_silently = oe_pillars_settings.1 }
        }
    }
}

# Settings-only, player initiated; no periodic AI or automation enumeration.
oe_settings_market_reduce = {
    type = owncountry
    show_in_gui_list = no
    show_message = no
    automation_tick = never
    ai_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { scope:actor = { oe_court_actor = yes } }
    select_trigger = {
        looking_for_a = location
        target_flag = target
        interaction_source_list = {
            every_market_in_world = {
                location = { add_to_list = source }
            }
        }
        name = oe_settings_market_reduce_select
        none_available_msg_key = oe_settings_market_reduce_none
        column = { data = name }
        visible = { is_market_center = yes }
        enabled = { is_market_center = yes }
        map_color = define:NMapColors|MAP_COLOR_HIGH
        only_color_selectable = yes
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_court_actor = yes }
                scope:target = { is_market_center = yes }
            }
            scope:target = {
                add_location_modifier = {
                    modifier = oe_settings_market_attraction_reduction
                    years = -1
                    size = 1
                    mode = add
                }
            }
        }
    }
}

# Settings-only, player initiated; no periodic AI or automation enumeration.
oe_settings_market_clear = {
    type = owncountry
    show_in_gui_list = no
    show_message = no
    automation_tick = never
    ai_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { scope:actor = { oe_court_actor = yes } }
    select_trigger = {
        looking_for_a = location
        target_flag = target
        # Populate actual location scopes explicitly when this player-only picker opens.
        interaction_source_list = {
            every_location_in_the_world = {
                limit = { has_location_modifier = oe_settings_market_attraction_reduction }
                add_to_list = source
            }
        }
        cache_interaction_source_list = yes
        name = oe_settings_market_clear_select
        none_available_msg_key = oe_settings_market_clear_none
        column = { data = name }
        visible = { has_location_modifier = oe_settings_market_attraction_reduction }
        enabled = { has_location_modifier = oe_settings_market_attraction_reduction }
        map_color = define:NMapColors|MAP_COLOR_HIGH
        only_color_selectable = yes
    }
    effect = {
        if = {
            limit = {
                scope:actor = { oe_court_actor = yes }
                scope:target = { has_location_modifier = oe_settings_market_attraction_reduction }
            }
            scope:target = {
                remove_location_modifier = oe_settings_market_attraction_reduction
            }
        }
    }
}
