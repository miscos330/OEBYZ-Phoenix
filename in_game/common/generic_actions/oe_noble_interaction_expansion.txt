oe_nf_redeem_1 = {
    type = internationalorganization
    show_in_gui_list = no
    show_message = no
    ai_tick = never
    automation_tick = never
    potential = { scope:actor = { oe_court_actor = yes } }
    allow = { international_organization:oe_byz_empire_pillars = { oe_nf_redeem_ready = { slot = 1 } } }
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    select_trigger = {
        # Explicit 1.4 target overlay; nonselectable land remains unhighlighted.
        only_color_selectable = yes
        map_color = define:NMapColors|MAP_COLOR_HIGH
        looking_for_a = location
        source = actor
        target_flag = target_location
        name = "OE_NF_CHOOSE_HOLDING"
        column = { data = name }
        interaction_source_list = { international_organization:oe_byz_empire_pillars = { every_in_list = { variable = oe_nf_locations_1 add_to_list = source } } }
        visible = { oe_nf_controls_slot = { slot = 1 } }
    }
    effect = {
        if = {
            limit = { scope:actor = { oe_court_actor = yes } scope:recipient = { this = international_organization:oe_byz_empire_pillars } }
            international_organization:oe_byz_empire_pillars = { oe_nf_redeem = { slot = 1 } }
        }
    }
}

        } }
        visible = { oe_nf_mediation_head = yes }
    }
    select_trigger = {
        looking_for_a = character
        source = actor
        target_flag = second_head
        name = "OE_NF_CHOOSE_SECOND_HEAD"
        column = { data = character_info }
        interaction_source_list = { international_organization:oe_byz_empire_pillars = {
            if = { limit = { has_variable = oe_nf_dynasty_1 exists = var:oe_nf_head_1 } var:oe_nf_head_1 = { add_to_list = source } }

            if = { limit = { has_variable = oe_nf_dynasty_4 exists = var:oe_nf_head_4 } var:oe_nf_head_4 = { add_to_list = source } }
        } }
        visible = { oe_nf_mediation_head = yes this != scope:first_head }
    }
    effect = {
        if = {
            limit = { scope:actor = { oe_court_actor = yes } scope:recipient = { this = international_organization:oe_byz_empire_pillars } }
