oe_pillars_borrow_chartered_bank = {
    type = internationalorganization
    show_message = no
    potential = { scope:actor = { tag = BYZ } }
    allow = {
        scope:actor = {
            tag = BYZ
            any_subject = { oe_chartered_bank_loan_target = yes }
        }
    }
    automation_tick = never
    ai_tick = never
    select_trigger = {
        looking_for_a = international_organization
        source = actor target_flag = recipient
        name = "choose_international_organization"
        none_available_msg_key = "leave_international_organization_none"
        column = { data = name }
        visible = { this = international_organization:oe_byz_empire_pillars }
    }
    select_trigger = {
        # Explicit 1.4 target overlay; nonselectable land remains unhighlighted.
        only_color_selectable = yes
        map_color = define:NMapColors|MAP_COLOR_HIGH
        looking_for_a = country target_flag = target_bank
        name = "OE_CHARTERED_BANK_LOAN_SELECT"
        none_available_msg_key = "OE_CHARTERED_BANK_LOAN_NONE"
        column = { data = name }
        column = { data = gold }
        visible = { is_subject_type = oe_bank is_subject_of = c:BYZ }
        enabled = { oe_chartered_bank_loan_target = yes }
    }
    effect = {
        if = {
            limit = {
                scope:actor = { tag = BYZ }
                scope:recipient = { this = international_organization:oe_byz_empire_pillars }
                scope:target_bank = { oe_chartered_bank_loan_target = yes }
            }
            scope:target_bank = {
                # Snapshot before native give_loan debits the lender and credits the borrower.
                save_scope_value_as = { name = oe_chartered_bank_principal value = { value = gold multiply = 0.10 } }
                give_loan = {
                    target = scope:actor
                    amount = scope:oe_chartered_bank_principal
                    interest = 0.04
                    months = 60
                    set_loc_key = "oe_chartered_bank_treasury_loan"
                }
            }
        }
    }
}
