# 6 Empire-Wide Subject Directives (type = owncountry, available to ANY nation with >= 2 subjects across all eras).

oe_wp47_directive_imperial_taxation = {
	type = owncountry
	sound = UI_action_government_generic
	show_message = no

	potential = {
		scope:actor = {
			oe_wp47_is_eligible_overlord = yes
		}
	}

	allow = {
		scope:actor = {
			num_subjects >= 2
			custom_tooltip = {
				text = oe_wp47_directive_ready_tt
				NOT = { has_variable = oe_wp47_directive_cd }
			}
		}
	}

	cooldown = {
		type = oe_wp47_mass_directive_cd
		years = 5
	}

	automation_tick = never
	ai_tick = monthly
	ai_tick_frequency = 24
	ai_prerequisite = {
		num_subjects >= 2
		is_rebel_country = no
		at_war = no
		gold < 200
		NOT = { has_variable = oe_wp47_directive_cd }
	}

	effect = {
		custom_tooltip = oe_wp47_directive_imperial_taxation_tt
		scope:actor = {
			oe_wp47_apply_directive_imperial_taxation = yes
		}
	}

	ai_will_do = {
		add = 0
		if = {
			limit = {
				scope:actor = {
					at_war = no
					gold < 150
					NOT = { any_subject = { subject_loyalty < 55 } }
				}
			}
			add = 25
		}
	}
}
