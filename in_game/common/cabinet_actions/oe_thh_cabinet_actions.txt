# 1. Kulm Law Ostsiedlung Settlement Charter (《库尔姆法》东向拓垦宪章)
oe_thh_cab_kulm_settlement = {
	ability = adm
	icon = develop_province
	helps_with = "location"
	allow_multiple = no

	potential = {
		oe_thh_is_generalkapitel_eligible = yes
	}

	allow = {
		OR = {
			scope:is_active ?= yes
			num_locations >= 3
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "oe_thh_cab_kulm_settlement_select_province"
		none_available_msg_key = "oe_thh_cab_kulm_settlement_no_provinces"
		column = { data = name }
		column = { data = population }
		visible = {
			any_location_in_province = {
				has_owner = yes
				owner = scope:actor
			}
			NOT = {
				cabinet_already_performing_same_task_on_target = {
					cabinet_action = oe_thh_cab_kulm_settlement
					interaction_target = target
				}
			}
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}

	country_modifier = {
		global_pop_assimilation_speed_modifier = 0.10
		peasants_estate_target_satisfaction = 0.04
		tax_income_efficiency = -0.03
	}

	location_modifier = {
		local_pop_assimilation_speed_modifier = 0.25
		local_monthly_development = 0.002
		local_population_growth = 0.002
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		add = { value = 12 }
		scope:actor = {
			if = {
				limit = { at_war = no }
				add = { value = 10 }
			}
		}
	}
}
