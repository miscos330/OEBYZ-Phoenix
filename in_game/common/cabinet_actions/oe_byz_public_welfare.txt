oe_byz_promote_births = {
    ability = adm
    allow_multiple = yes
    potential = { oe_byz_logothetes_services_available = yes }
    
    select_trigger = {
        looking_for_a = province
        source = actor
        target_flag = target
        name = oe_byz_welfare_choose_province
        none_available_msg_key = oe_byz_welfare_no_province
        column = { data = name }
        column = { data = population }
        visible = {
            any_location_in_province = { is_ownable = yes has_owner = yes owner = scope:actor }
            NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = oe_byz_promote_births interaction_target = target } }
        }
        # Assignment-only quota. Top-level allow is rechecked for running actions.
        enabled = {
            scope:actor = {
                custom_tooltip = {
                    text = OE_BYZ_WELFARE_MAX_THREE
                    NOT = {
                        any_cabinet = {
                            has_cabinet_action = yes
                            cabinet_action = cabinet_action:oe_byz_promote_births
                            count >= 3
                        }
                    }
                }
            }
        }
        map_color = define:NMapColors|MAP_COLOR_HIGH
        only_color_selectable = yes
    }
    is_finished = {
        OR = {
            NOT = { oe_byz_logothetes_services_available = yes }
            NOT = { scope:target = { any_location_in_province = { is_ownable = yes has_owner = yes owner = root } } }
        }
    }
    # Native ongoing modifier: automatically scaled by cabinet efficiency and removed on cancellation.
    # Like native reduced_paperwork: target coverage is handled by the cabinet system.
    location_modifier = {
        local_population_growth = 0.001
    }
    map_marker = { show_for_owner = yes }
}
