# Fixed transfers use the native ledger: debit the payer and credit the IO.
oe_pillars_nobles_contribution = {
    get_payer_list = {
        every_country_with_special_status_of_type = {
            type = oe_pillars_empire
            every_estate = {
                limit = { estate_type = estate_type:nobles_estate }
                add_to_local_variable_list = { name = payers target = this }
            }
        }
    }
    get_payee_list = { add_to_local_variable_list = { name = payees target = root } }
    price = oe_byz_bank_org_leader_pay_price
    uses_maintenance = no
    price_multiplier = { value = oe_pillars_nobles_monthly_income }
    proportion_for_payer = { value = 1 }
    proportion_for_payee = { value = 1 }
}

# Native synchronized maintenance setting is the budget control only.
# Zero native price prevents any charge to the country's treasury.
# The IO monthly effect makes the actual affordable debit and grants authority.
oe_pillars_authority_budget = {
    get_payer_list = {
        every_country_with_special_status_of_type = {
            type = oe_pillars_empire
            add_to_local_variable_list = { name = payers target = this }
        }
    }
    get_payee_list = { add_to_local_variable_list = { name = payees target = root } }
    price = oe_byz_bank_org_leader_pay_price
    price_multiplier = { value = 0 }
    uses_maintenance = yes
    min_slider_value = 0
    ai_maintenance_value = {
        value = 0
        if = {
            limit = {
                scope:recipient = { gold > 0 var:oe_pillars_authority < 90 oe_pillars_purse_monthly_income > 0 }
            }
            value = 0.5
        }
    }
    proportion_for_payer = { value = 1 }
    proportion_for_payee = { value = 1 }
}
