oe_byz_bank_org_big_bank_pay = {#监会向银行收费
	get_payer_list = {
		every_international_organization_member = {
			limit = { 
				NOT = {
					tag = BYZ
				}
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:oe_bank_org_big_bank
						international_organization = root
					}	
				}
			}

			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		every_country_with_special_status_of_type = {
			type = oe_bank_org_big_bank
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	uses_maintenance = yes
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|oe_byz_bank_org_big_bank_pay)" < 0.5 }
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|oe_byz_bank_org_big_bank_pay)"
				multiply = 2
			}
		}
		monthly_prestige = -0.05
		bank_interest = 0.1
	}
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|oe_byz_bank_org_big_bank_pay)" > 0.5 }
		scale = {
			value = "payment_maintenance(scope:recipient|oe_byz_bank_org_big_bank_pay)"
			subtract = 0.5
			multiply = 2
		}
		monthly_prestige = 0.05
		total_loan_capacity_modifier = 1
	}
	price = oe_byz_bank_org_leader_pay_price
	price_multiplier = {
		add = {
			desc = "BASE"
			every_international_organization_member = {
				limit = {
				NOT = {
					tag = BYZ
				}
				NOT = {
					has_special_status_in_international_organization = {
						type = special_status:oe_bank_org_big_bank
						international_organization = root
					}	
				}					
				}
				add = monthly_income_total
			}
			multiply = 0.2
			
		}
	}
	proportion_for_payer = {
		add = {
			desc = "BASE"
			value = monthly_income_total
		}
	}
	proportion_for_payee = {
		add = {
			desc = "BASE"
			value = 1
		}
	}
}
