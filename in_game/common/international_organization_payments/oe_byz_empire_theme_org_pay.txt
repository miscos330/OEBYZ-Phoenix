oe_byz_empire_theme_org_leader_pay_for_army = {#军饷
	get_payer_list = {
		every_international_organization_member = {
			limit = { 
					has_special_status_in_international_organization = {
						type = special_status:oe_pillars_empire
						international_organization = root
					}				
			}

			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
        every_country_with_special_status_of_type = {
            type = oe_pillars_theme
            add_to_local_variable_list = { name = payees target = this }
        }
        every_country_with_special_status_of_type = {
            type = oe_pillars_naval_theme
            add_to_local_variable_list = { name = payees target = this }
        }
        every_country_with_special_status_of_type = {
            type = oe_pillars_frontier_theme
            add_to_local_variable_list = { name = payees target = this }
        }
        every_country_with_special_status_of_type = {
            type = oe_pillars_latin_theme
            add_to_local_variable_list = { name = payees target = this }
        }
    }
	price = oe_byz_bank_org_leader_pay_price
	uses_maintenance = yes
	# r13 fix 1 (2026-11-19): AI Byzantium keeps this slider at 100%. The theme statuses scale loyalty 50 and morale 0.8
	# with (slider - 1), so the AI's saving slider at 0 cost every theme 50 loyalty (r13: 5 of 7 themes at 0% by 1344).
	ai_maintenance_value = { value = 1 }
	ai_maintenance_ignore_saving = yes
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|oe_byz_empire_theme_org_leader_pay_for_army)" < 0.5 }
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|oe_byz_empire_theme_org_leader_pay_for_army)"
				multiply = 2
			}
		}

		monthly_prestige = -0.05

	}
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|oe_byz_empire_theme_org_leader_pay_for_army)" > 0.5 }
		scale = {
			value = "payment_maintenance(scope:recipient|oe_byz_empire_theme_org_leader_pay_for_army)"
			subtract = 0.5
			multiply = 2
		}

		monthly_prestige = 0.05
	}
	price_multiplier = {
		add = {
			desc = "BASE"
			every_international_organization_member = {
				limit = {
					oe_theme_army_payment_trigger = yes
				}
				if = { limit = { oe_pillars_naval_theme_trigger = yes } add = navy_size }
                else = { add = army_size }
			}
			multiply = 1
		}
	}
	proportion_for_payer = {
		add = {
			desc = "BASE"
			value = 1
		}
	}
	proportion_for_payee = {
		add = {
			desc = "BASE"
			# fm_patches1 A2 (decision 281): shared out by what price_multiplier counts (ships for a naval theme)
			value = 0
			if = { limit = { oe_pillars_naval_theme_trigger = yes } add = navy_size }
			else = { add = army_size }
		}
	}
}
