    is_special = yes
    buildable = yes
    hide = no
    empty = no
    levy = no
    strength_damage_done = 0.10
    morale_damage_done = 0.33
    initiative = 3
    movement_speed = -0.25

    construction_demand = expensive_heavy_cavalry_construction
    maintenance_demand = expensive_heavy_cavalry_maintenance
    limit = {
        # Native limit is per type. Reserve room for all OTHER types already in service.
        # The engine accounts for this type's own recruitment queue.
        value = oe_hetaireia_cavalry_unit_limit
        subtract = oe_hetaireia_cavalry_total_count
        add = total_sub_unit_type_count:oe_rum_cataphracts_1
        min = 0
    }
    upgrades_to_only = oe_rum_cataphracts_2
    country_potential = {
        tag = BYZ
        NOT = { has_variable = oe_rum_pronoia_enacted }
    }
    location_trigger = {
        owner = {

        NOT = { has_variable = oe_rum_pronoia_enacted }

            NOT = { has_advance = professional_armies_advance }

        }
    }
    gfx_tags = { oe_rum_cataphract_tag heavy_tag spear_tag }
    location_potential = { has_owner = yes owner = {
