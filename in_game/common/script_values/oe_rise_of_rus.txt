# Engine manpower/population/army-strength units are thousands.
# Levy score uses native available land levies; regular_army_size is the
# native established regular-army strength (not hired mercenaries).
oe_rus_own_score = {
    value = country_tax_base
    add = country_total_army_levy_size
    add = { value = regular_army_size multiply = 10 }
    add = { value = manpower multiply = 0.5 }
    add = { value = total_population divide = 10 }
    every_owned_non_rural_location = {
        limit = { OR = { location_rank = location_rank:city location_rank = location_rank:megalopolis } }
        add = 10
    }
}
