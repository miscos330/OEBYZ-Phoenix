# Country-scope values: the unique IO is explicit, including in special-status modifiers.
oe_pillars_current_public_opinion = {
	value = 0
	if = {
		limit = { exists = international_organization:oe_byz_empire_pillars }
	    value = international_organization:oe_byz_empire_pillars.var:oe_pillars_public_opinion
	    min = 0
	    max = 100
	}
}
oe_pillars_opinion_discontent_scale = {
    value = 50
    subtract = oe_pillars_current_public_opinion
    divide = 50
}
# Desired unmodified stability cost: 1.5 - opinion / 100 (150%, 100%, 50%).
oe_pillars_opinion_stability_cost_factor = {
    value = 1.5
    subtract = { value = oe_pillars_current_public_opinion divide = 100 }
}
# EU5 1.3 uses cost efficiency, not the removed stability_cost modifier.
# Cost = base / (1 + efficiency). Native efficiency modifiers stack additively.
oe_pillars_opinion_stability_efficiency = {
    value = 1
    divide = oe_pillars_opinion_stability_cost_factor
    subtract = 1
}
