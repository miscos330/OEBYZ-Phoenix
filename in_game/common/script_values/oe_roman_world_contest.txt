# Native rise_of_the_ottomans formula; no prestige, population or territorial bonuses.
oe_roman_world_contest_score = {
    add = military_strength
    add = country_tax_base
}

# Explicit script values for the GUI tooltip, rather than passing engine triggers as value names.
oe_roman_world_contest_military = { value = military_strength }
oe_roman_world_contest_tax = { value = country_tax_base }

# Organization scope: a new campaign displays 0 rather than an unset variable.
oe_rw_renaissance_progress = {
    value = 0
    if = { limit = { has_variable = oe_rw_renaissance_score } add = var:oe_rw_renaissance_score }
}

# Country scope: one month of tax and trade income, never a negative payment.
oe_rw_one_month_income_cost = { value = monthly_income_trade_and_tax min = 0 }
