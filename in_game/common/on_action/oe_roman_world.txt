on_game_start = { on_actions = { oe_roman_world_reconcile } }
on_location_changed_owner = { on_actions = { oe_roman_world_constantinople_changed oe_roman_world_contest_changed } }
on_religion_changed = { on_actions = { oe_roman_world_reconcile } }
on_culture_changed = { on_actions = { oe_roman_world_reconcile } }
on_country_rank_change = { on_actions = { oe_roman_world_reconcile } }
oe_roman_world_constantinople_changed = {
    trigger = { current_year >= 1200 this = location:constantinople }
    effect = { international_organization:oe_byz_roman_world ?= { oe_roman_world_sync_effect = yes } }
}
oe_roman_world_reconcile = {
    trigger = { current_year >= 1200 }
    effect = { international_organization:oe_byz_roman_world ?= { oe_roman_world_sync_effect = yes } }
}

# Recalculate on territorial changes as well as on the daily pulse.
oe_roman_world_contest_changed = {
    trigger = { current_year >= 1200 }
    effect = { international_organization:oe_byz_roman_world ?= { oe_roman_world_recalculate_contest = yes } }
}
