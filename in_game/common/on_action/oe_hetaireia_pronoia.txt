# Catch units queued before an enactment/revocation, without any daily scan.
monthly_country_pulse = { on_actions = { oe_hetaireia_pronoia_reconcile } }
oe_hetaireia_pronoia_reconcile = {
    trigger = {
        tag = BYZ is_rebel_country = no
        OR = {
            AND = {
                has_variable = oe_rum_pronoia_enacted
                oe_hetaireia_cavalry_normal_count > 0
            }
            AND = {
                NOT = { has_variable = oe_rum_pronoia_enacted }
                oe_hetaireia_cavalry_pronoia_count > 0
            }
        }
    }
    effect = {
        if = {
            limit = { has_variable = oe_rum_pronoia_enacted }
            oe_hetaireia_pronoia_upgrade = yes
        }
        else = { oe_hetaireia_pronoia_downgrade = yes }
    }
}
