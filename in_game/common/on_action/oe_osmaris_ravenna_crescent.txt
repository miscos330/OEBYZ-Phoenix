on_war_declared = { on_actions = { oe_belisarius_check_crescent } }
on_join_war = { on_actions = { oe_belisarius_check_crescent } }
oe_belisarius_check_crescent = {
    trigger = {
        current_year >= 1200
        country_exists = c:BYZ
        c:BYZ = { has_variable = oe_belisarius_rising_presented NOT = { has_variable = oe_belisarius_crescent_presented } }
    }
    effect = { c:BYZ = { oe_belisarius_try_crescent = yes } }
}
monthly_country_pulse = { on_actions = { oe_belisarius_ravenna_crescent_monthly } }
oe_belisarius_ravenna_crescent_monthly = {
    trigger = { current_year >= 1200 tag = BYZ is_rebel_country = no has_variable = oe_belisarius_rising_presented }
    effect = { oe_belisarius_try_crescent = yes oe_belisarius_try_ravenna = yes }
}
