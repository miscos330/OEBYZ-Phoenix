on_game_start = { on_actions = { oe_character_progress_migration } }
oe_character_progress_migration = {
    trigger = { NOT = { has_global_variable = oe_character_progress_migrated_20261009 } }
    effect = {
        every_country = {
            every_character = {
                limit = { OR = {
                    has_trait = oe_character_war_score_trait_I
                    has_trait = oe_character_war_score_trait_II
                    has_trait = oe_character_war_score_trait_III
                    has_trait = oe_character_war_score_trait_IV
                    has_trait = oe_character_war_score_trait_V
                    has_trait = oe_character_cabinet_experience_trait_I
                    has_trait = oe_character_cabinet_experience_trait_II
                    has_trait = oe_character_cabinet_experience_trait_III
                    has_trait = oe_character_cabinet_experience_trait_IV
                    has_trait = oe_character_cabinet_experience_trait_V
                } }
                oe_migrate_character_progress = yes
            }
        }
        set_global_variable = oe_character_progress_migrated_20261009
    }
}
