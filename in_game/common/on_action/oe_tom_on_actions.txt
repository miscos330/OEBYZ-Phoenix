	trigger = {
		current_year >= 1200
		has_global_variable = oe_tom_crusade_crisis_active
		exists = scope:actor
		exists = scope:recipient
		scope:actor = {
			oe_tom_is_overlord_host = yes
			NOT = { has_variable = oe_teu_secularized_pru }
			NOT = { has_country_modifier = oe_tom_mod_papal_censure }
		}
		scope:recipient = {
			religion ?= religion:catholic
			OR = {
				tag = POL
				tag = LIT
				tag = PLC
			}
		}
	}
	effect = {
		scope:actor = {
			trigger_event_non_silently = oe_tom.16
		}
		scope:recipient = {
			add_country_modifier = { modifier = oe_tom_mod_pl_constance_defense years = 10 mode = replace }
		}
	}
}

# Tracks recent defeat by POL/LIT/PLC for Prussian Confederation Bund Tension (+6/yr for 10y).

			OR = {

				tag = LIT
				tag = PLC
			}
		}
	}
	effect = {
		scope:loser = {

		}
	}
}
