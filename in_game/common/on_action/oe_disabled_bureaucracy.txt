on_game_start = { on_actions = { oe_disabled_bureaucracy_start } }
yearly_country_pulse = { on_actions = { oe_disabled_bureaucracy_monthly } }
oe_disabled_bureaucracy_start = { effect = { every_country = { oe_remove_disabled_bureaucracies = yes } } }
oe_disabled_bureaucracy_monthly = {
    trigger = {
        current_year >= 1200
        tag = BYZ is_rebel_country = no
        NOT = { has_variable = oe_settings_bureaucracy_restored }
        any_current_bureaucracy = { count > 0 }
    }
    effect = { oe_remove_disabled_bureaucracies = yes }
}
