# Monthly indexed counters catch recruits queued before a privilege change.
# No daily checks or world-location loops.
monthly_country_pulse = { on_actions = { oe_roman_siegers_reconcile } }
oe_roman_siegers_reconcile = {
    trigger = {
        tag = BYZ is_rebel_country = no
        OR = {
            AND = { has_variable = oe_ir_artillery_done oe_roman_sieger_normal_count > 0 }
            AND = { NOT = { has_variable = oe_ir_artillery_done } oe_roman_sieger_imperial_count > 0 }
        }
    }
    effect = {
        if = {
            limit = { has_variable = oe_ir_artillery_done }
            oe_roman_siegers_convert = { FORM = imperial }
        }
        else = { oe_roman_siegers_convert = { FORM = normal } }
    }
}
