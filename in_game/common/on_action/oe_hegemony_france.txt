on_location_occupied = { on_actions = { oe_heg_france_occupation } }
on_siege_won = { on_actions = { oe_heg_france_occupation } }
oe_heg_france_occupation = {
    trigger = { current_year >= 1200 }
    effect = { scope:target = { oe_heg_france_capture = yes } }
}
# Eligibility checks live participants on every occupation, so separate/final peace disables it
# immediately. This monthly pass only removes inert country bookkeeping; it does not govern expiry.
monthly_country_pulse = { on_actions = { oe_heg_france_cleanup } }
oe_heg_france_cleanup = {
    trigger = { current_year >= 1200 is_rebel_country = no has_variable = oe_heg_france_campaign_target NOT = { oe_heg_france_target_active = yes } }
    effect = { remove_variable = oe_heg_france_campaign_target }
}
