# NEW 2026-11-22 v14.0 Stage Integrator: Cross-Wave Telemetry 2.0 & 1836 Capstone Honors Hooks (0 slots, 0 rules, 0 vanilla overrides).

monthly_country_pulse = {
	on_actions = { oe_bal2_telemetry_pulse }
}

yearly_country_pulse = {
	on_actions = { oe_bal2_1836_capstone_hook }
}

oe_bal2_telemetry_pulse = {
	trigger = {
		is_rebel_country = no
		oe_bal_rule_is_on = yes
		current_month = 1
		NOT = { global_var:oe_bal2_ran_y ?= current_year }
		OR = {
			AND = { tag = BYZ is_rebel_country = no }
			AND = { NOT = { country_exists = c:BYZ } tag = TUR is_rebel_country = no }
			AND = { NOT = { country_exists = c:BYZ } NOT = { country_exists = c:TUR } tag = FRA is_rebel_country = no }
			AND = { NOT = { country_exists = c:BYZ } NOT = { country_exists = c:TUR } NOT = { country_exists = c:FRA } tag = CHI is_rebel_country = no }
			AND = { NOT = { country_exists = c:BYZ } NOT = { country_exists = c:TUR } NOT = { country_exists = c:FRA } NOT = { country_exists = c:CHI } is_ai = no }
			AND = { NOT = { country_exists = c:BYZ } NOT = { country_exists = c:TUR } NOT = { country_exists = c:FRA } NOT = { country_exists = c:CHI } is_great_power = yes }
		}
	}
	effect = {
		set_global_variable = { name = oe_bal2_ran_y value = current_year }
		oe_bal2_emit_extended_snapshot = yes
	}
}

oe_bal2_1836_capstone_hook = {
	trigger = {
		is_rebel_country = no
		is_ai = no
		current_year >= 1835
		NOT = { has_variable = oe_bal2_capstone_done }
	}
	effect = {
		set_variable = { name = oe_bal2_capstone_done value = 1 }
		trigger_event_non_silently = oe_bal2.1
	}
}
