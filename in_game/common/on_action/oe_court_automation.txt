on_cabinet_assigned = { on_actions = { oe_court_auto_cabinet_changed } }
on_character_estate_change = { on_actions = { oe_court_auto_character_changed } }
on_character_moved_country = { on_actions = { oe_court_auto_character_changed } }
# Native root is the country, scope:target the new cabinet appointee.
oe_court_auto_cabinet_changed = {
    trigger = { current_year >= 1200 tag = BYZ is_rebel_country = no scope:target = { has_variable = oe_court_office_held } }
    effect = { oe_court_clear_invalid = yes oe_court_auto_fill = yes }
}
# Both native hooks have a character root. Country-move scope:owner is the previous country.
oe_court_auto_character_changed = {
    trigger = { current_year >= 1200 country_exists = c:BYZ has_variable = oe_court_office_held }
    effect = { c:BYZ = { oe_court_clear_invalid = yes oe_court_auto_fill = yes } }
}
