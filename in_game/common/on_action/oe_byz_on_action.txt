            if = { limit = { NOT = { ruler = character:byz_ioannes_synadenos } } set_new_ruler = character:byz_ioannes_synadenos }
        }

			add_manpower = 3
			# FIX 2026-11-17 (author decision): no food-access relation on founding (popup spam; themes cancelled it the next month).

		}

		oe_theme_regions_assign = yes
		c:BYT01 = { set_variable = { name = oe_theme_region_id value = 8 } set_variable = { name = oe_theme_distance value = 2 } }
		c:BYZ = {
			every_owned_location = {
				limit = { var:oe_theme_region ?= 8 }
				construct_building = {
					building_type = building_type:oe_byz_theme_system
					owner = c:BYT01
				cost_multiplier = 0
				cost_multiplier_reason = "game_concept_event"
				}
			}
		}
		location:thessaloniki = {

				count = 3

					type = oe_byz_theme_subject_soldier_I
					owner = c:BYT01
				}
			}
			while =  {
				count = 6
				create_sub_unit_with_owner = {

					owner = c:BYT01
				}
			}

oe_byz_cabinet_experience = {#内阁经验
	trigger = {
		tag = BYZ is_rebel_country = no
	}
	effect = {
		every_cabinet_character = {
			limit = {
				owner = c:BYZ			
			}
		if = {
			limit = {
					NOT = {
						has_variable = oe_byz_cabinet_experience_var
					}				
			}
			set_variable = {
				name = oe_byz_cabinet_experience_var
				value = 0
			}		
		} 	
			change_variable = {
				name = oe_byz_cabinet_experience_var
				add = 1
			}
		if = {
			limit = {
					var:oe_byz_cabinet_experience_var >= 1	
					var:oe_byz_cabinet_experience_var <= 12	
					NOT = { has_character_modifier = oe_character_cabinet_experience_modifier_I }		
			}
				add_character_modifier = { modifier = oe_character_cabinet_experience_modifier_I years = -1 mode = replace }			
		}
		if = {
			limit = {
					var:oe_byz_cabinet_experience_var > 12
					var:oe_byz_cabinet_experience_var <= 48	
					NOT = { has_character_modifier = oe_character_cabinet_experience_modifier_II }
			}
				add_character_modifier = { modifier = oe_character_cabinet_experience_modifier_II years = -1 mode = replace }	
				remove_character_modifier = oe_character_cabinet_experience_modifier_I		
		}
		if = {
			limit = {
					var:oe_byz_cabinet_experience_var > 48
					var:oe_byz_cabinet_experience_var <= 96	
					NOT = { has_character_modifier = oe_character_cabinet_experience_modifier_III }			
			}
				add_character_modifier = { modifier = oe_character_cabinet_experience_modifier_III years = -1 mode = replace }	
				remove_character_modifier = oe_character_cabinet_experience_modifier_II		
		}
		if = {
			limit = {
					var:oe_byz_cabinet_experience_var > 96
					var:oe_byz_cabinet_experience_var <= 192	
					NOT = { has_character_modifier = oe_character_cabinet_experience_modifier_IV }			
			}
				add_character_modifier = { modifier = oe_character_cabinet_experience_modifier_IV years = -1 mode = replace }
				remove_character_modifier = oe_character_cabinet_experience_modifier_III		
		}
		if = {
			limit = {
					var:oe_byz_cabinet_experience_var > 192	
					NOT = { has_character_modifier = oe_character_cabinet_experience_modifier_V }			
			}
				add_character_modifier = { modifier = oe_character_cabinet_experience_modifier_V years = -1 mode = replace }
				remove_character_modifier = oe_character_cabinet_experience_modifier_IV		
		}				
		}
	}
}

oe_byzt_theme_house = {#军区治所
	effect = {
		if = {
			limit = {
				scope:winner ?= c:BYZ
				NOT = { oe_theme_location_occupied_trigger = yes }
			}
			# NEW 2026-11-17 theme-rework: inside a historical region, join whichever theme (any type) holds the region.
			if = {
				limit = { has_variable = oe_theme_region }
				save_scope_as = oe_theme_new_location
				c:BYZ = {
					random_owned_location = {
						limit = {
							var:oe_theme_region ?= scope:oe_theme_new_location.var:oe_theme_region
							any_buildings_in_location = { oe_theme_building_trigger = yes owner ?= { oe_theme_member_trigger = yes } }
						}
						random_buildings_in_location = {
							limit = { oe_theme_building_trigger = yes owner ?= { oe_theme_member_trigger = yes } }
							owner = { save_scope_as = oe_theme_region_owner }
						}
					}
				}
				if = {
					limit = { exists = scope:oe_theme_region_owner }
					oe_theme_construct_post = { owner = scope:oe_theme_region_owner }
				}
			}
			# Outside the historical regions: original area rule (ordinary themes), now guarded against a missing seat.
			else = {
		root.area = {
			every_location_in_area = {
				limit = { has_building = building_type:oe_byz_theme_system_noble }
				every_buildings_in_location = {
					limit = { building_type = building_type:oe_byz_theme_system_noble }
					save_scope_as = target_building
				}
			}
		}
				if = {
					limit = { exists = scope:target_building }
						construct_building = {
					building_type = building_type:oe_byz_theme_system
					owner = scope:target_building.owner
					cost_multiplier = 0
					cost_multiplier_reason = "game_concept_event"
				}
				}
			}
			}
	}
}

    }

oe_character_war_score = {#战功
	trigger = {
		root = {
			OR = {
				owner = c:BYZ
				owner = {
					is_subject_type = oe_empire_theme
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				root = {
					NOT = {
						has_variable = oe_character_war_score_var
					}
				}
			}
			root = {
			set_variable = {
				name = oe_character_war_score_var
				value = 0
			}
		}
		} 
		root = {
			change_variable = {
				name = oe_character_war_score_var
				add = 1
			}
		}
		if = {
			limit = {
				root = {
					var:oe_character_war_score_var = 1
				}
			}
			root = {
				add_character_modifier = { modifier = oe_character_war_score_modifier_I years = -1 mode = replace }
			}
		}
		if = {
			limit = {
				root = {
					var:oe_character_war_score_var >= 2
					var:oe_character_war_score_var < 5
					NOT = { has_character_modifier = oe_character_war_score_modifier_II }
				}
			}
			root = {
				add_character_modifier = { modifier = oe_character_war_score_modifier_II years = -1 mode = replace }
				remove_character_modifier = oe_character_war_score_modifier_I
			}
		}
		if = {
			limit = {
				root = {
					var:oe_character_war_score_var >= 5
					var:oe_character_war_score_var < 15
					NOT = { has_character_modifier = oe_character_war_score_modifier_III }
				}
			}
			root = {
				add_character_modifier = { modifier = oe_character_war_score_modifier_III years = -1 mode = replace }
				remove_character_modifier = oe_character_war_score_modifier_II
			}
		}
		if = {
			limit = {
				root = {
					var:oe_character_war_score_var >= 15
					var:oe_character_war_score_var < 25
					NOT = { has_character_modifier = oe_character_war_score_modifier_IV }
				}
			}
			root = {
				add_character_modifier = { modifier = oe_character_war_score_modifier_IV years = -1 mode = replace }
				remove_character_modifier = oe_character_war_score_modifier_III
			}
		}
		if = {
			limit = {
				root = {
					var:oe_character_war_score_var >= 25
					NOT = { has_character_modifier = oe_character_war_score_modifier_V }
				}
			}
			root = {
				add_character_modifier = { modifier = oe_character_war_score_modifier_V years = -1 mode = replace }
				remove_character_modifier = oe_character_war_score_modifier_IV
			}
		}		
	}
}
