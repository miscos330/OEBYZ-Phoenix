# Vanilla supports multiple on_actions entries for these hardcoded dispatchers.
on_game_start = { on_actions = { oe_pillars_opening_resources oe_pillars_reconcile } }
on_subject_created = { on_actions = { oe_pillars_reconcile } }
on_subject_type_changed = { on_actions = { oe_pillars_reconcile } }
on_transfer_subject = { on_actions = { oe_pillars_reconcile } }
oe_pillars_reconcile = {
    trigger = { current_year >= 1200 }
    effect = {
        international_organization:oe_byz_empire_pillars ?= { oe_pillars_sync_effect = yes }
    }
}

oe_pillars_opening_resources = {
    trigger = { current_year >= 1200 }
    effect = {
        international_organization:oe_byz_empire_pillars ?= {
            oe_pillars_initialize_resources_effect = yes
        }
    }
}
