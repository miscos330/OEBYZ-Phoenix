# One 2% roll per Byzantine month, then one equally weighted eligible petition.
monthly_country_pulse = { on_actions = { oe_noble_petitions_monthly } }
oe_noble_petitions_monthly = {
    trigger = { current_year >= 1200 tag = BYZ is_rebel_country = no OR = { current_month = 2 current_month = 5 current_month = 8 current_month = 11 } oe_start_is_1337 = yes oe_court_actor = yes }
    random_events = {
        chance_to_happen = 6
        chance_of_no_event = 0
        1 = oe_noble_petitions.1
        1 = oe_noble_petitions.2
        1 = oe_noble_petitions.3
    }
}
