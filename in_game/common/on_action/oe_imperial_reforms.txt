monthly_country_pulse = { on_actions = { oe_ir_monastery_monthly oe_ir_court_reform_repair oe_ir_tagmata_policy_unlock } }
oe_ir_monastery_monthly = {
    trigger = { has_estate_privilege = estate_privilege:oe_noble_family_monasteries }
    effect = { add_stability = 0.02 }
}

on_battle_won = { on_actions = { oe_ir_army_battle_won } }
oe_ir_army_battle_won = {
    # Native on_battle_won passes the victorious unit as scope:actor.
    trigger = {
        country_exists = c:BYZ
        exists = scope:actor
        scope:actor = { is_army = yes owner = c:BYZ }
    }
    effect = {
        if = {
            limit = {
                exists = scope:actor.leader
                exists = c:BYZ.ruler
                scope:actor.leader = c:BYZ.ruler
            }
            c:BYZ = { oe_ir_change_army_score = { math = add value = 2 } }
        }
        else = { c:BYZ = { oe_ir_change_army_score = { math = add value = 1 } } }
    }
}

# Repair saves that completed the offices action before its reform upgrade was connected.
# No costs or other action rewards are repeated, and an existing good reform is never re-added.
oe_ir_court_reform_repair = {
    trigger = {
        tag = BYZ is_rebel_country = no
        has_variable = oe_ir_offices_done
        OR = {
            NOT = { has_variable = oe_byz_unlocked_reformed_system }
            has_reform = government_reform:oe_byz_badsystem
            NOT = { has_reform = government_reform:oe_byz_goodsystem }
        }
    }
    effect = { oe_ir_upgrade_court_reform = yes }
}

# Existing monthly country pulse: unlock once for saves that already completed Army Reform II.
# No reform costs/rewards are repeated, and no country/location iteration is added.
oe_ir_tagmata_policy_unlock = {
    trigger = {
        tag = BYZ is_rebel_country = no
        has_variable = oe_ir_army_2_done
        NOT = { has_unlocked_policy_trigger = { type = byz_tagmata_policy_upgraded } }
    }
    effect = { unlock_policy_effect = { type = byz_tagmata_policy_upgraded } }
}
