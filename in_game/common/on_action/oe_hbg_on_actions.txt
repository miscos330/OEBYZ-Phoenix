on_game_started_after_lobby = {
	on_actions = {
		oe_hbg_after_lobby
	}
}

yearly_country_pulse = {
	on_actions = {
		oe_hbg_yearly_pulse
	}
}

on_new_country_formed = {
	on_actions = {
		oe_hbg_on_country_formed
	}
}

oe_hbg_after_lobby = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = oe_hbg_init_done } }
			oe_hbg_init_once = yes
			set_global_variable = { name = oe_hbg_init_done value = 1 }
		}
	}
}

oe_hbg_yearly_pulse = {
	trigger = {
		is_rebel_country = no
		num_locations > 0
		NOT = { global_var:oe_hbg_ran_y ?= current_year }
	}
	effect = {
		oe_hbg_kernel_yearly = yes
	}
}

oe_hbg_on_country_formed = {
	effect = {
		oe_hbg_cleanup_check = yes
	}
}
