# Native root is the country; scope:old_ruler is the deceased ruler.
on_ruler_death = { on_actions = { oe_nf_political_ally_ruler_death } }
oe_nf_political_ally_ruler_death = {
    trigger = {
        current_year >= 1200
        oe_start_is_1337 = yes
        tag = BYZ
        is_rebel_country = no
        exists = international_organization:oe_byz_empire_pillars
    }
    effect = {
        international_organization:oe_byz_empire_pillars ?= { oe_nf_clear_political_ally = yes }
        oe_nf_refresh = yes
    }
}
