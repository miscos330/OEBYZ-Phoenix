# Once per new campaign; no daily or monthly location scan.
on_game_start = {
    on_actions = { oe_14_byzantine_silk_setup }
}
oe_14_byzantine_silk_setup = {
    # fm_patches1 A4 (decision 281): only in a game begun at the 1337 start (oe_start_1337, gen_mainline_lock.py);
    # on_game_start runs its handlers in no fixed order, so while the flag is not set yet: the test oe_ml_mark_start uses.
    trigger = { current_year >= 1200 OR = { oe_start_is_1337 = yes AND = { NOT = { has_global_variable = oe_start_1337 } current_year >= 1337 current_year < 1338 } } }
    effect = {
        location:constantinople = {
            if = {
                limit = { NOT = { has_location_modifier = byz_Constantinople_silk_base } }
                add_location_modifier = {
                    modifier = byz_Constantinople_silk_base
                    days = -1
                    mode = add_and_extend
                }
            }
        }
    }
}
