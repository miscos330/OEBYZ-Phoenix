		current_year <= 1450
		is_rebel_country = no
		num_locations > 0
		NOT = { has_variable = oe_ywp13_ev_cd }
		oe_start_is_1337 = yes
		OR = {
			tag = YUA
			AND = {
				tag = CHI
				NOT = { has_variable = rtr_is_new_china }
				NOT = { has_or_had_tag = MNG }
				NOT = { has_or_had_tag = MCH }
				culture = { has_culture_group = culture_group:mongolian_group }
			}
		}
	}
	effect = {
		if = {
			limit = {
				NOT = { has_variable = oe_ywp13_ev3_done }
				current_year >= 1342
				current_year <= 1352
				owns = location:dadu
			}
			set_variable = { name = oe_ywp13_ev3_done value = 1 }
			set_variable = { name = oe_ywp13_ev_cd value = 1 years = 3 }
			trigger_event_non_silently = oe_ywp13.3
		}
		else_if = {
			limit = {
				NOT = { has_variable = oe_ywp13_ev4_done }
				current_year >= 1370
				NOT = { is_situation_active = situation:red_turban_rebellions }
				owns = location:dadu
				OR = {
					AND = {
						country_exists = c:BYZ
						c:BYZ = { owns = location:constantinople }
					}
					AND = {
						country_exists = c:ROM
						c:ROM = { owns = location:constantinople }
					}
				}
			}
			set_variable = { name = oe_ywp13_ev4_done value = 1 }
			set_variable = { name = oe_ywp13_ev_cd value = 1 years = 3 }
			trigger_event_non_silently = oe_ywp13.4
		}
	}
