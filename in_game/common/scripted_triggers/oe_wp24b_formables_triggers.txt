# Restored Roman Empire (重光罗马帝国)
oe_wp24b_is_rom_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	has_or_had_tag = ROM
}

# Latin Empire (拉丁帝国)
oe_wp24b_is_lat_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	has_or_had_tag = LAT
}

# Imperial Prussian Order (普鲁士帝国军政体系)
oe_wp24b_is_pru_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	OR = {
		has_or_had_tag = PRU
		AND = {
			tag = TEU
			OR = {
				government_type = government_type:monarchy
				government_type = government_type:republic
				current_age_or_later = { age = age_4_reformation }
			}
		}
	}
}

# Restored Great Ming (大明中兴天朝)
oe_wp24b_is_mng_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	has_or_had_tag = MNG
}

# Great Qing Banner Empire (大清八旗帝国)
oe_wp24b_is_mch_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	has_or_had_tag = MCH
}

# Northern Yuan Resurgence (北元中兴大汗国)
oe_wp24b_is_yua_host = {
	oe_main_mod_old_world_event_allowed = yes
	is_rebel_country = no
	OR = {
		has_or_had_tag = YUA
		AND = {
			tag = CHI
			culture ?= {
				OR = {
					has_culture_group = culture_group:mongolian_group
					has_culture_group = culture_group:steppe_group
				}
			}
		}
	}
}

# True when country is any of the 6 Project Phoenix WP24-B flagship formable realms.
oe_wp24b_any_formable_host = {
	OR = {
		oe_wp24b_is_rom_host = yes
		oe_wp24b_is_lat_host = yes
		oe_wp24b_is_pru_host = yes
		oe_wp24b_is_mng_host = yes
		oe_wp24b_is_mch_host = yes
		oe_wp24b_is_yua_host = yes
	}
}
