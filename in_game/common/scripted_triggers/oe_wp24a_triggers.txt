# Country scope: Teutonic Order host (TEU, secularized PRU from TEU, or independent LIV if TEU gone).
oe_wp24a_is_teu_host = {
	is_rebel_country = no
	NOT = { oe_is_fourth_rome_exile_country = yes }
	OR = {
		tag = TEU
		AND = {
			tag = PRU
			has_or_had_tag = TEU
		}
		AND = {
			tag = LIV
			is_subject = no
			NOT = { country_exists = c:TEU }
		}
	}
}

# Country scope: Reformation-era closed lower-bound guard (R31 safe: strictly >= 1517).
oe_wp24a_is_teu_reformation_era = {
	current_year >= 1517
	OR = {
		current_age_or_later = { age = age_4_reformation }
		current_year >= 1525
		religion:lutheran = { is_religion_enabled = yes }
	}
}

# Country scope: Northern Yuan / Great Yuan Mongol host (blocks Ming/Qing/Han New China and pre-1337 starts).
oe_wp24a_is_yuan_host = {
	is_rebel_country = no
	NOT = { oe_is_fourth_rome_exile_country = yes }
	current_year >= 1337
	OR = {
		tag = YUA
		tag = MGE
		tag = OIR
		AND = {
			tag = CHI
			NOT = { has_variable = rtr_is_new_china }
			NOT = { has_or_had_tag = MNG }
			NOT = { has_or_had_tag = MCH }
			culture = { has_culture_group = culture_group:mongolian_group }
		}
	}
}

# Country scope: Steppe retreat or Mongolian heartland anchor for Northern Yuan Comeback Arc.
oe_wp24a_yuan_steppe_anchor = {
	oe_wp24a_is_yuan_host = yes
	OR = {
		NOT = { owns = location:dadu }
		owns = location:karakorum
		owns = location:shangdu
		any_owned_location = { region = region:mongolia_region }
	}
}
