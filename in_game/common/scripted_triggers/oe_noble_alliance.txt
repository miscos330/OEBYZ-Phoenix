# All persistent political alliance state belongs to the Empire Pillars IO.
oe_nf_is_political_ally = {
    trigger_if = {
        limit = { has_variable = oe_nf_dynasty_$slot$ has_variable = oe_nf_political_ally }
        var:oe_nf_political_ally = var:oe_nf_dynasty_$slot$
    }
    trigger_else = { always = no }
}
oe_nf_ally_form_ready = {
    c:BYZ = { oe_court_actor = yes OR = { has_ruler = yes has_regent = yes } legitimacy >= 5 has_any_active_disaster = no }
    NOT = { has_variable = oe_nf_political_ally }
    oe_nf_alliance_has_head = yes
}
oe_nf_pressure_ready = {
    c:BYZ = { oe_court_actor = yes }
    var:oe_pillars_authority >= 2
    NOT = { has_variable = oe_nf_pressure_cooldown }
    oe_nf_alliance_has_head = yes
}
oe_nf_honors_ready = {
    oe_nf_interaction_ready = { slot = $slot$ cost = 0 }
    var:oe_nf_influence_$slot$ > 100
    gold >= oe_nf_honors_price
    NOT = { has_variable = oe_nf_honors_cooldown }
}
oe_nf_ally_support_ready = {
    c:BYZ = { oe_court_actor = yes has_ruler = yes }
    oe_nf_is_political_ally = { slot = $slot$ }
    NOT = { has_variable = oe_nf_ally_support_cooldown }
}
oe_nf_ally_pressure_ready = {
    oe_nf_interaction_ready = { slot = $slot$ cost = 1 }
    oe_nf_is_political_ally = { slot = $slot$ }
    NOT = { has_variable = oe_nf_ally_pressure_cooldown }
    oe_nf_alliance_has_rival = yes
}
oe_nf_alliance_has_head = {
    OR = {
        AND = { oe_nf_head_valid = { slot = 1 }  }
        AND = { oe_nf_head_valid = { slot = 2 }  }
        AND = { oe_nf_head_valid = { slot = 3 }  }
        AND = { oe_nf_head_valid = { slot = 4 }  }
    }
}
oe_nf_alliance_head_target = {
    oe_nf_living = yes
    save_temporary_scope_as = oe_nf_alliance_checked_head
    international_organization:oe_byz_empire_pillars = {
        OR = {
            AND = { oe_nf_alliance_matches_head = { slot = 1 }  }
            AND = { oe_nf_alliance_matches_head = { slot = 2 }  }
            AND = { oe_nf_alliance_matches_head = { slot = 3 }  }
            AND = { oe_nf_alliance_matches_head = { slot = 4 }  }
        }
    }
}
oe_nf_alliance_has_rival = {
    OR = {
        AND = { oe_nf_head_valid = { slot = 1 } var:oe_nf_influence_1 > 100 NOT = { oe_nf_is_political_ally = { slot = 1 } } }
        AND = { oe_nf_head_valid = { slot = 2 } var:oe_nf_influence_2 > 100 NOT = { oe_nf_is_political_ally = { slot = 2 } } }
        AND = { oe_nf_head_valid = { slot = 3 } var:oe_nf_influence_3 > 100 NOT = { oe_nf_is_political_ally = { slot = 3 } } }
        AND = { oe_nf_head_valid = { slot = 4 } var:oe_nf_influence_4 > 100 NOT = { oe_nf_is_political_ally = { slot = 4 } } }
    }
}
oe_nf_alliance_rival_target = {
    oe_nf_living = yes
    save_temporary_scope_as = oe_nf_alliance_checked_head
    international_organization:oe_byz_empire_pillars = {
        OR = {
            AND = { oe_nf_alliance_matches_head = { slot = 1 } var:oe_nf_influence_1 > 100 NOT = { oe_nf_is_political_ally = { slot = 1 } } }
            AND = { oe_nf_alliance_matches_head = { slot = 2 } var:oe_nf_influence_2 > 100 NOT = { oe_nf_is_political_ally = { slot = 2 } } }
            AND = { oe_nf_alliance_matches_head = { slot = 3 } var:oe_nf_influence_3 > 100 NOT = { oe_nf_is_political_ally = { slot = 3 } } }
            AND = { oe_nf_alliance_matches_head = { slot = 4 } var:oe_nf_influence_4 > 100 NOT = { oe_nf_is_political_ally = { slot = 4 } } }
        }
    }
}

oe_nf_alliance_matches_head = {
    trigger_if = {
        limit = { oe_nf_head_valid = { slot = $slot$ } }
        var:oe_nf_head_$slot$ = scope:oe_nf_alliance_checked_head
    }
    trigger_else = { always = no }
}
