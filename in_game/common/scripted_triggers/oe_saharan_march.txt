# Location scope. Only directly owned Byzantine holdings are transferred.
oe_saharan_march_grant_location = {
    is_ownable = yes
    has_owner = yes
    owner = c:BYZ
    OR = { area = area:desert_area area = area:western_sahara_area }
}

oe_saharan_march_allowed = {
    tag = BYZ
    NOT = { has_variable = oe_saharan_march_founded }
    custom_tooltip = {
        text = OE_SAHARAN_MARCH_TERRITORY_REQUIRED
        OR = {
            area:desert_area = { any_location_in_area = { oe_saharan_march_grant_location = yes } }
            area:western_sahara_area = { any_location_in_area = { oe_saharan_march_grant_location = yes } }
        }
    }
}
