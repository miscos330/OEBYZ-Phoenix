# Country scope. Italy has its own planning gate; no Alexandria requirement.
oe_heg_plan_italy_allowed = {
    oe_heg_campaign_access = yes
    NOT = { has_variable = oe_heg_italy_plan_done }
    gold >= 1000
}
oe_heg_italy_action_allowed = {
    oe_heg_campaign_access = yes
    custom_tooltip = { text = OE_HEG_ITALY_PLAN_REQUIRED has_variable = oe_heg_italy_plan_done }
}
# Location scope: already-owned bloc land counts for coverage, but only enemy occupation transfers.
oe_heg_south_italy_covered = {
    OR = {
        owner = { oe_heg_byz_bloc = yes }
        oe_heg_military_occupation = yes
    }
}
oe_heg_drive_out_normans_allowed = {
    oe_heg_italy_action_allowed = yes
    custom_tooltip = {
        text = OE_HEG_SOUTH_ITALY_COVERED
        area:apulia_area = {
            NOT = { any_location_in_area = { is_ownable = yes NOT = { oe_heg_south_italy_covered = yes } } }
        }
        area:basilicata_area = {
            NOT = { any_location_in_area = { is_ownable = yes NOT = { oe_heg_south_italy_covered = yes } } }
        }
    }
    custom_tooltip = {
        text = OE_HEG_SOUTH_ITALY_OCCUPATION
        OR = {
            area:apulia_area = { any_location_in_area = { oe_heg_military_occupation = yes } }
            area:basilicata_area = { any_location_in_area = { oe_heg_military_occupation = yes } }
        }
    }
}
