# BYZ country scope. The native occupation/siege hook supplies the actual commander.
oe_belisarius_command_context = {
    tag = BYZ has_variable = oe_osmaris_varna_resolved
    NOT = { has_variable = oe_osman_expelled }
    oe_osmaris_belisarius_available = yes
    exists = scope:character
    scope:character = character:oe_osmaris_belisarius
    exists = scope:target
}
oe_belisarius_rising_ready = {
    tag = BYZ has_variable = oe_belisarius_fort_victory
    NOT = { has_variable = oe_belisarius_rising_presented }
    NOT = { has_variable = oe_osman_expelled }
}
oe_belisarius_war_in_italy = {
    any_enemy = { any_owned_location = { region = region:italy_region } }
}
oe_belisarius_war_art_ready = {
    oe_court_actor = yes has_variable = oe_belisarius_rising_presented
    NOT = { has_variable = oe_belisarius_war_art_presented }
    NOT = { has_variable = oe_osman_expelled }
    oe_osmaris_belisarius_available = yes
    oe_belisarius_war_in_italy = yes
}
# Location scope. Never transfer peaceful, neutral or friendly territory.
oe_belisarius_italian_enemy_land = {
    NOT = { oe_conquest_is_native_civil_war = { recipient = c:BYZ } }
    is_ownable = yes has_owner = yes region = region:italy_region
    owner = { is_at_war_with = c:BYZ NOT = { oe_heg_byz_bloc = yes } }
}
oe_belisarius_italian_capture_context = {
    oe_belisarius_command_context = yes
    has_variable = oe_belisarius_italian_command
    NOT = { has_variable = oe_belisarius_capture_in_progress }
    scope:target = {
        oe_belisarius_italian_enemy_land = yes
        controller = { oe_heg_byz_bloc = yes }
    }
}
