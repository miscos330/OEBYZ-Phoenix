# Country scope. An owned location is counted once regardless of subject tier.
oe_ir_italy_half_owned = {
    region:italy_region = {
        any_ownable_location_in_region = {
            percent >= 0.5
            exists = owner
            owner = { OR = { this = c:BYZ is_subject_or_below_of = c:BYZ } }
        }
    }
}
oe_italian_theme_creation_unlocked = {
    OR = { has_variable = oe_theme_italian_unlocked has_variable = oe_theme_latin_unlocked }
}
oe_pillars_italian_theme_trigger = {
    country_type = building
    is_subject_of = c:BYZ
    is_subject_type = oe_empire_theme_italian
    has_reform = government_reform:oe_byz_theme_italian_subject
}
oe_pillars_latin_theme_trigger = {
    country_type = building
    is_subject_of = c:BYZ
    is_subject_type = oe_empire_theme_latin

}
