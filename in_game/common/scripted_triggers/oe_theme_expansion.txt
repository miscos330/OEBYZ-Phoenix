# Province scope; only the standard theme creation action uses this range (legacy fallback outside R01-R79).
oe_theme_creation_region_trigger = {
    OR = {
        region = region:anatolia_region
        region = region:balkan_region
        area = area:armenian_highlands_area
        area = area:wallachia_area
        AND = {
            global_var:oe_phoenix_empire ?= { has_variable = oe_create_eastern_themes_done }
            OR = {
                area = area:jazira_area
                area = area:iraq_arabi_area
                area = area:levant_area
            }
        }
        AND = {
            global_var:oe_phoenix_empire ?= { has_variable = oe_create_austrian_themes_done }
            OR = {
                area = area:transdanubia_area
                area = area:upper_austria_area
                area = area:austria_area
                area = area:styria_area
                area = area:salzburg_area
                area = area:carinthia_area
                area = area:tirol_area
            }
        }
        AND = {
            global_var:oe_phoenix_empire ?= { has_variable = oe_create_swiss_themes_done }
            OR = {
                area = area:eastern_switzerland_area
                area = area:central_switzerland_area
                area = area:western_switzerland_area
            }
        }
    }
}

    oe_owned_by_phoenix_empire = yes
    has_variable = oe_theme_picker_country
    var:oe_theme_picker_country = {
        oe_theme_convertible_trigger = yes
        capital = prev
    }
