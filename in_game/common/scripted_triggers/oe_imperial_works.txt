# Country scope; engineering is available from the start, independent of hegemony.
oe_works_actor = {
    tag = BYZ
    is_member_of_international_organization = international_organization:oe_byz_roman_world
}

# Location scope, with the native GenericAction actor saved scope.
oe_works_thrace_food_target = {
    area = area:thrace_area
    owner = scope:actor
    location_rank = location_rank:rural_settlement
    raw_material ?= { is_food = yes }
    NOT = { has_location_modifier = oe_supply_the_capital }
}

oe_works_thrace_food_allowed = {
    oe_works_actor = yes
    gold >= 1000
    custom_tooltip = {
        text = OE_WORKS_FIVE_PROJECTS_TT
        OR = {
            NOT = { has_variable = oe_works_thrace_food_projects }
            var:oe_works_thrace_food_projects < 5
        }
    }
    custom_tooltip = {
        text = OE_WORKS_FOOD_TARGET_TT
        area:thrace_area = { any_location_in_area = { oe_works_thrace_food_target = yes } }
    }
}

oe_works_naval_base_target = {
    has_owner = yes
    owner = scope:actor
    oe_works_naval_base_port = yes
    NOT = { has_variable = oe_works_naval_base_site }
    NOT = {
        any_buildings_in_location = {
            building_type = building_type:oe_imperial_naval_base
            OR = { building_level > 0 building_levels_under_construction > 0 }
        }
    }
}

# Country scope; the allowance is cumulative, never reset by Our Sea.
oe_works_naval_base_slots_available = {
    OR = {
        NOT = { has_variable = oe_works_naval_base_projects }
        var:oe_works_naval_base_projects < 5
        AND = {
            has_variable = oe_ir_sea_done
            var:oe_works_naval_base_projects < 10
        }
    }
}
