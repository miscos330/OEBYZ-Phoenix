    area:georgia_area = {
        any_location_in_area = { is_ownable = yes }
        NOT = { any_location_in_area = {
            is_ownable = yes
            NOT = { AND = { has_owner = yes owner = c:BYZ } }
        } }
    }
}

    is_ownable = yes
    has_owner = yes
    owner = c:BYZ
    OR = {
        province_definition = province_definition:tuapse_province
        AND = { region = region:caucasus_region NOT = { area = area:armenian_highlands_area } }
    }
}

    tag = BYZ
    NOT = { has_variable = oe_caucasian_march_founded }
    custom_tooltip = { text = OE_MARCH_CAUCASIAN_TERRITORY_REQUIRED oe_caucasian_march_land_owned = yes }
}
