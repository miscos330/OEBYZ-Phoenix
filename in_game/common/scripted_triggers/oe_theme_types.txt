# Country scope. Italy is intentionally excluded by the four-type whitelist.
oe_theme_convertible_trigger = {
    country_type = building
    oe_subject_of_phoenix_empire = yes
    exists = capital
    OR = {
        is_subject_type = oe_empire_theme
        is_subject_type = oe_empire_theme_naval
        is_subject_type = oe_empire_theme_administrative
        is_subject_type = oe_empire_theme_frontier
    }
}
oe_theme_military_law_trigger = {
    oe_theme_member_trigger = yes
    exists = international_organization:oe_byz_empire_pillars
    international_organization:oe_byz_empire_pillars = {
        OR = {
            international_organization_has_policy = policy:oe_byz_theme_governor_administration
            international_organization_has_policy = policy:oe_byz_empire_theme_org_law_army_rule_II_I
        }
    }
}
oe_theme_central_law_trigger = {
    oe_theme_member_trigger = yes
    exists = international_organization:oe_byz_empire_pillars
    international_organization:oe_byz_empire_pillars = {
        international_organization_has_policy = policy:oe_byz_empire_theme_org_law_army_rule_II_I
    }
}
# Building scope. The owner here is the building country, not the host country.

    OR = {
        building_type = building_type:oe_byz_theme_system_noble
        building_type = building_type:oe_byz_theme_naval_seat
        building_type = building_type:oe_byz_theme_administrative_seat
        building_type = building_type:oe_byz_theme_frontier_seat
        building_type = building_type:oe_byz_theme_system
        building_type = building_type:oe_byz_theme_naval_post
        building_type = building_type:oe_byz_theme_administrative_post
        building_type = building_type:oe_byz_theme_frontier_post

# Location scope. Used only for selecting/converting already-owned theme buildings.
oe_theme_conversion_location_trigger = {
    oe_owned_by_phoenix_empire = yes
    any_buildings_in_location = {
        owner = scope:oe_theme_converted_country
        oe_theme_building_trigger = yes
    }
}

    country_type = building

    OR = {
        is_subject_type = oe_empire_theme
        is_subject_type = oe_empire_theme_naval

        is_subject_type = oe_empire_theme_frontier

        is_subject_type = oe_empire_theme_latin
