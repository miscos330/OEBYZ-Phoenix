oe_rights_imperial_mint = {
    kept_at_conquest = no
    potential = {
        OR = {
            has_or_had_tag = BYZ
            has_or_had_tag = ROM
            is_subject_or_below_of = c:BYZ
        }
    }
    allow = {
        custom_tooltip = { text = oe_mint_coinage_reformed_tt oe_mint_coinage_reformed = yes }
        scope:target = {
            custom_tooltip = {
                text = oe_imperial_mint_location_tt
                OR = {
                    this = location:thessaloniki
                    AND = {
                        this = location:genoa
                        has_variable = oe_genoa_imperial_mint_unlocked
                    }
                }
            }
            NOR = {
                has_town_rights = town_rights_type:oe_rights_thessaloniki_mint
                has_town_rights = town_rights_type:oe_rights_imperial_mint
            }
        }
    }
    location_modifier = {
        local_jewelry_guild_building_levels = 10
        local_jewelry_output_modifier = 0.5
    }
    country_modifier = { minting_inflation_threshold = 0.02 }
    color = rgb { 196 142 36 }
}
