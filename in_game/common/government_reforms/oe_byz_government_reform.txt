                        international_organization_has_policy = policy:oe_byz_theme_governor_administration

                    }
                }
            }
        }
        scale = { value = oe_theme_law_pay_ratio }
        army_maintenance_efficiency = 8
    }
    country_modifier = {
        potential_trigger = {
            custom_tooltip = {
                text = OE_THEME_MILITARY_LAW_ACTIVE
                is_subject_type = oe_empire_theme
                is_subject_of = c:BYZ

                    OR = {
                        international_organization_has_policy = policy:oe_byz_theme_governor_administration

    country_modifier = {
        potential_trigger = {
            custom_tooltip = {
                text = OE_THEME_CENTRAL_LAW_ACTIVE

	days = 3
