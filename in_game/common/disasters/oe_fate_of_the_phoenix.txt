oe_fate_of_the_phoenix = {
	content_priority = 700
	image = "gfx/interface/illustrations/events/oe_ev_byz_siege_walls.dds"
	
	monthly_spawn_chance = 1

	can_start = {
		tag = BYZ
		NOT = { has_variable = oe_ely_is_elysia }
		NOT = { has_variable = oe_ely_voyager }
		NOT = { has_global_variable = oe_ely_settled }
		current_age = age_1_traditions
		
		current_year >= 1337
		current_year < 1338
	}
	
	can_end = {
		OR = {
			has_variable = oe_ely_is_elysia
			oe_phoenix_disaster_can_end = yes
		}
	}

	modifier = {
		mercenary_maintenance_efficiency = 0.5
		
# PAUSED 2026-09-22: custom bureaucracy/corruption subsystem.
# global_bureaucracy_maintenance_cost_modifier = 0.1

		casus_belli_creation_speed_modifier = -0.2
	}
	
	on_start = {
		if = {
			limit = {
				is_situation_active = situation:rise_of_the_ottomans
				situation:rise_of_the_ottomans = {
					has_variable = strongest_beylik_variable
					exists = var:strongest_beylik_variable.ruler_or_regent
				}
			}
			set_variable = {
				name = strongest_belik_variable
				value = situation:rise_of_the_ottomans.var:strongest_beylik_variable.ruler_or_regent
			}
		}
        c:BYZ = { trigger_event_non_silently = oe_fate_of_the_phoenix.21 }
    }

	on_end = {
        oe_phoenix_auto_bandit_stop = yes
		every_owned_location = {
			limit = { has_building = building_type:oe_byz_bad_thief_greek }
			destroy_building = "building(building_type:oe_byz_bad_thief_greek|c:BYZ)"
		}
		if = {
			limit = { has_variable = strongest_belik_variable }
			remove_variable = strongest_belik_variable
		}
		if = {
			limit = { NOT = { has_variable = oe_ely_is_elysia } }
			set_variable = won_fate_of_the_phoenix_disaster_variable
			trigger_event_non_silently = oe_fate_of_the_phoenix.1000
			oe_byz_check_nirvana_complete = yes
		}
	}
	
	on_monthly = {
		if = {
			limit = {
				is_situation_active = situation:rise_of_the_ottomans
				situation:rise_of_the_ottomans = {
					has_variable = strongest_beylik_variable
					exists = var:strongest_beylik_variable.ruler_or_regent
				}
			}
			set_variable = {
				name = strongest_belik_variable
				value = situation:rise_of_the_ottomans.var:strongest_beylik_variable.ruler_or_regent
			}
		}
        hidden_effect = {
            # Fixed denominator: unavailable categories become empty outcomes.
            # One roll, at most one raid/month; each eligible category stays at 1%.
            random_list = {
                96 = {}
                1 = { if = { limit = { oe_raid_ser_available = yes } trigger_event_non_silently = oe_fate_of_the_phoenix.201 } }
                1 = { if = { limit = { oe_raid_turk_available = yes } trigger_event_non_silently = oe_fate_of_the_phoenix.200 } }
                1 = { if = { limit = { oe_raid_pirate_available = yes } trigger_event_non_silently = oe_fate_of_the_phoenix.203 } }
                1 = { if = { limit = { oe_raid_bandit_available = yes } trigger_event_non_silently = oe_fate_of_the_phoenix.204 } }
            }
        }
        hidden_effect = { oe_phoenix_auto_bandit_monthly = yes }
    }
}
