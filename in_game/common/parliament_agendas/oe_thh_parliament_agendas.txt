# Confirm Commandery Chapel Exemptions (clergy_estate)
oe_thh_pa_clergy_chapter_tithe = {
	type = country
	estate = clergy_estate
	importance = 12

	potential = {
		oe_thh_is_generalkapitel_eligible = yes
		country_has_estate = estate_type:clergy_estate
		NOT = { has_country_modifier = oe_thh_mod_pa_chapter_tithe }
	}

	on_accept = {
		add_country_modifier = { modifier = oe_thh_mod_pa_chapter_tithe years = 5 mode = replace }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_bonus }
	}

	can_bribe = {
		can_bribe_default_trigger = yes
	}

	on_bribe = {
		on_bribe_default_effect = yes
	}

	ai_will_do = {
		add = 20
		subtract = {
			if = {
				limit = {
					estate_satisfaction = { estate_type = scope:target value > 0.6 }
				}
				value = 12
			}
		}
	}

	chance = 10
}
