# Ostsiedlung Land Charter (《东向拓垦授田宪章》)
oe_thh_pi_ostsiedlung_charter = {
	type = country
	estate = peasants_estate

	modifier_when_in_debate = {
		global_population_growth = 0.0015
	}

	potential = {
		OR = {
			parliament_type = parliament_type:oe_teu_generalkapitel
			oe_thh_is_generalkapitel_eligible = yes
		}
	}

	allow = {
		NOT = { has_country_modifier = oe_thh_mod_pi_ostsiedlung }
	}

	chance = {
		add = 8
		add = {
			if = {
				limit = { parliament_type = parliament_type:oe_teu_generalkapitel }
				value = 5
			}
		}
		add = {
			if = {
				limit = { at_war = no }
				value = 3
			}
		}
	}

	wants_this_parliament_issue_bias = {
		add = 5
		add = {
			if = {
				limit = { at_war = no }
				value = 10
			}
		}
	}

	on_debate_passed = {
		add_country_modifier = { modifier = oe_thh_mod_pi_ostsiedlung years = 12 mode = replace }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}
