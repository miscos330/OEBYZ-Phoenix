oe_v59_sys_pi_maritime_charter = {
	type = country
	estate = burghers_estate

	modifier_when_in_debate = {
		trade_sea_efficiency = 0.06
	}

	potential = {
		current_year >= 1337
		is_rebel_country = no
		any_owned_location = { is_port = yes }
	}

	allow = {
		NOT = { has_country_modifier = oe_v59_sys_mod_pi_maritime_charter }
	}

	chance = {
		add = 8
		add = {
			if = {
				limit = { any_owned_location = { is_port = yes } }
				value = 4
			}
		}
	}

	wants_this_parliament_issue_bias = {
		add = 5
		add = {
			if = {
				limit = { any_owned_location = { is_port = yes } }
				value = 8
			}
		}
	}

	on_debate_passed = {
		add_country_modifier = {
			modifier = oe_v59_sys_mod_pi_maritime_charter
			years = 10
			mode = replace
		}
		add_navy_tradition = 4
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}
