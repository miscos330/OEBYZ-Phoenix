# 1. Commission the 8-Peak Anatolian Beacon Chain (Phryktoria)
oe_wp32_dec_anatolian_beacon_chain = {
	decision_category = infrastructure_decisions
	only_once = yes
	only_once_scope = country
	potential = {
		oe_wp32_is_byz_host = yes
		oe_wp32_owns_beacon_corridor = yes
		NOT = { has_variable = oe_wp32_beacon_chain_chartered }
	}
	allow = {
		is_rebel_country = no
		at_war = no
		gold >= 160
		owns = location:constantinople
	}
	ai_will_do = {
		value = 0
		if = { limit = { gold >= 300 at_war = no } add = 30 }
	}
	illustration_tags = { 10 = exterior }
	illustration_scope = { event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:laborers } }
	option = {
		name = oe_wp32_dec_anatolian_beacon_chain.a
		effect = {
			custom_tooltip = oe_v55_perf_main_tt_beacon_chain
			set_variable = { name = oe_wp32_beacon_chain_chartered value = 1 }
			add_gold = -160
			add_country_modifier = { modifier = oe_wp32_mod_beacon_chain_active years = 40 mode = replace }
			hidden_effect = {
				every_owned_location = {
					limit = {
						OR = {
							this = location:tarsus
					this = location:nigde
					this = location:aksaray
					this = location:konya
					this = location:emirdag
					this = location:kutahya
					this = location:iznik
					this = location:izmit
						}
						NOT = { has_building = building_type:oe_wp32_gp_phryktoria_beacon }
						can_build_building = building_type:oe_wp32_gp_phryktoria_beacon
					}
					construct_building = {
						building_type = building_type:oe_wp32_gp_phryktoria_beacon
						owner = root
						instant = yes
						cost_multiplier = 0.25
						cost_multiplier_reason = "oe_wp32_beacon_chain_reason"
					}
				}
			}
			trigger_event_non_silently = { id = oe_wp32_gp.1 days = 2 }
		}
	}
}
