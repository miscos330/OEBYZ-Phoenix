		oe_vic_has_any_active_viceroy = yes

	allow = {
		is_subject = no
		gold >= 75
		NOT = { has_variable = oe_vic_audit_cd }
	}

	illustration_scope = { event_illustration_poptype_effect = { foreground = pop_type:nobles background = pop_type:burghers } }

			add_gold = -75
			set_variable = { name = oe_vic_audit_cd months = 60 }
			add_country_modifier = { modifier = oe_vic_audit_harmonized_mod years = 5 mode = replace }

				limit = { oe_vic_is_viceroyalty_subject = yes }

				oe_vic_sync_self_tier = yes
			}
		}
	}
}
