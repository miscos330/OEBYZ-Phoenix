oe_v59_sys_dec_mp_rihla = {
	decision_category = government_decisions
	only_once = yes
	potential = {
		current_year >= 1337
		current_year < 1550
		is_rebel_country = no
		OR = {
			religion.group = religion_group:muslim
			culture = {
				OR = {
					has_culture_group = culture_group:maghrebi_group
					has_culture_group = culture_group:arabic_group
					has_culture_group = culture_group:arabian_group
				}
			}
		}
		exists = capital
		NOT = { has_global_variable = oe_v59_sys_mp_rihla_ibn_battuta_done }
		NOT = { has_variable = oe_v59_sys_mp_rihla_ibn_battuta_asked }
	}
	allow = {
		gold >= 120
		stability >= 5
		NOT = { has_global_variable = oe_v59_sys_mp_rihla_ibn_battuta_done }
		NOT = { has_variable = oe_v59_sys_mp_rihla_ibn_battuta_asked }
	}
	ai_will_do = {
		value = 0
		if = {
			limit = { gold >= 240 stability >= 10 }
			add = 25
		}
	}
	illustration_tags = { 10 = interior }
	option = {
		name = oe_v59_sys_dec_mp_rihla.a
		ai_chance = { value = 100 }
		effect = {
			set_global_variable = { name = oe_v59_sys_mp_rihla_ibn_battuta_done value = 1 }
			set_variable = { name = oe_v59_sys_mp_rihla_ibn_battuta_asked value = 1 }
			add_gold = -120
			trigger_event_non_silently = oe_v59_sys.1
		}
	}
}
