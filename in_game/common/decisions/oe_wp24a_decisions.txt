	decision_category = government_decisions
	only_once = yes
	only_once_scope = country
	potential = {
		oe_wp24a_holds_oriens = yes
		NOT = { has_variable = oe_wp24a_oriens_chartered }
	}
	allow = {
		is_rebel_country = no
		gold >= 120
		stability >= 5

			owns_or_non_sovereign_subject_owns = location:antioch
			owns_or_non_sovereign_subject_owns = location:damascus
			owns_or_non_sovereign_subject_owns = location:alexandria
		}
	}
	ai_will_do = {
		value = 0
		if = {
			limit = { gold >= 240 stability >= 10 }
			add = 25
		}
	}
	illustration_tags = { 10 = interior }
	illustration_scope = { event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:crown_estate } }
	option = {
		name = oe_wp24a_dec_pref_oriens_charter.a
		effect = {
			set_variable = { name = oe_wp24a_oriens_chartered value = 1 }
			add_gold = -120
			trigger_event_non_silently = oe_wp24a_pref.1
		}
	}
}

# WP12 Deepening — Teutonic Order 1525 Reformation Fork (2 Mutually-Exclusive Decisions)
# =============================================================================

oe_wp24a_dec_teu_secularize_prussia = {
	decision_category = religious_decisions
	only_once = yes
	only_once_scope = country
	potential = {
		oe_wp24a_is_teu_host = yes
		oe_wp24a_is_teu_reformation_era = yes
		NOT = { has_variable = oe_wp24a_teu_branch_secular }
		NOT = { has_variable = oe_wp24a_teu_branch_catholic }
		NOT = { has_variable = oe_teu_counter_reformation_order }
	}
	allow = {
		is_rebel_country = no
		at_war = no
		current_year >= 1517
		OR = {
			current_year >= 1525
			has_variable = oe_teu_secularized_pru
			AND = {
				religion:lutheran = { is_religion_enabled = yes }
				"religion_percentage_in_country(religion:lutheran)" >= 0.10
			}
		}
		OR = {
			owns = location:konigsberg
			owns = location:malbork
			num_locations >= 5
		}
	}
	ai_will_do = {
		value = 0
		if = {
			limit = {
				OR = {
					has_variable = oe_teu_secularized_pru
					AND = {
						current_year >= 1525
						religion:lutheran = { is_religion_enabled = yes }
						"religion_percentage_in_country(religion:lutheran)" >= 0.15
					}
				}
			}
			add = 45
		}
	}
	illustration_tags = { 10 = interior }
	illustration_scope = { event_illustration_estate_effect = { foreground = estate_type:crown_estate background = estate_type:nobles_estate } }
	option = {
		name = oe_wp24a_dec_teu_secularize_prussia.a
		effect = {
			set_variable = { name = oe_wp24a_teu_branch_secular value = 1 }
			set_variable = { name = oe_wp24a_teu_1_done value = 1 }
			trigger_event_non_silently = oe_wp24a_teu.1
		}
	}
}
