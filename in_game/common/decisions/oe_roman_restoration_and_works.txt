oe_restore_anatolia = {
    decision_category = expansion_decisions
    only_once = yes
    only_once_scope = country
    potential = { oe_gw_roman_country = yes NOT = { has_variable = oe_restore_anatolia_done } }
    allow = { oe_restore_anatolia_territory = yes exists = international_organization:oe_byz_empire_pillars }
    ai_will_do = { value = 100 }
    illustration_tags = { 10 = armed 10 = exterior }
    illustration_scope = { event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers } }
    option = { name = oe_restore_anatolia.a ai_chance = { value = 100 } effect = {
        if = { limit = { oe_restore_anatolia_territory = yes NOT = { has_variable = oe_restore_anatolia_done } }
            set_variable = oe_restore_anatolia_done
            add_prestige = 20
            international_organization:oe_byz_empire_pillars ?= { change_variable = { name = oe_pillars_authority add = 10 } }
            add_political_influence = 50
        }
    } }
}
